from pathlib import Path
from collections import defaultdict, Counter
import json, hashlib, math
import numpy as np
from PIL import Image, ImageDraw

out = Path(__file__).resolve().parent
root = out.parents[1]
import sys
job_id=sys.argv[1]
assert job_id in ['rom090','obj901_01']
source=root/'jobber'/job_id/'original_1x.png'
job=json.loads((source.parent/'jobb.json').read_text())
src = np.asarray(Image.open(source).convert('RGB'))
height, width = src.shape[:2]
assert (width,height)==tuple(job["region"][2:])
palette = sorted({tuple(map(int, p)) for row in src for p in row})
assert len(palette)==(41 if job_id=="rom090" else 8)
directed = defaultdict(set)
neighbors = defaultdict(set)
unique = set()

def add(color, a, b):
    directed[color].add((a, b))
    key = tuple(sorted((a, b)))
    if key not in unique:
        unique.add(key)
        neighbors[a].add(b)
        neighbors[b].add(a)

for y in range(height):
    for x in range(width):
        c = tuple(map(int, src[y, x]))
        if y == 0 or not np.array_equal(src[y-1, x], src[y, x]):
            add(c, (x, y), (x+1, y))
        if x == width-1 or not np.array_equal(src[y, x+1], src[y, x]):
            add(c, (x+1, y), (x+1, y+1))
        if y == height-1 or not np.array_equal(src[y+1, x], src[y, x]):
            add(c, (x+1, y+1), (x, y+1))
        if x == 0 or not np.array_equal(src[y, x-1], src[y, x]):
            add(c, (x, y+1), (x, y))

def distance(p, a, b):
    dx, dy = b[0]-a[0], b[1]-a[1]
    if dx == dy == 0:
        return math.hypot(p[0]-a[0], p[1]-a[1])
    t = max(0, min(1, ((p[0]-a[0])*dx+(p[1]-a[1])*dy)/(dx*dx+dy*dy)))
    return math.hypot(p[0]-a[0]-t*dx, p[1]-a[1]-t*dy)

def rdp(points, eps=.8):
    if len(points) <= 2:
        return points
    distances = [distance(p, points[0], points[-1]) for p in points[1:-1]]
    peak = max(distances, default=0)
    if peak <= eps:
        return [points[0], points[-1]]
    i = distances.index(peak)+1
    return rdp(points[:i+1], eps)[:-1]+rdp(points[i:], eps)

def area(points):
    return sum(a[0]*b[1]-b[0]*a[1] for a,b in zip(points, points[1:]+points[:1]))/2

def simplify(points):
    if len(points) <= 5:
        return points
    if points[0] != points[-1]:
        return rdp(points)
    # A closed isolated boundary is split into two arcs; small colored marks
    # remain actual vector polygons rather than collapsing to a diagonal.
    i = max(range(1,len(points)-1), key=lambda i: math.dist(points[0], points[i]))
    result = rdp(points[:i+1])[:-1]+rdp(points[i:])
    if len(set(result)) < 3 or area(result)*area(points) <= 0:
        return points
    return result

unused = set(unique)
chains = []
edge_map = {}

def trace(a, b):
    path = [a, b]
    unused.remove(tuple(sorted((a,b))))
    while len(neighbors[b]) == 2 and b != path[0]:
        nxt = next(p for p in neighbors[b] if p != a)
        key = tuple(sorted((b,nxt)))
        if key not in unused:
            break
        unused.remove(key)
        path.append(nxt)
        a,b = b,nxt
    index = len(chains)
    cleaned = simplify(path)
    chains.append({'source': path, 'vector': cleaned})
    for u,v in zip(path,path[1:]):
        edge_map[(u,v)] = (index,1)
        edge_map[(v,u)] = (index,-1)

for a in sorted(neighbors):
    if len(neighbors[a]) != 2:
        for b in sorted(neighbors[a]):
            if tuple(sorted((a,b))) in unused:
                trace(a,b)
while unused:
    a,b = min(unused)
    trace(a,b)
assert len(edge_map) == 2*len(unique)

def direction(a,b):
    return {(1,0):0,(0,1):1,(-1,0):2,(0,-1):3}[(b[0]-a[0],b[1]-a[1])]

regions = {}
source_loop_count = 0
for color, edges in directed.items():
    pending = set(edges)
    following = defaultdict(set)
    for a,b in edges:
        following[a].add(b)
    loops = []
    while pending:
        start,b = min(pending)
        a = start
        loop = []
        while True:
            edge = (a,b)
            pending.remove(edge)
            loop.append(edge)
            if b == start:
                break
            candidates = [p for p in following[b] if (b,p) in pending]
            assert candidates, (color,b)
            old = direction(a,b)
            # Keep diagonally touching source regions separate at a vertex.
            rank = {1:0,0:1,3:2,2:3}
            nxt = min(candidates, key=lambda p:rank[(direction(b,p)-old)%4])
            a,b = b,nxt
        ids = [edge_map[e][0] for e in loop]
        different = [i for i in range(len(ids)) if ids[i] != ids[i-1]]
        if different:
            i = different[0]
            loop = loop[i:]+loop[:i]
        result = []
        last = None
        for edge in loop:
            chain_id, orientation = edge_map[edge]
            if chain_id == last:
                continue
            points = chains[chain_id]['vector']
            if orientation == -1:
                points = list(reversed(points))
            if result:
                assert result[-1] == points[0], (color,chain_id)
                result.extend(points[1:])
            else:
                result.extend(points)
            last = chain_id
        assert result[0] == result[-1]
        assert len(set(result)) >= 3 and area(result) != 0, (color,result)
        loops.append(result)
        source_loop_count += 1
    regions[color] = loops

# Shared boundaries are simplified ONCE then reused on both sides. This is a
# source-derived planar vector graph, no generated silhouette or pixel mask.
svg = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{width*4}" height="{height*4}" viewBox="0 0 {width} {height}">']
for color, loops in sorted(regions.items()):
    parts = []
    for points in loops:
        parts.append('M '+' L '.join(f'{x},{y}' for x,y in points[:-1])+' Z')
    svg.append(f'<path fill="rgb{color}" fill-rule="evenodd" d="'+ ' '.join(parts)+'"/>')
svg.append('</svg>')
svg_text = '\n'.join(svg)+'\n'
assert '<image' not in svg_text and 'gradient' not in svg_text.lower()
(out/f'{job_id}-motiv.svg').write_text(svg_text)

# Render those same vector paths at 16 samples per original pixel, then
# area-average to 4x. Flat interiors keep the exact source palette; only edges
# contain coverage colors. Rasterizing vector paths does not upscale a bitmap.
samples = 16
# Scanline coverage at sample centers follows SVG even-odd fill exactly.
# Half-open ranges avoid PIL inclusive-endpoint bleeding beyond native bounds.
raster=np.zeros((height*samples,width*samples,3),dtype=np.uint8)
coverage=np.zeros((height*samples,width*samples),dtype=np.uint8)
for color, loops in sorted(regions.items()):
    edges=[(a,b) for points in loops for a,b in zip(points,points[1:]) if a[1]!=b[1]]
    y0=max(0,int(min(min(a[1],b[1]) for a,b in edges)*samples))
    y1=min(height*samples,int(max(max(a[1],b[1]) for a,b in edges)*samples))
    for row in range(y0,y1):
        yy=(row+0.5)/samples
        xx=sorted(a[0]+(yy-a[1])*(b[0]-a[0])/(b[1]-a[1]) for a,b in edges if min(a[1],b[1])<=yy<max(a[1],b[1]))
        assert len(xx)%2==0
        for left,right in zip(xx[::2],xx[1::2]):
            lo=max(0,math.ceil(left*samples-0.5-1e-9));hi=min(width*samples,math.ceil(right*samples-0.5-1e-9))
            raster[row,lo:hi]=color;coverage[row,lo:hi]=1
assert coverage.min()==1,'Vector graph has a sample coverage gap'
render=Image.fromarray(raster)
master = render.resize((width*4,height*4),Image.Resampling.BOX)
master.save(out/f'{job_id}-motiv.png')
canvas=Image.new('RGB',(1536,1024),(128,128,128))
px,py,pw,ph=job['bilde_i_lerret']
assert master.size==(pw,ph)
canvas.paste(master,(px,py))
canvas.save(out/f'{job_id}-kandidat.png')

    'maximum_chain_simplification_distance_source_px':0.8,
    'ordinary_boundary_tolerance_source_px':0.8,
    'render_samples_per_source_pixel':samples,
    'rasterizer':'SVG even-odd scanline fill at sample centers, half-open ranges',
    'output_scale':4,
    'flat_original_colors':True,
    'new_facets':0,
    'bitmap_embedded_in_svg':False,
    'coverage_gaps':0,

    'runtime':False,

}
(out/f'{job_id}-geometri.json').write_text(json.dumps(detail,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({k:v for k,v in detail.items() if k not in ['geometry','source_palette']},ensure_ascii=False))
