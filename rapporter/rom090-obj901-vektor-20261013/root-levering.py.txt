sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

def update_queue(entries, note):
    p = root/'GRAFIKKLISTE.md'
    text = p.read_text()
    for e in entries:
        assert e['klar_for_mottak'] and e['egenkontroll'] != 'FEIL'
        old = '- [ ] `'+e['jobb']+'`'
        assert text.count(old) == 1
        text = text.replace(old, '- [x] `'+e['jobb']+'`')
    text += '\n'+note+'\n'
    ids = [j for j in re.findall(r'^- \[x\] `([^`]+)`', text, re.M) if (root/'jobber'/j/'jobb.json').is_file()]
    metas = {j:json.loads((root/'jobber'/j/'jobb.json').read_text()) for j in ids}
    latest = {}
    for f in sorted((root/'rapporter').glob('bestilling*-20261013/fremdrift.json')):
        for e in json.loads(f.read_text())['leveranser']:
            latest[e['jobb']] = e
    for f in sorted((root/'rapporter').glob('*vektor*-20261013/fremdrift.json')):
        for e in json.loads(f.read_text())['leveranser']:
            latest[e['jobb']] = e
    bgs = [j for j,m in metas.items() if m['type']=='rom']
    objs = [j for j,m in metas.items() if m['type']=='objekt']
    layers = [m for m in metas.values() if m['type']=='lag']
    rooms = {metas[j]['rom'] for j in bgs}
    complete = [v for v in rooms if len([j for j in bgs if metas[j]['rom']==v]) == max(metas[j].get('deler',1) for j in bgs if metas[j]['rom']==v)]
    count = lambda js:sum(latest.get(j,{}).get('egenkontroll')=='SJEKK' for j in js)
    stamp = datetime.now(ZoneInfo('Europe/Oslo')).strftime('%Y-%m-%d %H:%M')
    text = re.sub(r'Oppdatert [^\n]+', 'Oppdatert '+stamp+' (norsk tid)', text, count=1)
    text = re.sub(r'\| Rombakgrunner \| [^|]+ \|', f'| Rombakgrunner | {len(complete)} hele rom levert ({len(bgs)} deler), inkludert {count(bgs)} SJEKK, pilotrom visuelt godkjent |', text, count=1)
    text = re.sub(r'\| Store objektbilder \(nærbilder, kart og store deler\) \| [^|]+ \|', f'| Store objektbilder (nærbilder, kart og store deler) | {len(objs)} levert, inkludert {count(objs)} SJEKK |', text, count=1)
    text = re.sub(r'(\| Små objektbilder i lag [^|]+\|) [^|]+ \|', r'\1 '+str(sum(len(v.get('objekter',[])) for v in layers))+' bilder levert ('+str(len(layers))+' lagjobber) |', text, count=1)
    p.write_text(text)
    return {'grafikkjobber':len(ids),'romdeler':len(bgs),'hele_rom':len(complete),'store_objekter':len(objs),'lagjobber':len(layers),'sma_objekter':sum(len(v.get('objekter',[])) for v in layers)}

def seal_and_push(folder, entries, message):
    for p in list(folder.rglob('*'))+[root/'jobber'/e['jobb']/'notat.md' for e in entries]:
        # Prompt and recipe bytes are provenance, including their exact final newline.
        if p.is_file() and p.suffix == '.md':
            p.write_text('\n'.join(line.rstrip() for line in p.read_text().splitlines()).rstrip()+'\n')
    files = sorted(p for p in folder.rglob('*') if p.is_file() and p.name!='SHA256SUMS.txt')+[root/'jobber'/e['jobb']/name for e in entries for name in ['resultat.png','notat.md']]
    (folder/'SHA256SUMS.txt').write_text(''.join(sha(p)+'  '+str(p.relative_to(root))+'\n' for p in files))
    assert subprocess.check_output(['git','branch','--show-current'],cwd=root,text=True).strip()=='gpt-arbeid'
    assert not subprocess.check_output(['git','diff','--cached','--name-only'],cwd=root,text=True).strip()
    paths = [str(folder.relative_to(root)), 'GRAFIKKLISTE.md','todo.md','log.md']+[str((root/'jobber'/e['jobb']/name).relative_to(root)) for e in entries for name in ['resultat.png','notat.md']]
    subprocess.run(['git','add','--',*paths],cwd=root,check=True)
    subprocess.run(['git','diff','--cached','--check'],cwd=root,check=True)
    subprocess.run(['git','commit','-m',message],cwd=root,check=True,capture_output=True,text=True)
    subprocess.run(['git','push','origin','gpt-arbeid'],cwd=root,check=True,capture_output=True,text=True)
    return subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
