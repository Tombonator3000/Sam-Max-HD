from pathlib import Path
import hashlib,json,runpy,re
from datetime import datetime
from zoneinfo import ZoneInfo
r=Path(__file__).resolve().parent;root=r.parents[1];sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
c=runpy.run_path(str(root/'rapporter/rom090-obj901-vektor-20261013/root-levering.py.txt'),run_name='delivery_helpers')
seal=runpy.run_path(str(root/'rapporter/bestilling051-20261013/root-git-mottak.py.txt'))['seal_and_push']
m=json.loads((r/'fremdrift.json').read_text());s=json.loads((r/'sluttkontroll.json').read_text());assert s['utfall_antall']=={'OK':9,'SJEKK':1,'FEIL':0} and s['figurruter']==137
for e in json.loads((r/'kildehash.json').read_text()):assert sha(root/e['fil'])==e['sha256']
for e in m['leveranser']:assert sha(root/e['resultat'])==e['sha256'] and e['klar_for_mottak'] and not e['runtime_verifisert']
stamp=datetime.now(ZoneInfo('Europe/Oslo')).isoformat(timespec='seconds')
note='Bestilling 53: ti figurark med 137 Sam/Max-ruter levert, ni OK og én dokumentert SJEKK for to tilsiktede halsskjøter. Se `rapporter/bestilling053-20261013/RAPPORT.md`. Arbeidet fortsetter.'
counts=c['update_queue'](m['leveranser'],note)
q=root/'GRAFIKKLISTE.md';qt=q.read_text();ids=[j for j in re.findall(r'^- \[x\] `([^`]+)`',qt,re.M) if (root/'jobber'/j/'jobb.json').is_file()]
metas=[json.loads((root/'jobber'/j/'jobb.json').read_text()) for j in ids];icons=[e for e in metas if e['type']=='ikon'];figures=[e for e in metas if e['type']=='figur']

assert counts['grafikkjobber']==490 and counts['figurruter']==235 and counts['ikonfelt']==46
qt=re.sub(r'(\| Ikonark [^|]+\|) [^|]+ \|',lambda mm:mm.group(1)+' '+str(counts['ikonfelt'])+' felt levert ('+str(counts['ikonark'])+' ark), inkludert 2 SJEKK |',qt,count=1)
qt=re.sub(r'(\| Figurark [^|]+\|) [^|]+ \|',lambda mm:mm.group(1)+' '+str(counts['figurruter'])+' ruter levert ('+str(counts['figurark'])+' ark), pilotstil godkjent |',qt,count=1);q.write_text(qt)
(r/'root-mottak.json').write_text(json.dumps({'dato':stamp,'telling':counts,'root_visuell_motivkontroll':True,'alle_137_figurruter_kontrollert':True,'halsskjoter_SJEKK_beholdt':True,'motorimport':False,'runtime':False,'ny_brukergodkjenning':False},ensure_ascii=False,indent=2)+'\n')
with (root/'todo.md').open('a') as f:f.write('\n- [x] Levert ti jobber i bestilling 53: 71 Sam-ruter og 66 Max-ruter. Én SJEKK for tilsiktede halsskjøter er dokumentert.\n')
with (root/'log.md').open('a') as f:f.write('\n## '+stamp+' (Codex, Astra)\n\n- '+note+'\n- Figurark har ekte RGBA; 71 Sam-ruter er bevist kildeidentiske og gjenbruker nye komponenter. Max-ansikter er korrigert og kontrollert lokalt. Ingen motorimport, spilltest eller ny individuell bildeaksept.\n')
commit=seal(r,m['leveranser'],'GPT: 137 Sam og Max-ruter med kildetro kontroll, bestilling 53')
print(json.dumps({'commit':commit,'telling':counts},ensure_ascii=False))
