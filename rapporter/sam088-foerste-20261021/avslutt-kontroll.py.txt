from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, re, shutil, subprocess

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
C = O / 'mod-sam0883521'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(p, d):
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')

b = read(R / 'grunnlag-produksjon.json')
d = read(R / 'fremdrift.json')
motor = read(R / 'motor-sammenligning.json')
vis = read(R / 'motor-visuell-plan.json')
assert motor['sett'] == 50 and motor['motorlogget_hele_familien'] == 64
assert motor['kontrollantall'] == 1440 and motor['nullkontroller'] == motor['grunnlag_nullkontroller'] == 1436
assert motor['alle100_originalflater_SHA_identiske'] and motor['alle100_utenfor_figurflate'] == 0
assert all(motor[k] for k in ['primaerskript201_kjort', 'avvikskoordinater_og_rgb_par_identiske', 'ingen_nye_feilpiksler', 'avviksinnhold_og_rekkefolge_identiske'])
assert motor['binar_sha256'] == '1754ab80def20fa885817c13a9067ed23c69de6252c83a708e78be8bf87aa708'
# Eksplisitt sett i denne gjennomgangen. De øvrige hele motorbildene er
# sett i sine hele før/etter-par, men ikke åpnet separat i full oppløsning.
full_opplosning = {'costume088_1922', 'costume088_1951', 'costume088_1969'}
for gruppe in ['varianter', 'hele_for_etter', 'hele_motorbilder']:
    for e in vis[gruppe]:
        assert sha(R / e['fil']) == e['sha256']
        e['visuelt_sett'] = gruppe != 'hele_motorbilder' or e['bilde'] in full_opplosning
        if gruppe == 'hele_motorbilder':
            e['sett_i_helt_for_etter_par'] = True
vis.update(visuelt_sett=True, hele_motorbilder_sett_separat=3, hele_for_etter_sett=9,
           hele_varianter_sett=29, tid_utc=datetime.now(timezone.utc).isoformat(),
           vurdering='Opp/ned-faser, fall, bevegelsesmerker, svimmelhetsring, kjegletopp og tomme klammer sett mot hele originalen og faktisk rom. Ingen mistet del, ny bakgrunn eller utilsiktet avklipping observert. Andre aktører og romskript er synlige. Egenkontroll er ikke Tom-/Claude-godkjenning.')
assert len(vis['varianter']) == 8 and len(vis['hele_for_etter']) == 9
assert sum(len(e['ruter']) for e in vis['varianter']) == 29
save(R / 'motor-visuell-plan.json', vis)
motor.update(visuelt_sett=True, hele_varianter_sett=29, hele_motorbilder_sett_separat=3, hele_for_etter_sett=9)
save(R / 'motor-sammenligning.json', motor)
assert len(list((A / 'jobber').glob('fig*/resultat.png'))) == 494
warnings = []
for e in d['leveranser']:
    assert sha(A / e['resultat']) == e['sha256']
    text = (R / (e['jobb'] + '-egenkontroll.txt')).read_text()
    assert 'FEIL' not in text and 'Utfall: SJEKK' in text
    warnings += [dict(jobb=e['jobb'], tekst=line) for line in text.splitlines() if line.startswith('SJEKK costume')]
    e.update(visuelt_kontrollert=True, motorobservert=True)
assert len(warnings) == 11
save(R / 'omrissvarsler-vurdert.json', dict(varsler=warnings, visuelt_vurdert=True,
     begrunnelse='Normalisert arkomriss varierer med hele maleriets myke kanter. Alle elleve er sett mot kilden og mot faktisk motorvisning. De hele motivene, positurene og smådelene er bevart. Native kontroll av hver hele fil bruker uendrede porter; ingen FEIL aksepteres.', CORE=128, IoU=.6, dekning=.8))
auto = read(R / 'automatisk-mottak50.json')
assert auto['ruter_ok'] == 50 and all(auto[k] == 0 for k in ['ruter_sjekk', 'ruter_mangler', 'ruter_teknisk'])
native = read(R / 'egen-native.json')
assert len(native['ruter']) == 50
for e in native['ruter']:
    assert e['egen_alfa'] and not e['originalmaske'] and not e['originalfragmenter']
    assert sha(A / e['prototype']) == sha(C / e['fil']) == e['sha256']
protected = read(R / 'beskyttede-samlet1380.json')
assert len(protected) == 1380 and all(sha(A / f) == h for f, h in protected.items())
assert subprocess.check_output(['git', 'stash', 'list', '--format=%H'], cwd=P, text=True).splitlines() == b['stash']
files = read(R / 'kandidatfilhash.json')
assert len(files) == 21764 and all(sha(C / f) == h for f, h in files.items())
assert {f for f, h in b['filer'].items() if files[f] != h} == {'mod.json'}
assert len(set(files) - set(b['filer'])) == 50
calls = list((R / 'kall').glob('*/verktøyresultat.json'))
assert len(calls) == 42
for p in calls:
    cc = p.parent
    call = read(p)
    args = read(cc / 'faktiske-argumenter.json')
    assert sha(cc / 'raa.png') == call['raa_sha256']
    assert sha(cc / 'faktiske-argumenter.json') == call['argumenter_sha256']
    assert call['referanser'] == {f: sha(Path(f)) for f in args['referenced_image_paths']}
d.update(visuelt_sett=True, motorobservert=True, kilde_og_stilanker_sett=True,
         ark_sjekk=5, ark_ok=0, ark_omrissvarsler=11, native_ok=50,
         alle1380_beskyttede_SHA_like=True, alle_tidligere_filer_bevart=True)
save(R / 'fremdrift.json', d)
save(R / 'sluttkontroll.json', d)
