from pathlib import Path
import json,re,hashlib,csv
from PIL import Image,ImageDraw
import numpy as np

r=Path(__file__).resolve().parent;root=r.parents[1]
q=(root/'GRAFIKKLISTE.md').read_text().split('## Bestilling 35\n',1)[1].split('## Bestilling 36\n',1)[0]
jobs=re.findall(r'^- \[ \] `([^`]+)`',q,re.M);assert len(jobs)==10
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
locks=[];records=[];fields=[];anchors={}
register={(int(e['rom']),int(e['objekt_id']),e['tilstand_hex']):e for e in csv.DictReader((root/'konsept/2026-10-12/objektbilder-959.csv').open())}
contact=Image.new('RGB',(1536,5*286),(190,190,190));d=ImageDraw.Draw(contact)
for k,j in enumerate(jobs):
    folder=root/'jobber'/j;m=json.loads((folder/'jobb.json').read_text());im=Image.open(folder/'original_1x.png').convert('RGB');a=np.asarray(im)
    for name in ['referanse.png','original_1x.png','prompt.txt','jobb.json']:locks.append({'fil':str((folder/name).relative_to(root)),'sha256':sha(folder/name)})
    basefolder=(root/m['bilde_to']).parent;base=Image.open(basefolder/'original_1x.png').convert('RGB');anchors[m['bilde_to']]=sha(root/m['bilde_to'])
    if m['rom']==29:
        bm=json.loads((basefolder/'jobb.json').read_text());room=Image.open(root/'jobber/rom029/original_1x.png').convert('RGB');ox,oy,ow,oh=bm['objekt_i_rom'];room.paste(base,(ox,oy));base=room;anchors['jobber/rom029/resultat.png']=sha(root/'jobber/rom029/resultat.png')
    rx,ry,rw,rh=m['region'];px,py,pw,ph=m['bilde_i_lerret'];mask=np.zeros(a.shape[:2],bool)
    for ob in m['objekter']:
        x,y,w,h=ob['rekt'];x0=max(0,x-rx);y0=max(0,y-ry);x1=min(rw,x-rx+w);y1=min(rh,y-ry+h)
        mask[y0:y1,x0:x1]=True;crop=im.crop((x0,y0,x1,y1));c=np.asarray(crop)
        fields.append({'jobb':j,'objekt':ob['bilde'],'rekt':ob['rekt'],'lokalt_klipp':[x0,y0,x1,y1],'hd_rekt':[px+(x-rx)*4,py+(y-ry)*4,w*4,h*4],'register':register.get((m['rom'],ob['objekt'],ob['tilstand'])),'palett_antall':len(np.unique(c.reshape(-1,3),axis=0)),'kilde_rgb_sha256':hashlib.sha256(crop.tobytes()).hexdigest()})
    diff=np.any(a!=np.asarray(base),axis=2);yy,xx=np.nonzero(diff&~mask)
    records.append({'jobb':j,'metadata':m,'endret_totalt':int(diff.sum()),'endret_utenfor_egen_union':int((diff&~mask).sum()),'bbox_utenfor_union':None if not len(xx) else [int(xx.min()+rx),int(yy.min()+ry),int(xx.max()+rx+1),int(yy.max()+ry+1)]})
    ref=Image.open(folder/'referanse.png').convert('RGB');ref.thumbnail((768,256));x=k%2*768;y=k//2*286;contact.paste(ref,(x,y+20));d.text((x+5,y+5),j,fill=(0,0,0))
contact.save(r/'original-kontaktark.png')
for page in range((len(fields)+7)//8):
    c=Image.new('RGB',(1200,4*266),(190,190,190));dr=ImageDraw.Draw(c)
    for i,e in enumerate(fields[page*8:page*8+8]):
        im=Image.open(root/'jobber'/e['jobb']/'original_1x.png').convert('RGB').crop(e['lokalt_klipp']);im=im.resize((im.width*4,im.height*4),Image.Resampling.NEAREST);im.thumbnail((580,230),Image.Resampling.NEAREST)
        x=i%2*600;y=i//2*266;c.paste(im,(x,y+30));dr.text((x+4,y+5),e['jobb']+' '+e['objekt']+' '+str(e['rekt']),fill=(0,0,0))
    c.save(r/f'a-objektfelt-{page+1}.png')
(r/'kildehash.json').write_text(json.dumps(locks,ensure_ascii=False,indent=2)+'\n')
(r/'a-lagbevis.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
(r/'a-feltbevis.json').write_text(json.dumps(fields,ensure_ascii=False,indent=2)+'\n')
(r/'a-ankerhash.json').write_text(json.dumps(anchors,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'jobber':jobs,'kilder':len(locks),'objektfelt':len(fields),'navn':[(e['jobb'],e['objekt'],e['register']['internt_originalnavn'] if e['register'] else None) for e in fields]},ensure_ascii=False))
