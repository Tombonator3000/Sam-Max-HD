from pathlib import Path
import json, hashlib, subprocess, shutil, zipfile

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
S = A / 'rapporter/sam047-slutt-20261020'
E = A / 'rapporter/sam047-effektretur-20261020'
N = A / 'rapporter/sam251-foerste-20261020'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
save = lambda p, d: p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')
api = lambda p: json.loads(subprocess.check_output(['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/' + p], cwd=P, text=True))
state = read(O / 'bygg-startet.json')
tag, code, mod = (state[k] for k in ['release_tag', 'kode', 'hd_mod'])
manifest = read(R / 'moddfiler-2683.json')
old = read(A / 'rapporter/felles-release-20261020-gang047/moddfiler-2634.json')
assert len(manifest) == 20926 and all(sha(P / 'mods/gpt' / f) == h for f, h in manifest.items())
assert set(old) <= set(manifest) and all(manifest[f] == h for f, h in old.items() if f.endswith('.png'))
assert all(f'costumes/costume164_{n}_hd.png' not in manifest for n in [1943, 1944])
protected = read(N / 'beskyttede-samlet.json')
assert len(protected) == 832 and all(sha(A / f) == h for f, h in protected.items())
base = read(S / 'grunnlag-produksjon.json')
assert subprocess.check_output(['git', 'stash', 'list', '--format=%H'], cwd=P, text=True).splitlines() == base['stash']
assert len(list((A / 'jobber').glob('fig*/resultat.png'))) == 371
motor = read(R / 'linux-kontroll.json')
files = read(O / 'moddfiler-kontroll.json')
assets = read(O / 'assetkontroll.json')
assert motor['native'] == 2683 and motor['nye_native_observert'] == 49
assert motor['runder'] == motor['grunnlagsrunder_samme_binar'] == 7
assert motor['skjermbufferkontroller'] == 464 and motor['nullkontroller'] == 461
assert not motor['nye_eller_oekte_avvik'] and motor['kontroller_fra_figurstart'] == 198
assert not motor['helprove_uten_avvik']
assert files['moddfiler_per_pakke'] == 20925 and files['begge_faktiske_nedlastinger_verifisert']
assert files['begge_modder_byteidentiske'] and files['kode'] == code and files['hd_mod'] == mod
assert len(assets) == 4
for v in assets:
    p = O / v['plattform'] / v['fil']
    assert p.stat().st_size == v['bytes'] and sha(p) == v['sha256']
shots = read(R / 'motorbilder.json')
assert len(shots) == 12 and all(sha(R / v['fil']) == v['sha256'] for v in shots)
assert all(v['utenfor_nye_cels_og_marg'] == 0 for v in read(R / 'pakke-pikselkontroll.json'))
review = read(R / 'visuell-kontroll.json')
assert review['hele_motorbilder_sett'] == 12 and review['foer_etter_sett'] == 3
assert not review['ny_Tom_Claude_godkjenning']
obs = read(R / 'motorobservasjoner49.json')
assert obs['sett'] == len(obs['tegninger']) == 49
assert all(sha(Path(v['fil'])) == v['sha256'] and manifest['costumes/' + n + '_hd.png'] == v['native_sha256'] for n, v in obs['tegninger'].items())
assert read(R / 'rasterkontroll-mot-visuelt-sett.json')['rasteridentiske'] == 49
logs = {}
with zipfile.ZipFile(R / 'linux-motorlogger.zip', 'w', zipfile.ZIP_DEFLATED) as z:
    for p in sorted(O.glob('*/scummvm.log')):
        rel = p.relative_to(O).as_posix()
        z.write(p, rel)
        logs[rel] = sha(p)
assert len(logs) == 14
save(R / 'linux-motorlogger-manifest.json', logs)
save(R / 'bevaringskontroll.json', dict(beskyttede_kilder_SHA_uendret=832, tre_stasher_uendret=True,
     alle_tidligere_filstier_bevart=True, alle_tidligere_png_byteidentiske=True, filer2683=20926,
     figurjobber_med_resultat=371, figur251_kildeklare_ark=5, figur251_nye_resultater=0))
lr, wr = (state['plattform'][p]['run_id'] for p in ['linux', 'windows'])
source = read(S / 'mottak-publisert.json')['kildecommit']
body = f'''Linux og Windows inneholder **2683 native HD-bildelag**, med **49 nye ruter fra seks hele ark** av Sam og Max og effektlagene i gangen utenfor kontorene. Hele denne figursekvensen, kostyme047, har nå74 native ruter. Tidligere grafikk og rettinger er med. Resten av HD-grafikken er fortsatt under arbeid.

Last ned **SamMaxHD-Windows.zip** for Windows eller **SamMaxHD-x86_64.AppImage** for Linux. Spill og tale følger med. Gjør Linux-filen kjørbar og start den. Ctrl+H bytter grafikk, F5 åpner originalmenyen.

Begge faktiske nedlastinger og SHA-filer er kontrollert. Alle 20 925 runtime moddfiler per pakke stemmer med publisert modd og er like mellom plattformene. Linux-pakken er prøvd med inventar, akvarium, cafeteria, 820 bilder fra alkovens skript og tre gangsekvenser. Alle49 nye ruter er observert. Tolv hele pakkebilder og tre før/etter er visuelt sett. Alle49 nye pakkemotorbilder er rasteridentiske med de sette produksjonsbildene.

464 skjermbufferkontroller ga461 null. De tre tvungne gangprøvene har samme kjente160-pikslers overgangsavvik før figurstart som2634-grunnlaget med samme nye binær. Alle198 kontroller fra figurstart er null. Ingen nye eller økte avvik. Helprøven er ikke avviksfri.23 omrissvarsler er visuelt vurdert og beholdt. Normale aktører2/3 er synlige under den tvungne kontrollen;dette er ikke naturlig gjennomspilling. Fig164_04 med1943/1944 er fortsatt originalfallback etter forrige bildeverktøyavvisning og er ikke telt som egen HD.

Full naturlig gjennomspilling, interaktiv Windows-test og ny webtest gjenstår. Tidligere testhoppstans, liten inventarkant i rom35 og Windows SDL-startfeil er fortsatt åpne. Ingen ny Tom-/Claude-stilgodkjenning eller motor-/pipelineendring i denne grafikkrunden.

Kode: {code}
hd-mod: {mod}
Grafikkilde: {source}
Linux: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{lr}
Windows: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{wr}
Kontrollrapport: https://github.com/Tombonator3000/Sam-Max-HD/blob/gpt-arbeid/rapporter/{R.name}/RAPPORT.md

Spillet og grafikken tilhører Disney/Lucasfilm.
'''
(O / 'utgivelse-notat.md').write_text(body)
shutil.copy2(O / 'utgivelse-notat.md', R / 'utgivelse-notat.md')
prior = api('releases?per_page=100')
before = read(O / 'releaser-foer-bygg.json')
assert not any(v['tag_name'] == tag for v in prior)
assert {v['id'] for v in before} <= {v['id'] for v in prior}
aset = lambda v: {(a['id'], a['name'], a['size'], a['digest']) for a in v['assets']}
for v in before:
    if v['tag_name'] not in ['linux', 'windows']:
        assert aset(v) == aset(next(x for x in prior if x['id'] == v['id']))
assert api('git/ref/heads/hd-mod')['object']['sha'] == mod
for D in [O, R]:
    save(D / 'releaser-foer.json', prior)
(R / 'RAPPORT.md').write_text(body + f'\nLinux-binær SHA256 {motor["binar_sha256"]}. Alle{len(prior)} tidligere releaser og vedlegg bevares.371 av1758 figurjobber har resultat,1387 gjenstår pluss returer/katalogfeil/sluttkontroll.832 beskyttede kilder og tre stasher er uendret. Neste fem251-ark er bare kildeklargjort,ingen nye kall eller resultater. Faktisk publisering/Latest kvitteres i publisert.json.\n')
for n in ['start.py', 'hent-status.py', 'hent-pakker.py', 'kontroller-pakker.py', 'klargjor-linux.py', 'kontroller-linux.py', 'vis-pakke.py', 'forbered-release.py']:
    shutil.copy2(O / n, R / (n + '.txt'))
print('Begge faktiske pakker,49 nye,461/464 null og visuell kontroll bekreftet; klar for fellesrelease.', flush=True)
