def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def box128(part):
    a = np.asarray(part)[..., 3] >= 128
    ys, xs = np.nonzero(a)
    if not len(xs):
        return None
    return (int(xs.min()), int(ys.min()), int(xs.max()+1), int(ys.max()+1))

def extract(box):
    p = im.crop(box)
    a = np.asarray(p).copy()
    core = Image.fromarray((a[..., 3] >= 128).astype('uint8') * 255)
    reach = np.asarray(core.filter(ImageFilter.MaxFilter(5))) > 0
    a[..., 3][~reach | (a[..., 3] < 40)] = 0
    a[a[..., 3] == 0, :3] = 0
    p = Image.fromarray(a, 'RGBA')
    b = box128(p)
    native = [b[0] + box[0], b[1] + box[1], b[2] - b[0], b[3] - b[1]]
    return p.crop((max(0,b[0]-4), max(0,b[1]-4), min(p.width,b[2]+4), min(p.height,b[3]+4))), native

def fit(p, w, h):
    b = box128(p)
    sw, sh = b[2]-b[0], b[3]-b[1]
    center = (round(p.width*w/sw), round(p.height*h/sh))
    best = None
    for dw in range(-3,4):
        for dh in range(-3,4):
            size = (max(1,center[0]+dw), max(1,center[1]+dh))
            q = p.convert('RGBa').resize(size, Image.Resampling.LANCZOS).convert('RGBA')
            bb = box128(q)
            score = abs(bb[2]-bb[0]-w)+abs(bb[3]-bb[1]-h)
            rank = (score,abs(dw)+abs(dh))
            if best is None or rank < best[0]:
                best = (rank,q,bb)
    _, q, b = best
    assert (b[2]-b[0],b[3]-b[1]) == (w,h), (b,w,h)
    return q,b

def assembly(cel_list,kind='hd'):
    canvas=Image.new('RGBA',(440,360))
    anchor=(210,320)
    for cel in cel_list:
        c=cels[cel]
        if kind=='original':
            q=original_part(cel);x=anchor[0]+c['rel_x']*4;y=anchor[1]+c['rel_y']*4
        else:
            q,b=parts[cel];r=routes[cel]
            x=anchor[0]+(c['rel_x']+r['synlig'][0]-r['rekt'][0])*4-b[0]
            y=anchor[1]+(c['rel_y']+r['synlig'][1]-r['rekt'][1])*4-b[1]
        canvas.alpha_composite(q,(x,y))
    return canvas
