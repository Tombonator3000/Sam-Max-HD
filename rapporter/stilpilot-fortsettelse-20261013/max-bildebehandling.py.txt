def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def bbox(im):
    a=np.array(im.convert('RGBA'))[:,:,3]
    y,x=np.nonzero(a>=CORE)
    if not len(x): raise ValueError('Ingen alfakjerne')
    return [int(x.min()),int(y.min()),int(x.max()-x.min()+1),int(y.max()-y.min()+1)]
def transformed(source, before, target):
    x,y,w,h=before
    _,_,tw,th=target
    pad=8
    margin=max(12,math.ceil(pad/max(0.1,(tw-2)/w))+2,math.ceil(pad/max(0.1,(th-2)/h))+2)
    crop=source.crop((x-margin,y-margin,x+w+margin,y+h+margin)).convert('RGBa')
    for dw,dh in sorted(((dx,dy) for dx in [0,1,-1,2,-2] for dy in [0,1,-1,2,-2]),key=lambda t:abs(t[0])+abs(t[1])):
        sx,sy=(tw+dw)/w,(th+dh)/h
        sample=(margin-pad/sx,margin-pad/sy,margin+w+pad/sx,margin+h+pad/sy)
        tile=crop.resize((tw+dw+2*pad,th+dh+2*pad),Image.Resampling.LANCZOS,box=sample).convert('RGBA')
        measured=bbox(tile)
        if measured[2:]==[tw,th]:
            return tile,measured,{'skala_x':sx,'skala_y':sy,'avrunding_px':[dw,dh]}
    raise ValueError(f'Maal kan ikke tilpasses: {before} -> {target}')
