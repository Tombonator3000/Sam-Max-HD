    shutil.copy2(generated2, raw2)

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

assert a.size == b.size == (1536, 1024)

# avgrensede overgangsfeltet blandes RGB med en myk smoothstep-vekt.
full = Image.new('RGB', (1952, 800))
full.paste(part1, (0, 0))
full.paste(part2, (offset, 0))
full.paste(part1.crop((0, 0, blend_left, 800)), (0, 0))
left_patch = part1.crop((blend_left, 0, blend_right, 800))
right_patch = part2.crop((blend_left-offset, 0, blend_right-offset, 800))
weights = []
for x in range(blend_width):
    t = x / (blend_width - 1)
    weights.append(round(255 * t*t*(3-2*t)))
mask = Image.new('L', (blend_width, 800))
mask.putdata(weights * 800)
full.paste(Image.composite(right_patch, left_patch, mask), (blend_left, 0))
fullpath = report / 'kontor-helrom.png'

def contact(items, columns, width, height, target):
    gap, label = 16, 42
    rows = (len(items) + columns - 1) // columns
    sheet = Image.new('RGB', (columns*width+(columns+1)*gap, rows*(height+label)+(rows+1)*gap), '#eeeeee')
    draw = ImageDraw.Draw(sheet)
    for i, (title, im) in enumerate(items):
        x = gap + (i % columns) * (width+gap)
        y = gap + (i // columns) * (height+label+gap)
        draw.text((x,y+4), title, fill='#111111', font=font)
        if im.size != (width,height):
            im = im.resize((width,height), Image.Resampling.LANCZOS)
        sheet.paste(im, (x,y+label))
    sheet.save(target)
