def tile(cid,hd):
    r=routes[cid];sx0,sy0,sx1,sy1=r['synlig']
    if hd:return out.crop((64+4*sx0,32+4*sy0,64+4*sx1,32+4*sy1))
    t=original.crop((sx0,sy0,sx1,sy1)).convert('RGBA');a=np.array(t)
    a[:,:,3]=np.where(np.abs(a[:,:,:3].astype(int)-np.array([0,200,200])).sum(axis=2)>24,255,0).astype(np.uint8)
    return Image.fromarray(a).resize((4*(sx1-sx0),4*(sy1-sy0)),Image.Resampling.NEAREST)

def paste_cel(im,cid,hd,anchor):
    r=routes[cid];c=cels[cid];t=tile(cid,hd)
    dx,dy=4*(r['synlig'][0]-r['rekt'][0]),4*(r['synlig'][1]-r['rekt'][1])
    xy=(anchor[0]+4*c['rel_x']+dx,anchor[1]+4*c['rel_y']+dy)
    im.alpha_composite(t,xy)
    return xy

def compose(ids,hd,wh=(300,288),anchor=(150,250)):
    im=Image.new('RGBA',wh,'#f2f3f5')
    for cid in ids:paste_cel(im,cid,hd,anchor)
    return im.convert('RGB')
