from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, shutil, subprocess, zipfile
import numpy as np
from PIL import Image
from smhd.myk import _iou, CORE, GOOD_IOU, GOOD_COVER

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
S = A / 'rapporter' / O.name
E = A / 'rapporter/sam047-effektretur-20261020'
C = O / 'mod-sam047-slutt'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads(p.read_text())
save = lambda p, d: p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')
resolve = lambda f: Path(f) if Path(f).is_absolute() else A / f

protected = read(S / 'beskyttede-samlet.json')
assert len(protected) == 812
assert all(sha(A / f) == h for f, h in protected.items())
base = read(S / 'grunnlag-produksjon.json')
assert subprocess.check_output(['git', 'stash', 'list', '--format=%H'], cwd=P, text=True).splitlines() == base['stash']
files = read(S / 'kandidatfilhash.json')
actual = {p.relative_to(C).as_posix(): sha(p) for p in C.rglob('*') if p.is_file()}
assert actual == files and len(files) == 20926
assert {f for f, h in base['filer'].items() if files[f] != h} == {'mod.json'}
assert sum(f.endswith('.png') for f in files) == 20920
assert read(C / 'mod.json')['costume_cels'] == 2683
assert (CORE, GOOD_IOU, GOOD_COVER) == (128, .6, .8)

native = read(S / 'native-fast4x-samlet.json')
rows = native['ruter']
assert len(rows) == len({v['bilde'] for v in rows}) == 49
assert set(files) - set(base['filer']) == {v['fil'] for v in rows}
for v in rows:
    assert sha(C / v['fil']) == v['sha256']
    assert sha(resolve(v['prototype'])) == v['prototype_sha256']
    a = np.array(Image.open(C / v['fil']).convert('RGBA'))
    proto = np.array(Image.open(resolve(v['prototype'])).convert('RGBA'))
    assert np.array_equal(a[:, :, 3], proto[:, :, 3])
    assert not a[a[:, :, 3] == 0, :3].any()
    assert not ((a[:, :, 0] > 220) & (a[:, :, 1] < 100) & (a[:, :, 2] > 220) & (a[:, :, 3] > 0)).any()
    src = np.array(Image.open(P / 'work/extract/costumes' / (v['bilde'] + '.png')).convert('RGBA'))
    assert hashlib.sha256(src.tobytes()).hexdigest() == v['canonical_rgba_sha256']
    h, w = src.shape[:2]
    assert a.shape[:2] == (h * 4 + 16, w * 4 + 16)
    iou, coverage = _iou(a[:, :, 3] >= CORE, np.repeat(np.repeat(src[:, :, 3] > 0, 4, 0), 4, 1), -8, -8)
    assert iou >= GOOD_IOU and coverage >= GOOD_COVER
    assert v['hard_originalmaske'] is False and v['originalfragment_fallback'] is False
assert sum(bool(v['eksakt_helt_alias']) for v in rows) == 16
native.pop('nye_hele_canonical_malerier', None)
native.update(hele_canonical_kilder_nye_i_native=33, nye_malerier_denne_okten=31,
              tidligere_egne_prototyper_gjenbrukt=2, tidligere_egne_prototyper=[1804, 1991])
save(S / 'native-fast4x-samlet.json', native)

raw_calls = []
for R, expected in [(E, 13), (S, 6)]:
    dirs = sorted(p for p in (R / 'kall').iterdir() if p.is_dir() and (p / 'argumenter.json').exists())
    assert len(dirs) == expected
    for d in dirs:
        args = read(d / 'argumenter.json')
        assert args['transparent_background'] is True
        assert (d / 'prompt-faktisk.txt').read_text().rstrip('\n') == args['prompt'].rstrip('\n')
        assert sha(d / 'raa.png') == read(d / 'raabilde.json')['sha256']
        refs = read(d / 'referansehasher.json')
        refs = list(refs.items()) if isinstance(refs, dict) else [(x['fil'], x['sha256']) for x in refs]
        assert all(sha(resolve(f)) == h for f, h in refs)
        assert all(Path(f).exists() for f in args['referenced_image_paths'])
        raw_calls.append(dict(kall=d.name, rapport=R.name, raabilde_sha256=sha(d / 'raa.png'),
                              argumenter_sha256=sha(d / 'argumenter.json'), referanser=len(refs)))

deliveries = []
warnings = 0
for R in [E, S]:
    progress = read(R / 'fremdrift.json')
    for v in progress['leveranser']:
        assert sha(A / v['resultat']) == v['sha256']
        assert Image.open(A / v['resultat']).size == (1536, 1024)
        route = read(A / v['rutekart'])[v['jobb']]
        assert route['sha256_resultat'] == v['sha256'] and len(route['ruter']) == v['ruter']
        result = (R / (v['jobb'] + '-egenkontroll.txt')).read_text()
        assert 'Utfall: FEIL' not in result
        warnings += sum(line.startswith('SJEKK ') for line in result.splitlines())
        v.update(visuelt_kontrollert=True, native_mottatt=True, motorobservert=True,
                 publisert=False, i_fellespakke=False, godkjent_av_Tom=False, claude_kontroll=False)
        deliveries.append(v)
    n = sum(v['ruter'] for v in progress['leveranser'])
    progress.update(native_mottatt=n, motorobservert=n, visuelt_kontrollert=True,
                    importert=True, publisert=False, i_fellespakke=False)
    save(R / 'fremdrift.json', progress)
