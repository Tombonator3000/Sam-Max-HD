from pathlib import Path
import sys, json, hashlib
import numpy as np
from PIL import Image
from scipy import ndimage as nd

root = Path(__file__).resolve().parents[2]
report = root / 'rapporter/opp-ned-20261016'
name = sys.argv[1]
jobdir = root / 'jobber' / name
assert not (jobdir / 'resultat.png').exists(), 'Bevar eksisterende resultat'
job = json.loads((jobdir / 'jobb.json').read_text())
raw_boxes = json.loads((report / (name+'-rawbokser.json')).read_text())
assert len(raw_boxes) == len(job['ruter'])
output = Image.new('RGBA', tuple(job['lerret']))
maps = []

def bounds(mask):
    y,x = np.nonzero(mask)
    return int(x.min()),int(y.min()),int(x.max())+1,int(y.max())+1

def clean(a, core):
    distance, nearest = nd.distance_transform_edt(~core, return_indices=True)
    keep = distance <= 1.5
    cut = a.copy()
    cut[:,:,3][~keep] = 0
    edge = keep & ~core
    cut[:,:,:3][edge] = a[nearest[0][edge],nearest[1][edge],:3]
    b = bounds(core)
    return Image.fromarray(cut).crop(b), b

for cel, spec in zip(job['ruter'], raw_boxes):
    raw_image = Image.open(report/spec['raw']).convert('RGBA').crop(spec['box'])
    if spec.get('rotate') == 180:
        raw_image = raw_image.transpose(Image.Transpose.ROTATE_180)
    raw = np.asarray(raw_image).copy()
    core = raw[:,:,3] > 220
    if spec['komponenter'] == 2:
        # Two original skin patches cover the eyes independently. Remove the
        # generated red fringe, and place the two new marks independently.
        core &= (raw[:,:,1]>30)&(raw[:,:,0]>raw[:,:,1]*1.2)&(raw[:,:,2]<120)
    labels, count = nd.label(core)
    sizes = np.bincount(labels.ravel());sizes[0]=0
    ids = sorted(np.argsort(sizes)[-spec['komponenter']:].tolist(),key=lambda i:bounds(labels==i)[0])
    assert len(ids) == spec['komponenter'] and all(i and sizes[i]>10 for i in ids)
    x0,y0,x1,y1 = cel['synlig']
    dx,dy = (job['plassering'][0]+x0)*4,(job['plassering'][1]+y0)*4
    if spec['komponenter'] == 1:
        image, box = clean(raw, labels==ids[0])
        image = image.resize(((x1-x0)*4,(y1-y0)*4),Image.Resampling.LANCZOS)
        output.alpha_composite(image,(dx,dy))
        source_boxes=[list(box)]
    else:
        source = np.asarray(Image.open(root.parents[1]/'work/extract/costumes'/(cel['bilde']+'.png')).convert('RGBA'))
        source_mask=source[:,:,3]>0;olab,on=nd.label(source_mask);assert on==2
        oboxes=sorted([bounds(olab==i) for i in range(1,on+1)])
        source_boxes=[]
        for component,tb in zip(ids,oboxes):
            image,box=clean(raw,labels==component)
            image=image.resize(((tb[2]-tb[0])*4,(tb[3]-tb[1])*4),Image.Resampling.LANCZOS)
            if spec.get('hudfarge_fra_hd_hode'):
                # The source patches hide the eyes rather than depict a brown
                # symbol. Match the already painted head's nearby skin color.
                head=next(c for c in job['ruter'] if c['rute']==spec['hudfarge_fra_hd_hode'])
                head_original=Image.open(root.parents[1]/'work/extract/costumes'/(head['bilde']+'.png')).convert('RGBA')
                hb=head_original.getchannel('A').getbbox()
                carrier_x=33 if cel['rute']==385 else 30
                hx=(16+head['synlig'][0]+carrier_x+tb[0]-hb[0])*4
                hy=(8+head['synlig'][1]+56+tb[1]-hb[1])*4
                skin=np.asarray(output.crop((hx-4,hy-4,hx+12,hy+12)))
                valid=(skin[:,:,3]>200)&(skin[:,:,0]>150)&(skin[:,:,1]>70)&(skin[:,:,2]<110)
                assert valid.any()
                color=np.median(skin[:,:,:3][valid],axis=0).astype(np.uint8)
                pixels=np.asarray(image).copy();pixels[:,:,:3]=color
                image=Image.fromarray(pixels)
            output.alpha_composite(image,(dx+tb[0]*4,dy+tb[1]*4));source_boxes.append(list(box))
    actual=output.crop((dx,dy,dx+(x1-x0)*4,dy+(y1-y0)*4)).getchannel('A').point(lambda a:255 if a>128 else 0).getbbox()
    assert actual is not None
    maps.append({'bilde':cel['bilde'],'kostyme':cel['kostyme'],'rute':cel['rute'],
                 'rect':[dx+actual[0],dy+actual[1],actual[2]-actual[0],actual[3]-actual[1]],
                 'raw':spec['raw'],'raw_boks':spec['box'],'komponentbokser':source_boxes})

if name == 'fig204_02':
    # Apply only the requested mouth-palette correction from image_gen's fifth
    # attempt. Preserve every other fitted pixel, avoiding a color shift in
    # the bodies and unaffected heads from the full-sheet edit.
    p = np.asarray(output).copy()
    color = p[:,:,:3].astype(float)
    red = (p[:,:,3]>0)&(color[:,:,0]>60)&(color[:,:,0]>color[:,:,1]*1.6)&((color[:,:,1]-color[:,:,2])<(color[:,:,0]-color[:,:,2])*.3)
    allowed = np.zeros(red.shape, bool)
    for i in [8,16,19]:
        x0,y0,x1,y1 = job['ruter'][i]['synlig']
        allowed[(8+y0)*4:(8+y1)*4,(16+x0)*4:(16+x1)*4] = True
    red &= allowed
    corrected = np.asarray(Image.open(report/'fig204_02-raa05.png').convert('RGBA'))
    assert corrected.shape == p.shape
    dark = (corrected[:,:,3]>220)&(corrected[:,:,:3].max(axis=2)<35)
    distance, nearest = nd.distance_transform_edt(~dark, return_indices=True)
    assert red.any() and distance[red].max()<12
    p[:,:,:3][red] = corrected[nearest[0][red],nearest[1][red],:3]
    output = Image.fromarray(p)
    (report/'fig204_02-palettrens.json').write_text(json.dumps({'endrede_piksler':int(red.sum()),'grunnlag':'fig204_02-raa05.png','bare_redigerte_munnfelter':True},indent=2)+'\n')

output.save(jobdir/'resultat.png')
sha=hashlib.sha256((jobdir/'resultat.png').read_bytes()).hexdigest()
(report/('rutekart-'+name+'.json')).write_text(json.dumps({name:{'sha256_resultat':sha,'ruter':maps}},indent=2)+'\n')
p=json.loads((report/'fremdrift.json').read_text());p['leveranser']=[row for row in p['leveranser'] if row['jobb']!=name]
p['leveranser'].append({'jobb':name,'resultat':'jobber/'+name+'/resultat.png','sha256':sha,'ruter':len(maps),
    'rutekart':'rapporter/opp-ned-20261016/rutekart-'+name+'.json','antall_bildekall':len(list(report.glob(name+'-*prompt*.txt'))),
    'egenkontroll':'ikke kjørt','motoravspilt':False,'importert':False,'godkjent_av_Tom':False})
(report/'fremdrift.json').write_text(json.dumps(p,ensure_ascii=False,indent=2)+'\n')
print(name,sha)
