from pathlib import Path

from scipy import ndimage as nd
import numpy as np,json,hashlib

plan=json.loads((R/'ruteplan-unik.json').read_text())['ruter'];groups={};sources={}

 im.save(R/f'smadeler-kilde-{bi//30+1:02d}.png')
