from pathlib import Path
import hashlib,json,subprocess,sys
from PIL import Image
import numpy as np
r=Path(__file__).resolve().parent;root=r.parents[1]
read=lambda n:json.loads((r/n).read_text());sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
for p,h in read('c-kildehash.json')['filer'].items():assert sha(root/p)==h,p
prov=read('c-proveniens.json');assert len(prov['raw'])==prov['kall']
for c in prov['raw']:
    assert sha(root/c['raw'])==c['raw_sha256']
    assert sha(root/c['prompt'])==c['prompt_sha256']
src={z['rute']:z for z in read('c-rutekart066.json')['ruter']};native=json.loads((root/'konsept/2026-10-12/kostymer-originalmetadata.json').read_text())

    p=root/'jobber'/job;m=json.loads((p/'jobb.json').read_text());z=next(x for x in m['ruter'] if x['rute']==cell)
    a=np.array(Image.open(p/'original_1x.png').convert('RGBA').crop(z['synlig']));mask=np.all(a[:,:,:3]==m['bakgrunn'],axis=2);a[:,:,3]=np.where(mask,0,255);a[mask,:3]=0
    return a

for d in donors:

    assert np.array_equal(source_crop(src[c]['jobb'],c),source_crop(p.parent.name,d['donor_cel']))
    donor=Image.open(p).convert('RGBA').crop(d['donor_box']);component=Image.open(root/d['komponent']).convert('RGBA')

    assert sha(root/d['komponent'])==d['komponent_sha256']
for e in read('c-leveranser.json')['leveranser']:
    j=e['jobb'];p=root/e['resultat'];im=Image.open(p);assert sha(p)==e['sha256'] and im.mode=='RGBA' and im.size==(1536,1024)
    out=Image.new('RGBA',im.size);mapping=read(f'c-{j}-rutekart.json');jm=json.loads((root/'jobber'/j/'jobb.json').read_text())
    for m in mapping:
        c=m['cel'];z=src[c];n=next(n for n in native['66']['cels'] if n['cel']==c);assert z['metadata']==n
        actual=next(x for x in jm['ruter'] if x['rute']==c);assert actual['synlig']==z['synlig'] and actual['rekt']==z['rekt']
        x,y,X,Y=z['synlig'];rx,ry,_,_=z['rekt'];rel=[n['rel_x']+x-rx,n['rel_y']+y-ry,n['rel_x']+X-rx,n['rel_y']+Y-ry]

        assert sha(root/m['komponent'])==m['sha256']==proof[c]['sha256'];assert hashlib.sha256(q.tobytes()).hexdigest()==proof[c]['RGBA_sha256']
        assert q.tobytes()==im.crop(box).tobytes();out.paste(q,box[:2]);records.append({'jobb':j,'cel':c,'rel':rel,'mål':box,'sha256':m['sha256']})
    assert out.tobytes()==im.tobytes()
    run=subprocess.run([sys.executable,str(root/'verktoy/sjekk_leveranse.py'),str(p)],capture_output=True,text=True);status={0:'OK',1:'SJEKK',2:'FEIL'}[run.returncode]
    assert status==e['egenkontroll'] and status!='FEIL'
    (r/f'root-{j}-kontroll.txt').write_text((run.stdout+run.stderr).rstrip()+'\n');deliveries.append({'jobb':j,'ruter':len(mapping),'sha256':sha(p),'egenkontroll':status})
eyes=[]
for e in read('c-oyefelt-bevis.json'):

print('C59:',len(records),'ruter,13 donorpar, metadata og byteidentisk fullpakking;',[(e['jobb'],e['egenkontroll']) for e in deliveries])
