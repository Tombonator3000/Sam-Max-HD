from pathlib import Path
import hashlib,json,runpy,re
from datetime import datetime
from zoneinfo import ZoneInfo
r=Path(__file__).resolve().parent;root=r.parents[1];sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
c=runpy.run_path(str(root/'rapporter/rom090-obj901-vektor-20261013/root-levering.py.txt'),run_name='delivery_helpers')
seal=runpy.run_path(str(root/'rapporter/bestilling051-20261013/root-git-mottak.py.txt'))['seal_and_push']

for e in json.loads((r/'kildehash.json').read_text()):assert sha(root/e['fil'])==e['sha256']
for e in m['leveranser']:assert sha(root/e['resultat'])==e['sha256'] and e['klar_for_mottak'] and not e['runtime_verifisert']
assert len(m['leveranser'])==10 and sum(e['ruter'] for e in m['leveranser'])==115
stamp=datetime.now(ZoneInfo('Europe/Oslo')).isoformat(timespec='seconds')
note='Bestilling59: ti figurark med115 ruter levert; Sam ved kista, Max med rasp/toalettnøkkel og20 kildeidentiske Max-felt. Ni OK, fig066_03 SJEKK for133s omriss. Se `rapporter/bestilling059-20261013/RAPPORT.md`. Arbeidet fortsetter.'
q=root/'GRAFIKKLISTE.md';before=len([j for j in re.findall(r'^- \[x\] `([^`]+)`',q.read_text(),re.M) if (root/'jobber'/j/'jobb.json').is_file()])
counts=c['update_queue'](m['leveranser'],note)
qt=q.read_text();ids=[j for j in re.findall(r'^- \[x\] `([^`]+)`',qt,re.M) if (root/'jobber'/j/'jobb.json').is_file()]
metas=[json.loads((root/'jobber'/j/'jobb.json').read_text()) for j in ids];icons=[e for e in metas if e['type']=='ikon'];figures=[e for e in metas if e['type']=='figur']
counts.update(ikonark=len(icons),ikonfelt=sum(len(e['objekter']) for e in icons),figurark=len(figures),figurruter=sum(len(e['ruter']) for e in figures));assert counts['grafikkjobber']==before+10
qt=re.sub(r'(\| Ikonark [^|]+\|) [^|]+ \|',lambda mm:mm.group(1)+' '+str(counts['ikonfelt'])+' felt levert ('+str(counts['ikonark'])+' ark), inkludert2 SJEKK |',qt,count=1)
qt=re.sub(r'(\| Figurark [^|]+\|) [^|]+ \|',lambda mm:mm.group(1)+' '+str(counts['figurruter'])+' ruter levert ('+str(counts['figurark'])+' ark), pilotstil godkjent |',qt,count=1);q.write_text(qt)
(r/'root-mottak.json').write_text(json.dumps({'dato':stamp,'telling':counts,'nye_jobber':10,'root_visuell_motivkontroll':True,'alle115_ruter_i_detaljark_sett':True,'full_komponentrekonstruksjon_byteidentisk':True,'rel_metadata_alle_og_sammensettinger':True,'motorimport':False,'runtime':False,'ny_brukergodkjenning':False},ensure_ascii=False,indent=2)+'\n')
with (root/'todo.md').open('a') as f:f.write('\n- [x] Levert bestilling59: ti figurark med115 ruter. Ni OK; Max066_03 beholder SJEKK for133s omriss. Øyelokk, donorhoder, kistelokk og kildebeviste rosa sålemerker kontrollert.\n')
with (root/'log.md').open('a') as f:f.write('\n## '+stamp+' (Codex, Astra)\n\n- '+note+'\n-16 native bildekall og33 nye donorutsnitt er dokumentert. Alle faktiske leveranser kan gjenoppbygges byteidentisk fra nye komponenter;63 unike kilde-/kontekstlåser er kontrollert. Ingen motorimport, spilltest eller ny individuell bildeaksept.\n')
for p in r.glob('*kontroll*.txt'):p.write_text(p.read_text().rstrip()+'\n')
for p in r.glob('*.md'):p.write_text(p.read_text().replace(' – ','; ').replace(' — ','; '))
commit=seal(r,m['leveranser'],'GPT: 115 ruter for Sam ved kista og Max med rasp, bestilling 59')
