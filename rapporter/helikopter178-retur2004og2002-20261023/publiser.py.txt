read = lambda p: json.loads(p.read_text())

def save(p, q): p.write_text(json.dumps(q, ensure_ascii=False, indent=2) + '\n')
def git(p, *args): return subprocess.check_output(['git', *args], cwd=p, text=True).strip()

def inventory(root):
    files = [p for p in root.rglob('*') if p.is_file() and '.git' not in p.relative_to(root).parts]
    with ThreadPoolExecutor(max_workers=8) as pool:
        return dict(pool.map(lambda p: (p.relative_to(root).as_posix(), sha(p)), files))
def copy(src, dst):
