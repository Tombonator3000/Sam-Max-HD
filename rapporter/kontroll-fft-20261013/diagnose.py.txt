def confidence(a,b):
 h,w=a.shape;win=np.outer(np.hanning(h),np.hanning(w))
 aa=np.fft.fft2((a-a.mean())*win);bb=np.fft.fft2((b-b.mean())*win)
 cross=bb*np.conj(aa);cross/=np.abs(cross)+1e-9
 c=np.fft.ifft2(cross).real;py,px=np.unravel_index(np.argmax(c),c.shape)
 yy,xx=np.indices(c.shape);ddy=np.minimum(abs(yy-py),h-abs(yy-py));ddx=np.minimum(abs(xx-px),w-abs(xx-px))
 side=c[~((ddy<=3)&(ddx<=3))]
 peak=float(c[py,px]);second=float(side.max());sd=float(side.std())
 return {'integer_peak_xy':[int(px),int(py)],'peak':peak,'second_outside_7x7':second,'peak_second_ratio':peak/second if second else None,'psr':(peak-float(side.mean()))/sd if sd else None,'correlation_zero_lag':float(c[0,0])}
def evaluate(o,hd):
 metrics=m.compare(o,hd);status,reasons=m.judge(metrics);eo,es=metrics.pop('eo'),metrics.pop('es')
 h,w=eo.shape;bw,bh=max(48,w//3),max(48,h//2);blocks=[]
 for by in range(0,max(1,h-bh+1),bh):
  for bx in range(0,max(1,w-bw+1),bw):
   a,z=eo[by:by+bh,bx:bx+bw],es[by:by+bh,bx:bx+bw]
   dx,dy=m.phase_shift(a,z)
   blocks.append({'xywh':[bx,by,a.shape[1],a.shape[0]],'shift':[dx,dy],'distance':math.hypot(dx,dy),'edge_std_original':float(a.std()),'edge_std_variant':float(z.std()),'edge_variance_original':float(a.var()),'edge_variance_variant':float(z.var()),'edge_corr':m.corr(a,z),'excluded_by_current_std_gate':bool(a.std()<1e-3 or z.std()<1e-3),**confidence(a,z)})
 return {'status':status,'reasons':reasons,'metrics':metrics,'blocks':blocks}
