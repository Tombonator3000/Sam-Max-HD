def moments(points, pixelvariance):
    center=points.mean(axis=0);cov=np.cov(points,rowvar=False,bias=True) if len(points)>1 else np.zeros((2,2))
    vals,vecs=np.linalg.eigh(cov+np.eye(2)*pixelvariance)
    # Orient the major axis consistently towards positive x.
    if vecs[0,1]<0:vecs[:,1]*=-1
    if np.linalg.det(vecs)<0:vecs[:,0]*=-1
    return center,vals,vecs

def place_moments(canvas,art,source_points,description):
    # Global geometric statistics only: center,2x2 covariance and ink area.
    # No per-pixel source outline,scanline,alpha or silhouette is copied to art.
    im,bbox=art;a=np.array(im);yy,xx=np.nonzero(a[:,:,3]>=128)
    mu,ev,U=moments(np.c_[xx,yy].astype(float),1/12)
    pts=np.array(source_points,float)*4+9.5;target,tv,V=moments(pts,16/12)
    scales=np.sqrt(tv/ev)
    scales[0]*=.9
    trans=V@np.diag(scales)@U.T;inv=np.linalg.inv(trans);offset=mu-inv@target
    z=im.transform(canvas.size,Image.Transform.AFFINE,(inv[0,0],inv[0,1],offset[0],inv[1,0],inv[1,1],offset[1]),Image.Resampling.BICUBIC)
    canvas.alpha_composite(z)
    operations.append(dict(metode='Egen hel komponent,geometrisk affin plassering med senter,2x2-spredning og samlet areal,ingen originalmaske',beskrivelse=description,maalt_senter=target.tolist(),maalt_spredning=tv.tolist(),samlet_areal=len(pts)*16,affin=trans.tolist()))
