from pathlib import Path
from PIL import Image, ImageDraw
from scipy import ndimage as nd
from smhd.myk import CORE, GOOD_IOU, GOOD_COVER, _iou
import numpy as np, json, hashlib, sys, shutil

P=Path('/home/tom/Sam-Max-HD'); A=P/'work/.gpt-gren'; O=Path(__file__).resolve().parent; R=A/'rapporter'/O.name
read=lambda p:json.loads(p.read_text()); sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest(); save=lambda p,d:p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
assert (CORE,GOOD_IOU,GOOD_COVER)==(128,.6,.8)
def bbox(mask):
    y,x=np.nonzero(mask); return [int(x.min()),int(y.min()),int(x.max()+1),int(y.max()+1)]
def clean(a):
    core=a[:,:,3]>=CORE; assert core.any()
    a[nd.distance_transform_edt(~core)>1.5]=0
    trust=a[:,:,3]>=245; assert trust.any()
    _,ix=nd.distance_transform_edt(~trust,return_indices=True)
    edge=(a[:,:,3]>0)&(a[:,:,3]<245); near=a[tuple(ix)][:,:,:3]; a[edge,:3]=near[edge]; a[a[:,:,3]==0,:3]=0
    return a
rows=read(R/'prototyper-forelopig.json') if (R/'prototyper-forelopig.json').exists() else []
for name in sys.argv[1:]:
    C=R/'kall'/name; plan=read(C/'plan.json'); assert len(plan['ruter'])==1
    e=plan['ruter'][0]; raw=Image.open(C/'raa.png').convert('RGBA'); src=np.array(Image.open(e['kilde']).convert('RGBA'))
    assert hashlib.sha256(src.tobytes()).hexdigest()==e['canonical_rgba_sha256']
    own=clean(np.array(raw)); sm=src[:,:,3]>0
    alpha_before=own[:,:,3].copy(); rgb_notes=[]
    if e.get('egne_lose_merker'):
        core=own[:,:,3]>=CORE; labels,n=nd.label(core,np.ones((3,3)))
        biggest=1+int(np.argmax(np.bincount(labels.ravel())[1:]))
        _,ix=nd.distance_transform_edt(~core,return_indices=True)
        expanded=labels[tuple(ix)]
        for label in range(1,n+1):
            if label==biggest: continue
            pix=own[labels==label,:3].astype(int)
            if len(pix)<5: continue
            yellow=(pix[:,0]>160)&(pix[:,1]>150)&(pix[:,2]<160)
            white=np.min(pix,axis=1)>150
            if str(label) in e.get('egen_merke_farge',{}): color=e['egen_merke_farge'][str(label)]
            elif e['egne_lose_merker']=='hvit': color=[255,255,255]
            elif yellow.mean()>.15: color=[239,231,91]
            elif white.mean()>.15: color=[255,255,255]
            else: raise ValueError((name,'Uavklart egen effektfarge',label))
            mask=(expanded==label)&(own[:,:,3]>0); own[mask,:3]=color
            rgb_notes.append(dict(type='egen_los_markering',egen_komponent=label,egen_core_bbox=bbox(labels==label),RGB=color,piksler=int(mask.sum()),egen_maske_sha256=hashlib.sha256(mask.tobytes()).hexdigest(),originalmaske=False))
    for patch in e.get('egne_rgb_flekker',[]):
        x0,y0,x1,y1=patch['egen_rekt']; region=np.zeros(own.shape[:2],bool);region[y0:y1,x0:x1]=True
        rgb=own[:,:,:3].astype(int)
        mask=region&(own[:,:,3]>0)&(rgb.sum(axis=2)<patch['maks_sum'])
        mask=nd.binary_dilation(mask,iterations=1)&region&(own[:,:,3]>0)
        assert mask.any(),(name,patch)
        search=np.zeros(own.shape[:2],bool);search[max(0,y0-18):y1+18,max(0,x0-18):x1+18]=True
        rr,gg,bb=rgb.transpose(2,0,1)
        trust=search&~mask&(own[:,:,3]>=245)&(rr>60)&(gg>rr+5)&(bb>rr+20)
        assert trust.any();_,ix=nd.distance_transform_edt(~trust,return_indices=True)
        near=own[tuple(ix)][:,:,:3];own[mask,:3]=near[mask]
        rgb_notes.append(dict(**patch,type='egen_fargeflekk',piksler=int(mask.sum()),egen_maske_sha256=hashlib.sha256(mask.tobytes()).hexdigest(),originalmaske=False))
    if e.get('egen_bla_kantrens'):
        rgb=own[:,:,:3].astype(int); rr,gg,bb=rgb.transpose(2,0,1)
        mask=(own[:,:,3]>0)&(bb>=70)&(bb>rr+45)&(bb>gg+35)&(gg<bb*.7)
        level=np.minimum(bb,103)
        own[mask,0]=0; own[mask,1]=np.rint(level[mask]*43/103).astype(np.uint8); own[mask,2]=level[mask].astype(np.uint8)
        rgb_notes.append(dict(type='egen_bla_kantfarge',kildepalettindeks=198,RGB_maks=[0,43,103],piksler=int(mask.sum()),egen_RGB_maske_sha256=hashlib.sha256(mask.tobytes()).hexdigest(),originalmaske=False,geometri_endret=False))
    assert np.array_equal(alpha_before,own[:,:,3])
    Image.fromarray(own).save(C/'egen-renset.png')
    full_source_bbox=bbox(sm)
    sx0,sy0,sx1,sy1=e.get('maalegrundlag_aktor_bbox',full_source_bbox); ox0,oy0,ox1,oy1=bbox(own[:,:,3]>=CORE)
    fx=(sx1-sx0)*4/(ox1-ox0); fy=(sy1-sy0)*4/(oy1-oy0)
    assert .88<=fx/fy<=1.12, (name,'For stor ulik skalering',fx,fy)
    out=Image.fromarray(own).transform((src.shape[1]*4+16,src.shape[0]*4+16),Image.Transform.AFFINE,(1/fx,0,ox0-(sx0*4+8)/fx,0,1/fy,oy0-(sy0*4+8)/fy),Image.Resampling.BICUBIC)
    a=clean(np.array(out)); out=Image.fromarray(a)
    iou,cov=_iou(a[:,:,3]>=CORE,np.repeat(np.repeat(sm,4,0),4,1),-8,-8)
    f=C/(e['bilde']+'-prototype.png');out.save(f)
    rec=dict(bilde=e['bilde'],valgt_kall=name,prototype=f.relative_to(A).as_posix(),prototype_sha256=sha(f),canonical_rgba_sha256=e['canonical_rgba_sha256'],snitt_union=iou,dekning=cov,native_ok=iou>=GOOD_IOU and cov>=GOOD_COVER,egen_alfa=True,originalmaske=False,originalfragmenter=False,egen_glodrens_hd=1.5,maalt_kilde_bbox=[sx0,sy0,sx1,sy1],egen_bbox=[ox0,oy0,ox1,oy1],egen_ruteutsnitt=[0,0,*raw.size],egen_hel_affin_plassering=[fx,fy],visuelt_sett=False)
    rec['egne_RGB_rettinger']=rgb_notes;rec['alfa_uendret_ved_RGB_rettinger']=True
    rec['hele_originalens_bbox']=full_source_bbox;rec['porter_mot_hele_originalens_alfa']=True;rec['bekreftet_underlag_piksler']=plan['underlag_piksler']
    rows=[v for v in rows if v['valgt_kall']!=name]+[rec]; save(C/'prototypekontroll.json',[rec])
    board=Image.new('RGBA',(1200,max(440,src.shape[0]*4+62)),'#596978');d=ImageDraw.Draw(board);d.text((12,8),f'{name} {e["bilde"]} original 4x / egen 4x IoU {iou:.3f} dekning {cov:.3f}',fill='white')
    board.alpha_composite(Image.fromarray(src).resize((src.shape[1]*4,src.shape[0]*4),Image.Resampling.NEAREST),(24,48));board.alpha_composite(out,(624,40));board.convert('RGB').save(C/'helkontroll.png')
    print(name,e['bilde'],round(iou,4),round(cov,4),'OK' if rec['native_ok'] else 'FEIL',flush=True)
save(R/'prototyper-forelopig.json',rows);shutil.copy2(__file__,R/'rens-plasser.py.txt')
