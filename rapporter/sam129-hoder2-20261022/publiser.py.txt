from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, shutil, subprocess, tarfile

read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,d):p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
def git(w,*args):return subprocess.check_output(['git',*args],cwd=w,text=True).strip()
def api(s):return json.loads(subprocess.check_output(['gh','api','repos/Tombonator3000/Sam-Max-HD/'+s],cwd=P,text=True))
def remote(branch):
    rows=git(P,'ls-remote','origin','refs/heads/'+branch).splitlines()
    assert len(rows)==1
    return rows[0].split()[0]
def inv(folder):return {p.relative_to(folder).as_posix():sha(p) for p in folder.rglob('*') if p.is_file() and '.git' not in p.relative_to(folder).parts}
