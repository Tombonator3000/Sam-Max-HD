from pathlib import Path
from PIL import Image
from own_parts import decode, component_on_canvas
import hashlib,json,numpy as np,shutil,sys
P=Path('/home/tom/Sam-Max-HD');A=P/'work/.gpt-gren';O=Path(__file__).resolve().parent;R=A/'rapporter'/O.name
right,left=sys.argv[1:3];K=R/'kall'/right;raw=Path(sys.argv[3]);sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_text())
def save(p,q):p.write_text(json.dumps(q,ensure_ascii=False,indent=2)+'\n')
assert not(K/'raa.png').exists()or sha(K/'raa.png')==sha(raw)
shutil.copy2(raw,K/'raa.png');plan=read(K/'plan.json');plan.update(utfort=True,raw_path=str(raw),raa_sha256=sha(raw),omfang='egen høyremodul og sammensatt helsprite');save(K/'plan.json',plan)
meta=read(K/'delplassering-plan.json');basis=Path(meta['egen_basis']);assert sha(basis)==meta['egen_basis_sha256'];base=np.array(Image.open(basis).convert('RGBA'));size=(base.shape[1],base.shape[0]);canvas=Image.new('RGBA',size);rows=[]
for cid in [left,right]:
    C=R/'kall'/cid;m=read(C/'delplassering-plan.json');assert m['egen_basis_sha256']==meta['egen_basis_sha256']
    part,clean=decode(C/'raa.png',read(C/'kodepalett.json')['tabell']);Image.fromarray(part).save(C/'egen-del-renset.png')
    placed,position=component_on_canvas(part,m['egen_basis_core_bbox'],size)
    canvas=Image.alpha_composite(canvas,Image.fromarray(placed));rows.append(dict(kall=cid,raa_sha256=sha(C/'raa.png'),egen_del_renset_sha256=sha(C/'egen-del-renset.png'),**clean,**position))
# Bare et smalt bånd fra vår egen tidligere tegnede hals bevares i skjøten.
# Det er et rett redigeringsvindu i egen tegning, aldri originalens pikselmaske.
band=meta['egen_skjoet_baand'];join=np.zeros_like(base);join[:,band[0]:band[1]]=base[:,band[0]:band[1]]
Image.fromarray(join).save(K/'egen-skjoet.png');canvas=Image.alpha_composite(canvas,Image.fromarray(join));canvas.save(K/'egen-sammensatt.png')
save(K/'egen-del-montering.json',dict(egne_moduler=rows,egen_basis=str(basis),egen_basis_sha256=sha(basis),egen_lerret_maal=list(size),
    egen_skjoet_baand=band,egen_skjoet_sha256=sha(K/'egen-skjoet.png'),gamle_egne_piksler_kun_i_skjoet=True,
    skjoet_valgt_fra_egen_alfa=True,originalfil_apnet_for_sammensetting=False,kildekoordinater_til_plassering=False,
    originalmaske=False,originalfragmenter=False,enkeltrekk_ikke_flyttet=True,sammensatt_sha256=sha(K/'egen-sammensatt.png')))
print('Egne moduler samlet ved egen halsgrense. Kilde ikke åpnet av monteringen.')
