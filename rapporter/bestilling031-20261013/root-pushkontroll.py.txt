from pathlib import Path
import json, hashlib, urllib.request, concurrent.futures

root=Path(__file__).resolve().parents[2]
r=Path(__file__).resolve().parent
checks=[]
for folder,commit in [('bestilling030-20261013','3508d931cfc6f88d3f31aed91b29da0424f75035')]:
    manifest=json.loads((root/'rapporter'/folder/'fremdrift.json').read_text())
    for e in manifest['leveranser']:checks.append((commit,e))
def verify(item):
    commit,e=item
    url='https://raw.githubusercontent.com/Tombonator3000/Sam-Max-HD/'+commit+'/'+e['resultat']
    data=urllib.request.urlopen(url,timeout=60).read()
    digest=hashlib.sha256(data).hexdigest()
    assert digest==e['sha256'],e['jobb']
    return {'jobb':e['jobb'],'commit':commit,'resultat':e['resultat'],'offentlig_sha256':digest,'stemmer':True}
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as pool:results=list(pool.map(verify,checks))
(r/'bestilling030-pushkontroll.json').write_text(json.dumps({'antall':len(results),'alle_stemmer':True,'kontroller':results},ensure_ascii=False,indent=2)+'\n')
print('Bekreftet offentlige PNG-hasher:',len(results))
