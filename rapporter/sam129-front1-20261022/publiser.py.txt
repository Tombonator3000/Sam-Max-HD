from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, shutil, subprocess, tarfile

P = Path('/home/tom/Sam-Max-HD')
A, M = P/'work/.gpt-gren', P/'mods/gpt'
O = Path(__file__).resolve().parent
R = A/'rapporter'/O.name
C, T = O/'mod-sam1293625', O/'publisering-hdmod3625'
H, BACKUP = O/'hentet-hdmod3625', O/'mod-grunnlag3619'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(p, data):
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2)+'\n')
def git(w, *args):
    return subprocess.check_output(['git', *args], cwd=w, text=True).strip()
def api(endpoint):
    return json.loads(subprocess.check_output(['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/'+endpoint], cwd=P, text=True))
def remote(branch):
    rows = git(P, 'ls-remote', 'origin', 'refs/heads/'+branch).splitlines()
    assert len(rows) == 1
    return rows[0].split()[0]
def inv(d):
    return {p.relative_to(d).as_posix():sha(p) for p in d.rglob('*') if p.is_file() and '.git' not in p.relative_to(d).parts}

b, files = read(R/'start.json'), read(R/'kandidatfilhash.json')

assert not any(p.exists() for p in [T, H, BACKUP, R/'publisering-forsokt.json'])
assert git(A, 'branch', '--show-current') == 'gpt-arbeid' and not git(A, 'status', '--porcelain')
art = git(A, 'rev-parse', 'HEAD')

assert remote('hd-mod') == b['hd_mod'] == '2aa8e71e342c7992256cd7600e71050220f64a2b'

assert m['ingen_nye_eller_okte_bufferavvik_i_denne_proven'] and m['utenfor_nye_hoder_og_dokumenterte_kildetilstander'] == 0

assert v['visuelt_sett'] and v['hele_varianter_sett'] == 6 and v['hele_motorbilder_sett'] == 4

assert len(selected) == 6 and {e['bilde'] for e in selected} == {f'costume129_{n}' for n in range(385,391)}
