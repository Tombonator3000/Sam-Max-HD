from pathlib import Path
from datetime import datetime
from zoneinfo import ZoneInfo
from PIL import Image
import hashlib, json, subprocess, shutil

P = Path('/home/tom/Sam-Max-HD')
A, O = P/'work/.gpt-gren', Path(__file__).resolve().parent
R, C = A/'rapporter'/O.name, O/'mod-sam1293625'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(p, data):
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2)+'\n')
start = read(R/'start.json')
selected = read(R/'valgte6.json')
native = read(R/'egen-native.json')
assert len(selected) == len(native['ruter']) == 6
assert len(list(A.glob('jobber/fig*/resultat.png'))) == 504
assert not (A/'jobber/fig129_02/resultat.png').exists()
assert all(sha(A/f) == h for f,h in start['beskyttede'].items())
assert subprocess.check_output(['git','stash','list','--format=%H'],cwd=P,text=True).splitlines() == start['stash']
for e in selected:
    call = R/'kall'/e['valgt_kall']
    args, result = read(call/'faktiske-argumenter.json'), read(call/'verktøyresultat.json')
    assert sha(call/'raa.png') == result['raa_sha256']
    assert result['referanser'] == {p:sha(Path(p)) for p in args['referenced_image_paths']}
    assert e['prototype_sha256'] == sha(A/e['prototype']) == sha(C/'costumes'/(e['bilde']+'_hd.png'))
    assert Image.open(A/e['prototype']).size == (220,344)
    assert e['native_ok'] and e['egen_alfa'] and not e['originalmaske'] and not e['originalfragmenter']
    e.update(visuelt_sett=True,motorobservert=True,godkjent_av_Tom=False,claude_kontroll=False)
save(R/'valgte6.json',selected)
route = read(R/'rutekart-fig129_02.json')
for e in route['fig129_02']['ruter']:
    e.update(prototype_visuelt_sett=True,motorobservert=True,publisert=False)
save(R/'rutekart-fig129_02.json',route)
visual = read(R/'motor-visuell-plan.json')
for group in ['varianter','hele_motorbilder']:
    for e in visual[group]:
        assert sha(R/e['fil']) == e['sha256']
        e['visuelt_sett'] = True
whole = visual['hele_frontanimasjon']
assert whole['alle17_hele_hode_og_kropp'] and len(whole['faser']) == 17
assert sha(R/whole['fil']) == whole['sha256']
whole['visuelt_sett'] = True
visual.update(visuelt_sett=True,hele_varianter_sett=6,hele_motorbilder_sett=4,hele_frontfaser_med_kropp_sett=17)
visual['andre_sette_bilder'] = [dict(fil=f,sha256=sha(R/f),visuelt_sett=True) for f in ['seks-slutthoder-original-native.png','fig129_02-delresultat.png']]
save(R/'motor-visuell-plan.json',visual)
m = read(R/'motor-sammenligning.json')
assert (m['grunnlag_null'],m['kandidat_null'],m['kontroller_pr_side']) == (320,320,320)
assert m['alle_originalaktorverdier_identiske'] and m['torso_utenfor_hoder_uendret']
m.update(visuelt_sett=True,indre_feste_kontrollert=True,palett_kontrollert=True,endelige_nativefiler_omprovd_etter_RGB_rens=True)
save(R/'motor-sammenligning.json',m)
rows = read(R/'motor-pikselkontroll.json')
for e in rows:
    assert sha(Path(e['kandidat_fil'])) == e['kandidat_sha256']
    assert e['native_sha256'] == sha(C/'costumes'/(e['bilde']+'_hd.png'))
    e['visuelt_sett'] = True
save(R/'motor-pikselkontroll.json',rows)
inner = read(R/'indre-feste-sluttkontroll.json')
assert inner['alle_maalte_senteravvik_HD_hoyst3'] and inner['hodet388_lukkede_oyer']
inner['visuelt_sett'] = True
save(R/'indre-feste-sluttkontroll.json',inner)
colors = read(R/'palettrens-kontroll.json')
assert [e['egne_oyefargekomponenter'] for e in colors] == [2,2,2,0,2,2]
progress = read(R/'fremdrift.json')
progress.update(visuelt_sett=6,motorobservert=6)
for e in progress['delvise_leveranser']:
    e.update(visuelt_kontrollert=True,motorobservert=True)
save(R/'fremdrift.json',progress)
save(R/'sluttkontroll.json',dict(tid_norsk=datetime.now(ZoneInfo('Europe/Oslo')).isoformat(),valgte_native=6,holdt_native=0,laget=6,egenkontrollert=6,native_mottatt=6,visuelt_sett=True,motorobservert=6,publisert=0,delvise_leveranser=progress['delvise_leveranser'],hele_ark_levert=0,figurark_med_resultat=504,total_figurark=1758,gjenstar_uten_resultat=1254,delark_egne_6_av23=True,alle_valgte_uten_FEIL=True,hele_arkkontroll_har_FEIL_for14_tomme_ikke_leverte_maal=True,diagnostiske_originalfragmenter_utelatt=['costume129_512','costume129_642','costume129_768'],ingen_beskyttede_endret=True,beskyttede=1420,stasher_bevart=start['stash'],native_totalt=3625,pakker_native=3596,pakker_mangler_native=29,Tom_godkjent=0,Claude_godkjent=0))
save(R/'hjelperstopp.json',dict(beskrivelse='Avgrensede lokale rapport-/analysefeil, ingen bildeverktøyfeil eller motorfeil. Rettet før publisering.',stopp=[dict(type='SyntaxError',arsak='Manglende mellomrom i else 2 og et tidligere hjelpeuttrykk. Ingen kjøring før retting.'),dict(type='palettaudit',arsak='Vid RGB-velger traff to store øyefelt og enkelte egne små fargepiksler utenfor. Bare de to egne sammenhengende øyefeltene korrigeres nå; øvrige råfarger og all alfa beholdes.'),dict(type='for_tidlig_filavlesning',arsak='Konteksthjelper lest før kandidatmotorens nye bilder fantes. Motor ble fullført før ny analyse og endelig audit.')],ingen_motorkodeendring=True,ingen_terskelsenkning=True,omprove_endelige_native_320_null=True,tidligere_filer_bevart_i=str(O/'foerste-palettrens')))
shutil.copy2(__file__,R/'sluttkontroll.py.txt')
print('Egenkontroll6, native6, visuelt6, motor6, publisert0.504 hele jobber uendret.')
