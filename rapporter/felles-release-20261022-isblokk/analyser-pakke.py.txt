def save(path, value):

@lru_cache(maxsize=16)
def picture(path, flat=False):
    path = Path(path)
    if flat:
        path = path.with_name(path.stem + '_flat.png')
    return np.array(Image.open(path).convert('RGB'))
def rectangle(event, padding=8):
    x,y = event['pos']; w,h = event['maal']
    return [max(0,x*4-padding),max(0,y*4-padding),min(1280,(x+w)*4+padding),min(800,(y+h)*4+padding)]
def buffer_details(proof, directory):
    rows=[]
    for check in proof['kontroller']:
        if not check['avvik']:continue
        frame=check['frame']
        screen=np.array(Image.open(directory/f'frame_{frame:06}_screen.png').convert('RGB'))
        files=[p for p in directory.glob(f'frame_{frame:06}_room*.png') if not p.stem.endswith('_flat')]
        assert len(files)==1
        whole=np.array(Image.open(files[0]).convert('RGB'))
        y,x=np.where(np.any(screen!=whole,2));assert len(x)==check['avvik']
        digest=hashlib.sha256(np.stack([x,y],1).tobytes()+screen[y,x].tobytes()+whole[y,x].tobytes()).hexdigest()
        rows.append(dict(frame=frame,avvik=len(x),fil=files[0].name,koordinater_og_begge_RGB_sha256=digest,foer_testaktor296=frame<40))
    return rows
