for i,name in enumerate(selected):
    CC = R/'kall'/name
    e = read(CC/'prototypekontroll.json')[0]
    args, tool, cp = [read(CC/f) for f in ['faktiske-argumenter.json','verktøyresultat.json','plan.json']]
    assert e['native_ok'] and e['egen_alfa'] and not e['originalmaske'] and not e['originalfragmenter']
    assert sha(A/e['prototype'])==e['prototype_sha256'] and sha(CC/'raa.png')==tool['raa_sha256']
    assert tool['argumenter_sha256']==sha(CC/'faktiske-argumenter.json') and tool['prompt_sha256']==sha(CC/'faktisk-prompt.txt')
    assert tool['referanser']=={p:sha(Path(p)) for p in args['referenced_image_paths']}
    assert cp['kilde_og_mastere_sett_foer_kall']

    chosen.append(e)
    src = Image.open(cp['ruter'][0]['kilde']).convert('RGBA')
    own = Image.open(A/e['prototype']).convert('RGBA')
    sa, oa = np.array(src), np.array(own)
    sy,sx = np.where((sa[:,:,3]>0)&np.all(sa[:,:,:3]==[255,199,135],2))
    rr,gg,bb = oa[:,:,:3].transpose(2,0,1).astype(int)
    eye = (oa[:,:,3]>=128)&(rr>=220)&(gg>=170)&(bb>=90)&(bb<235)&(gg>bb+12)
    y,x = np.where(eye)
    delta = [float(x.mean()-8-(sx.mean()*4+1.5)),float(y.mean()-8-(sy.mean()*4+1.5))]
    assert len(sx)==2 and len(x)>0 and max(abs(v) for v in delta)<=3,(name,delta)
    features.append(dict(kall=name,bilde=e['bilde'],kilde_punkt=[[int(x),int(y)] for x,y in zip(sx,sy)],native_lyse_piksler=len(x),native_oyerader_minus8=sorted(set(int(q)-8 for q in y)),forskyvning_senter_HD=delta,innen_eksisterende3_HD_grense=True))
    raw = np.array(Image.open(CC/'egen-renset.png'))
    labs,n = nd.label((raw[:,:,3]>=128)&np.all(raw[:,:,:3]==[255,199,135],2))
    assert n == 2,(name,'Uventet kremfelt',n)
    colors.append(dict(kall=name,bilde=e['bilde'],egne_oyefargekomponenter=n,RGB=[255,199,135],egen_alfa_uendret=True,geometri_endret=False,originalmaske=False))
    xx,yy = i%4*320,i//4*320
    dr.text((xx+5,yy+5),e['bilde']+' original / egen, 8x',fill='white')
    for j,im in enumerate([src.crop(src.getbbox()).resize((src.crop(src.getbbox()).width*4,src.crop(src.getbbox()).height*4),Image.Resampling.NEAREST),own.crop(own.getbbox())]):
        im = im.resize((im.width*2,im.height*2),Image.Resampling.NEAREST)
        visual.alpha_composite(im,(xx+5+j*158,yy+45))

    name = t['bilde']
    if name not in by:
        missing.append(name)
        continue
    e = by[name]
    source = next(e for e in plan['ruter'] if e['bilde']==name)
    src, own = Image.open(source['kilde']).convert('RGBA'), Image.open(A/e['prototype']).convert('RGBA')
    assert sha(Path(source['kilde'])) == source['sha256']
    assert hashlib.sha256(src.tobytes()).hexdigest() == e['canonical_rgba_sha256']
    assert own.size == (src.width*4+16,src.height*4+16)
    core, sm = np.array(own)[:,:,3]>=CORE, np.array(src)[:,:,3]>0

    assert not((np.array(board)[:,:,3]>0)&(np.array(layer)[:,:,3]>0)).any()
    board.alpha_composite(layer)
    yy,xx = np.nonzero(core)

    if new:
        native.append(dict(bilde=name,jobb='fig129_02',canonical=name,fil='costumes/'+name+'_hd.png',prototype=e['prototype'],sha256=e['prototype_sha256'],kall=e['valgt_kall'],snitt_union=iou,dekning=cov,egen_alfa=True,originalmaske=False,originalfragmenter=False))
