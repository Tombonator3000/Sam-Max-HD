from pathlib import Path
from PIL import Image, ImageDraw
import json, hashlib, shutil
import numpy as np

P = Path('/home/tom/Sam-Max-HD')
A, O = P/'work/.gpt-gren', Path(__file__).resolve().parent
R = A/'rapporter'/O.name
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(p, data):
    p.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
details = {
391:'The mouth is an asymmetrical slanting white-and-gray tooth shape across a dark opening. Preserve the diagonal white clusters, mid-gray and dark-gray patches, black ends and red lip outline exactly. This is not a generic horizontal smile. Do not invent individual tooth divisions.',
392:'Small narrow vertical open mouth: black upper opening, dark red interior, a warm skin-colored lower lip and red edge. NO white teeth in this cel.',
393:'Small pursed mouth displaced toward image right: red lips and skin shading with a tiny black slit on its right. NO white teeth. Preserve the crooked red shapes and exact source asymmetry.',
394:'Tall wide-open rectangular mouth: a white upper tooth band, black and dark-red interior, a white lower tooth band, mid-gray accents at the right corners and red lips. Keep both tooth bands at their exact source heights. No individual tooth divisions.',
395:'Small narrow vertical open mouth, slightly lower in this head: black upper opening, dark-red inside and warm skin-colored lower lip surrounded by red. NO white teeth.',
396:'Closed small flat red mouth under the long nose, a red upper stroke and warm skin-shade lower lip. NO black mouth opening and NO white teeth. The black pupils in the eyes stay as in the source.',
397:'Short asymmetrical mouth shifted toward image right: ONE narrow white UPPER tooth band, dark-red and black opening BELOW it and a warm skin-shade lower lip with red border. No lower white tooth band or separate tooth divisions.',
398:'Tall wide-open rectangular mouth: white upper and lower tooth bands, black and dark-red interior and mid-gray corner accents. Copy the red lower lip and short chin join. Preserve the slightly different head height and eye position in this phase.'}
plan = read(R/'kildeplan.json')
for i,n in enumerate(range(391,399),1):
    name = f'T{i:02}'
    C = R/'kall'/name
    assert not C.exists()
    C.mkdir(parents=True)
    e = next(e for e in plan['ruter'] if e['bilde']==f'costume129_{n}')
    src = Image.open(e['kilde']).convert('RGBA')
    assert sha(Path(e['kilde'])) == e['sha256']
    bbox = src.getbbox()
    visible = src.crop(bbox)
    w,h = visible.size
    arr = np.array(visible)
    eyes = np.argwhere((arr[:,:,3]>0)&np.all(arr[:,:,:3]==[255,199,135],2)).tolist()
    assert len(eyes) == 2 and eyes[0][0] == eyes[1][0]
    eye_y = eyes[0][0]
    atlas = Image.new('RGBA',(1024,1024),'#536576')
    factor = min(35,900//w,900//h)
    large = visible.resize((w*factor,h*factor),Image.Resampling.NEAREST)
    atlas.alpha_composite(large,((1024-large.width)//2,(1024-large.height)//2))
    atlas.convert('RGB').save(C/'originalatlas.png')
    src.save(C/'original-alfa.png')
    ctx = e['kontrollert_kontekst']
    assert ctx and sha(Path(ctx['fil'])) == ctx['sha256']
    shutil.copy2(ctx['fil'],C/'spillkontekst.png')
    refs = [C/'originalatlas.png',A/'hovedark/sam.png',A/'hovedark/max.png',C/'spillkontekst.png']
    prompt = f'''Use case: illustration-story.
Asset: ONE complete separate frontal speaking HEAD animation cel {e['bilde']}, Lee-Harvey from Sam & Max Hit the Road HD.
Input roles: Image1 is the ENTIRE visible original head, exactly {w}x{h} original pixels, enlarged with its true aspect ratio. Every original opaque pixel is included. Image2 and Image3 are COMPLETE approved Sam and Max STYLE masters only. Image4 is the actual original game context with this head on its independently drawn torso. SOURCE1 controls the face identity, pose, proportions, colors, expression and every feature position. Do not redraw the scene or borrow Sam/Max features.
Paint just this ENTIRE single head and short skin neck join in straight FRONTAL view, with no body, shoulders, clothes, props or background. Lee-Harvey has a tall asymmetric orange-brown quiff curling toward image left, pink narrow long angular face, long nose, small ears, brown sideburns and small suspicious half-lidded eyes. Copy the original hairstyle, exact hairline and red skin-shadow shapes. No invented facial hair, hair streaks, extra highlights or new face design. Keep the complete visible width/height ratio {w}:{h}.
CRITICAL REGISTRATION, zero-based INSIDE the complete visible {w}x{h} HEAD: hair top is row0; brown brow bars sit on row{eye_y-1}; tiny warm cream eye apertures occupy row{eye_y}, at x{eyes[0][1]} and x{eyes[1][1]}. Therefore the TOP of the eye aperture is EXACTLY {eye_y}/{h} of the full head height. Do not lengthen the forehead or push the eye band downward. Both cream apertures are only ONE original pixel each, warm cream RGB(255,199,135), never enlarged white eyeballs. Preserve the immediately adjacent black pupils. Keep the nose, mouth, ears and chin in their exact source-relative positions too.
Exact mouth phase: {details[n]}
Trace the original at4x. Keep the pose, silhouette, proportions and position. Turn every cluster of pixels into the shape it shows: clean continuous dark hand-drawn contours, flat original colors with one shade. Smooth antialiased own alpha edges, NO pixel stairs, hard original-mask clipping, texture, dithering, gradients, glow, cast shadow,3D, mirroring or rotation. Do not add a heavy black outline along the flat bottom neck join. Do not enlarge the head, eyes, nose, lips or mouth. Mouth and eye asymmetries are deliberate.
Output ONE complete head centered on a1024x1024 genuinely transparent PNG, large and clear, with transparent padding around every edge. No slate background, checkerboard, text, labels, numbers, frame or watermark. The final visible native extent is {w*4}x{h*4} HD pixels; its full native transparent canvas is {src.width*4+16}x{src.height*4+16}, including the original layer's transparent padding and equal8px margin. Paint the whole head anew; do not locally move or paste individual features.'''
    args = dict(prompt=prompt,referenced_image_paths=[str(p) for p in refs],transparent_background=True)
    save(C/'faktiske-argumenter.json',args)
    (C/'faktisk-prompt.txt').write_text(prompt+'\n')
    save(C/'plan.json',dict(ruter=[dict(e,canonical_rgba_sha256=e['rgba_sha256'],egen_kremoyefarge=[255,199,135])],detaljer=details[n],underlag_piksler=0,referanser={str(p):sha(p) for p in refs},kilde_og_mastere_sett_foer_kall=False,originalmaske=False,originalfragmenter=False,selvstendig_hel_hoderute=True,hele_kildelerret_sett=True,alle_originale_opake_piksler_i_forstorret_referanse=True,ingen_tidligere_tegning_som_stilanker=True,kilde_oyer_synlig=eyes))
    board = Image.new('RGB',(1900,1600),'#536576')
    dr = ImageDraw.Draw(board)
    dr.text((10,8),f'{name}, {n}: whole source head, full original layer, actual game context and COMPLETE approved masters',fill='white')
    im = Image.open(C/'originalatlas.png')
    im.thumbnail((650,760))
    board.paste(im,(10,35))
    im = Image.new('RGBA',(src.width*5,src.height*5),'#536576')
    im.alpha_composite(src.resize(im.size,Image.Resampling.NEAREST))
    board.paste(im.convert('RGB'),(660,35))
    im = Image.open(C/'spillkontekst.png').convert('RGB')
    im.thumbnail((980,740))
    board.paste(im,(920,35))
    for k,master in enumerate(['sam','max']):
        im = Image.open(A/'hovedark'/f'{master}.png').convert('RGBA')
        im.thumbnail((910,720))
        board.paste(im,(10+k*950,840),im)
    board.save(C/'kontekst-og-mastere.png')
shutil.copy2(__file__,R/'klargjor.py.txt')
shutil.copy2(P/'out/sam129-front1-20261022/bevar.py',O/'bevar.py')
shutil.copy2(P/'out/sam129-front1-20261022/rens-plasser.py',O/'rens-plasser.py')
print('Åtte hele frontale hoder klargjort. Original og godkjente mastere må ses per kall.')
