"""Plassering/rensing av ny grafikk, uten endring av beskyttede kilder."""
from pathlib import Path
from PIL import Image,ImageDraw
import numpy as np
import json,hashlib,subprocess,io
from scipy import ndimage
ROOT=Path(__file__).resolve().parents[2];R=Path(__file__).resolve().parent
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
BG=np.array([0,177,64.]);manifest=[]

def clean_key(im,key):
 a=np.array(im.convert('RGB'),float);key=np.array(key,float)
 dist=np.linalg.norm(a-key,axis=2);near=dist<70
 lab,_=ndimage.label(near);edge=np.unique(np.concatenate([lab[0],lab[-1],lab[:,0],lab[:,-1]]));flood=np.isin(lab,edge[edge>0])
 alpha=np.full(a.shape[:2],255.);fringe=flood & ndimage.maximum_filter(~flood,size=3);alpha[flood]=0;alpha[fringe]=np.clip((dist[fringe]-12)/58,0,1)*255
 part=(alpha>0)&(alpha<255);f=alpha[part,None]/255;a[part]=np.clip((a[part]-key*(1-f))/f,0,255);a[alpha==0]=0
 return Image.fromarray(np.dstack([a,alpha]).round().astype('uint8'))

def core(im):
 y,x=np.nonzero(np.array(im)[...,3]>=128);assert len(x)
 return int(x.min()),int(y.min()),int(x.max()+1),int(y.max()+1)

def fit(im,box,size):
 # Kun synlig boks, aldri originalkontur, styrer skalering/plassering.
 l,t,r,b=core(im);q=im.crop((max(0,l-2),max(0,t-2),min(im.width,r+2),min(im.height,b+2)))
 l,t,r,b=core(q);left,top,right,bottom=box;w,h=right-left,bottom-top
 target=(round(q.width*w/(r-l)),round(q.height*h/(b-t)))
 best=None
 for dx in range(-2,3):
  for dy in range(-2,3):
   z=q.convert('RGBa').resize((max(1,target[0]+dx),max(1,target[1]+dy)),Image.Resampling.LANCZOS).convert('RGBA');a,c,d,e=core(z);score=abs(d-a-w)+abs(e-c-h)
   if best is None or score<best[0]:best=(score,z,(a,c,d,e))
 assert best[0]<=1,(w,h,best[0]);_,z,(a,c,d,e)=best
 out=Image.new('RGBA',size);out.alpha_composite(z,(left-a,top-c));return out

def baseline(job,meta):
 p=ROOT/'jobber'/job/'resultat.png';backup=R/'for'/f'{job}.png';backup.parent.mkdir(exist_ok=True)
 if backup.exists():return Image.open(backup).convert('RGBA')
 old=subprocess.run(['git','show','6ffca8e8fba91c82efb43a0a26b17dfbd9ec3ab8:jobber/'+job+'/resultat.png'],cwd=ROOT,capture_output=True)
 if old.returncode==0:return Image.open(io.BytesIO(old.stdout)).convert('RGBA')
 return Image.open(ROOT/meta['bilde_to']).convert('RGBA')


raw=Image.open(R/'objektfaser-raa01.png');entries=json.loads((R/'objektfaser.json').read_text());canvases={};records=[]
for e in entries:
 name=e['bilde'];x,y,w,h=e['celle'];size=tuple(e['native_mal']);q=fit(clean_key(raw.crop((x,y,x+w,y+h)),BG),(0,0,*size),size);dest=R/'objects'/f'{name}.png';dest.parent.mkdir(exist_ok=True);q.save(dest)
 job=e['jobb'];meta=json.loads((ROOT/'jobber'/job/'jobb.json').read_text());o=next(o for o in meta['objekter'] if o['bilde']==name);x,y,w,h=o['rekt'];rx,ry,_,_=meta['region'];px,py=meta['plassering'];at=((px+x-rx)*4,(py+y-ry)*4)
 if job not in canvases:canvases[job]=baseline(job,meta)
 layer=q.copy();a=np.array(layer);hh,ww=a.shape[:2];yy,xx=np.mgrid[:hh,:ww];weight=np.minimum.reduce([xx/8,(ww-1-xx)/8,yy/8,(hh-1-yy)/8,np.ones((hh,ww))]);weight=np.clip(weight,0,1);a[:,:,3]=(a[:,:,3]*weight).round().astype('uint8');layer=Image.fromarray(a);canvases[job].alpha_composite(layer,at);records.append({**e,'fil':str(dest.relative_to(ROOT)),'sha256':sha(dest),'plassering_4x':at,'heltegning':True})
for job,im in canvases.items():im.convert('RGB').save(ROOT/'jobber'/job/'resultat.png')
(R/'objektleveranser.json').write_text(json.dumps(records,indent=2)+'\n')
