def components(mask):
 seen=np.zeros(mask.shape,bool);out=[]
 for y,x in zip(*np.where(mask)):
  if seen[y,x]:continue
  stack=[(y,x)];seen[y,x]=1;pts=[]
  while stack:
   a,b=stack.pop();pts.append((a,b))
   for da in [-1,0,1]:
    for db in [-1,0,1]:
     c,e=a+da,b+db
     if 0<=c<mask.shape[0] and 0<=e<mask.shape[1] and mask[c,e] and not seen[c,e]:seen[c,e]=1;stack.append((c,e))
  yy,xx=zip(*pts);out.append({'boks':[min(xx),min(yy),max(xx)-min(xx)+1,max(yy)-min(yy)+1],'piksler':len(pts)})
 return out
