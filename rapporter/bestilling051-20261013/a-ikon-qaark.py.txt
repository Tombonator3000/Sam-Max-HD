from pathlib import Path
from PIL import Image,ImageDraw
import json
r=Path(__file__).resolve().parent;root=r.parents[1]
for n in range(1,7):
 j=f'ikon{n:02d}';d=root/'jobber'/j
 if not (d/'resultat.png').exists():continue
 m=json.loads((d/'jobb.json').read_text());src=Image.open(d/'referanse.png');hd=Image.open(d/'resultat.png');can=Image.new('RGB',(1536,1048),(185,185,185));draw=ImageDraw.Draw(can)
 for i,(label,im) in enumerate([('Original',src),('Ny kandidat',hd)]):
  draw.text((5,i*524+4),j+' '+label,fill='black');can.paste(im.resize((768,512)),(0,i*524+20))
 can.save(r/f'a-{j}-helsideqa.png')
 for page in range((len(m['objekter'])+3)//4):
  can=Image.new('RGB',(1280,960),(185,185,185));draw=ImageDraw.Draw(can)
  for k,o in enumerate(m['objekter'][page*4:page*4+4]):
   x,y,w,h=o['rekt'];box=(64+x*4,32+y*4,64+(x+w)*4,32+(y+h)*4);draw.text((5,k*240+4),o['bilde']+' original / kandidat',fill='black')
   for col,im in enumerate([src,hd]):
    crop=im.crop(box);scale=min(625/crop.width,210/crop.height);crop=crop.resize((round(crop.width*scale),round(crop.height*scale)),Image.Resampling.NEAREST if col==0 else Image.Resampling.LANCZOS);can.paste(crop,(col*640,k*240+25))
  can.save(r/f'a-{j}-feltqa-{page+1}.png')
