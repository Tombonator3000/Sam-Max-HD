plan=json.loads((REPORT/'ruteplan-unik.json').read_text())['ruter']
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assignments={
 'A':(list(range(0,5)),[[3],[1],[2],[4],[5]],'01'),
 'B':(list(range(5,11)),[[3],[2],[1],[4],[6],[5]],'01'),
 'C':(list(range(11,17)),[[3],[1],[2],[5],[6],[4]],'01'),

specs={}
for group,(indices,ids,attempt) in assignments.items():

cache={};audit=[];prototypes={}
def drawing(raw,ids,size):
    if raw not in cache:
        a=np.array(Image.open(REPORT/raw).convert('RGBA'));labels,_=nd.label(a[:,:,3]>220);cache[raw]=(a,labels)
    a,labels=cache[raw];core=np.isin(labels,ids);yy,xx=np.nonzero(core);assert len(xx)>20
    x0,y0,x1,y1=int(xx.min()),int(yy.min()),int(xx.max())+1,int(yy.max())+1
    xa,ya,xb,yb=max(0,x0-2),max(0,y0-2),min(a.shape[1],x1+2),min(a.shape[0],y1+2)
    local=core[ya:yb,xa:xb];cut=a[ya:yb,xa:xb].copy()
    distance,nearest=nd.distance_transform_edt(~local,return_indices=True);permitted=distance<=1.5
    cut[:,:,3][~permitted|((labels[ya:yb,xa:xb]>0)&~local)]=0
    edge=permitted&~local;cut[:,:,:3][edge]=cut[nearest[0][edge],nearest[1][edge],:3]
    im=Image.fromarray(cut).crop((x0-xa,y0-ya,x1-xa,y1-ya)).resize(size,Image.Resampling.LANCZOS)
    clean=np.array(im);hit=(clean[:,:,:3]==[255,0,255]).all(2)&(clean[:,:,3]>0)
    if hit.any():
        assert (clean[:,:,3][hit]<=2).all()
        _,near=nd.distance_transform_edt(clean[:,:,3]<128,return_indices=True)
        clean[:,:,:3][hit]=clean[near[0][hit],near[1][hit],:3]
    return Image.fromarray(clean),{'raw':raw,'komponenter':ids,'raw_boks':[x0,y0,x1,y1],'mal':list(size),'rgb_avrundingsrens':int(hit.sum()),'original_pikselmaske_brukt':False}
for pose in plan:
    name=pose['bilde'];spec=specs[name];w,h=pose['rekt'][2:];im,detail=drawing(spec['raw'],spec['ids'],(w*4,h*4))
    dest=REPORT/(name+'-prototype.png');im.save(dest);prototypes[name]=im
    audit.append({'bilde':name,'prototype_sha256':sha(dest),**detail})

 'leveranser':[],'merknad':'Elleve faktiske bildekall, alle gav bilde. Ni råbilder brukes i finalen. F02 gjenoppretter kildebekreftede små brune grep. I retter den kompakte mekanismen i1923/1795. D sin siste pose er erstattet av H med riktig løftet fot. E02 retter håndtilbehørets linjer. 46 ulike originale RGBA-utsnitt, 82 spillruter. Ingen hard originalmaske.'}
for number in range(1,7):
    name=f'fig152_{number:02d}';d=ART/'jobber'/name;j=json.loads((d/'jobb.json').read_text());out=Image.new('RGBA',(1536,1024));maps=[]
    for cel in j['ruter']:
        pose=next(p for p in plan if any(t['bilde']==cel['bilde'] for t in p['malruter']))
        x,y,w,h=cel['rekt'];src=Image.open(d/'original_1x.png').convert('RGBA').crop((x,y,x+w,y+h))
        assert hashlib.sha256(src.tobytes()).hexdigest()==pose['rgba_sha256']
        im=prototypes[pose['bilde']];assert im.size==(w*4,h*4)
        dx,dy=(j['plassering'][0]+x)*4,(j['plassering'][1]+y)*4;out.alpha_composite(im,(dx,dy))
        yy,xx=np.nonzero(np.array(im)[:,:,3]>128)
        maps.append({'bilde':cel['bilde'],'kostyme':152,'rute':cel['rute'],
          'rect':[dx+int(xx.min()),dy+int(yy.min()),int(xx.max()-xx.min()+1),int(yy.max()-yy.min()+1)],
          'representant':pose['bilde'],'kilde_rgba_sha256':pose['rgba_sha256'],
          'prototype':pose['bilde']+'-prototype.png','raw':specs[pose['bilde']]['raw']})
    out.save(d/'resultat.png');digest=sha(d/'resultat.png')
    (REPORT/f'rutekart-{name}.json').write_text(json.dumps({name:{'sha256_resultat':digest,'ruter':maps}},ensure_ascii=False,indent=2)+'\n')

      'egenkontroll':'ikke kjørt','native_mottatt':False,'visuelt_kontrollert':False,'motoravspilt':False,
      'importert':False,'publisert':False,'godkjent_av_Tom':False,'claude_kontroll':False})
