from pathlib import Path
import hashlib,json,subprocess,datetime,shutil

repo=Path('/home/tom/Sam-Max-HD')
art=repo/'work/.gpt-gren'
r=art/'rapporter/sam-gripearm-20261019'
candidate=repo/'out/sam-gripearm-20261019/mod-kandidat'
fetched=repo/'mods/gpt'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def snapshot(root):
    return {p.relative_to(root).as_posix():sha(p) for p in root.rglob('*') if p.is_file()}
base=json.loads((r/'grunnlag-filer-bd65ce87.json').read_text())
actual=snapshot(fetched);expected=snapshot(candidate)
assert len(actual)==19978 and set(actual)==set(expected)
assert {p for p in expected if expected[p]!=actual[p]}=={'SHA256SUMS'}
assert all(actual[p]==h for p,h in expected.items() if p.endswith('.png'))
assert all(actual[p]==h for p,h in base.items() if p.endswith('.png'))
assert {p for p,h in base.items() if actual[p]!=h}=={'mod.json','SHA256SUMS'}
entries=json.loads((r/'tillegg.json').read_text())['tillegg']
assert len(entries)==82 and all(actual[e['fil']]==e['sha256'] for e in entries)
assert sum(p.endswith('.png') for p in actual)==19972
assert len(list((fetched/'costumes').glob('*_hd.png')))==1736
assert json.loads((fetched/'mod.json').read_text())['costume_cels']==1736
protected=json.loads((r/'beskyttede-kilder.json').read_text())
assert len(protected)==24 and all(sha(art/p)==h for p,h in protected.items())
source=subprocess.check_output(['git','rev-parse','HEAD'],cwd=art,text=True).strip()
assert source=='140f44b4c0fd329c3c4211eac80acb4c123f4dff'
native=subprocess.check_output(['git','rev-parse','origin/hd-mod'],cwd=repo,text=True).strip()
remote=subprocess.check_output(['git','ls-remote','origin','refs/heads/hd-mod'],cwd=repo,text=True).split()[0]
assert remote==native=='69d5f1ff15fd1b9c7b388a74041b65ccf83d8ec1'
stash=subprocess.check_output(['git','stash','list'],cwd=repo,text=True).splitlines()
assert len(stash)==3
f=json.loads((r/'fremdrift.json').read_text())
for e in f['leveranser']:
    assert sha(art/e['resultat'])==e['sha256']
    e.update(native_mottatt=True,visuelt_kontrollert=True,motoravspilt=True,importert=True,publisert=True,kildecommit=source,hd_mod_sha=native)
(r/'fremdrift.json').write_text(json.dumps(f,ensure_ascii=False,indent=2)+'\n')
receipt={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'kildecommit':source,'hd_mod_grunnlag':'bd65ce87f0f690ac19aa1c316f376a2e23810193','hd_mod_sha':native,
    'nye_native':82,'native_totalt':1736,'alle_kandidat_png_verifisert':19972,'tidligere_png_byteidentiske':19890,'tidligere_filer_bevart':19896,'endrede_tidligere_filer':['mod.json','SHA256SUMS'],
    'nye_native_sha_verifisert':True,'nedlastet_igjen':True,'sha256sums_kontrollert':True,'beskyttede_kilder_uendret':24,'tre_stasher_bevart':True,'figurjobber_med_resultat':170,'figurjobber_totalt':1758,
    'egenkontroll':{'ok':75,'sjekk':7,'feil':0},'native_mottak':{'ok':82,'sjekk':0,'mangler':0,'teknisk':0},'visuelt_sett':82,'motorobservert':82,'motor_kontroller':300,'fra_figurstart':264,'avvik_fra_figurstart':0,'kjente_oppstartsavvik':{'antall':4,'frame':36,'avvik':160,'identisk_med_grunnlag':True},
    'naturlig_gjennomspilling':False,'ny_Tom_godkjenning':False,'separat_Claude_kontroll':False,
    'nedlastingspakker':{'release':'sam-max-hd-2026-10-19-kveld','native':1654,'mangler_denne_bunken':82,'oppdateres_ved_samlepunkt':True},'tillegg':entries}
(r/'mottak-publisert.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n')
(r/'RAPPORT.md').write_text((r/'RAPPORT.md').read_text()+f'\nPublisering bekreftet: kildecommit {source}, hd-mod {native}.
