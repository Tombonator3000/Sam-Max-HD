def box(mask):
 yy,xx=np.where(mask);return [int(xx.min()),int(yy.min()),int(xx.max()+1),int(yy.max()+1)]

def shadow_mask(shape,rect):
 m=np.zeros(shape[:2],bool)
 if rect:
  x0,y0,x1,y1=rect;m[y0:y1,x0:x1]=True
 return m
def clean(a,rect):

 return a
