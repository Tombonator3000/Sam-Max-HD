 d=json.loads((C/'plan.json').read_text());d.update(ingen_kall_enna=False,kall_gjort=True,raa_sha256=sha(C/'raa.png'));save(C/'plan.json',d)
 print(c,Image.open(C/'raa.png').size,sha(C/'raa.png')[:12])
