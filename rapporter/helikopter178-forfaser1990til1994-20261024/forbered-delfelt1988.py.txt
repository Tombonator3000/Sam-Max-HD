from pathlib import Path

import hashlib,json,sys

P=Path('/home/tom/Sam-Max-HD'); A=P/'work/.gpt-gren'; O=Path(__file__).resolve().parent
R=A/'rapporter'/O.name; S=A/'rapporter/helikopter178-forfaser1995til1999-20261023/kilder-neste/1988'

 'overblad':([36,0,67,13],[700,345,1160,530]),

 out=Image.new('RGBA',(im.width+2*margin,im.height+2*margin));out.alpha_composite(im,(margin,margin));return out
source=Image.open(S/'original-1x.png').convert('RGBA').crop(crop)
