from pathlib import Path
from PIL import Image
import numpy as np, json, hashlib
P=Path('/home/tom/Sam-Max-HD'); R=P/'work/.gpt-gren/rapporter/helikopter178-forfaser1990til1994-20261024'
K=R/'kall/H248'; V=K/'eget-lite-graafelt'; V.mkdir()
base=R/'kall/H246/eget-blad-alfa/egen-heltegning.png'
new=K/'utsnitt-sammensatt/egen-heltegning.png'
old=np.array(Image.open(base).convert('RGBA')); full=np.array(Image.open(new).convert('RGBA'))
# Whole rectangular detail in our own images. No source pixels or source mask.
box=[548,777,602,812];x,y,X,Y=box;H,W=Y-y,X-x
yy,xx=np.indices((H,W));edge=np.minimum.reduce([xx,yy,W-1-xx,H-1-yy]);t=np.clip(edge/6,0,1);w=t*t*(3-2*t)
a=old[y:Y,x:X].astype(float)/255;b=full[y:Y,x:X].astype(float)/255
alpha=a[:,:,3]*(1-w)+b[:,:,3]*w
pre=a[:,:,:3]*a[:,:,3,None]*(1-w[:,:,None])+b[:,:,:3]*b[:,:,3,None]*w[:,:,None]
rgb=np.divide(pre,alpha[:,:,None],out=np.zeros_like(pre),where=alpha[:,:,None]>0)
out=old.copy();out[y:Y,x:X]=np.uint8(np.clip(np.rint(np.dstack((rgb,alpha))*255),0,255))
mask=np.ones(old.shape[:2],bool);mask[y:Y,x:X]=False;assert np.array_equal(out[mask],old[mask])
Image.fromarray(out).save(V/'egen-heltegning.png')

(R/'avgrens-eget-graafelt.py.txt').write_bytes(Path(__file__).read_bytes())
