def put(p, d):
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')
def inventory(root):
    fs = [f for f in root.rglob('*') if f.is_file()]
    with ThreadPoolExecutor(max_workers=6) as ex:
        return dict(ex.map(lambda f: (str(f.relative_to(root)), sha(f)), fs))
