from pathlib import Path
from PIL import Image
from datetime import datetime, timezone
import json, hashlib, sys

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
R = A / 'rapporter/helikopter178-forfaser1990til1994-20261024'
cid = sys.argv[1]
assert cid == 'H248'
K = R / 'kall' / cid
K.mkdir()
base = R / 'kall/H246/eget-blad-alfa/egen-heltegning.png'
roi = [470, 640, 890, 830]
Image.open(base).crop(roi).save(K / 'egen-kroppside.png')
source = Image.open(R / 'kilder/1990/original-1x.png').convert('RGBA').crop((25, 13, 54, 27))
flat = Image.new('RGBA', source.size, (215, 92, 156, 255))
flat.alpha_composite(source)
flat.resize((1160, 560), Image.Resampling.NEAREST).save(K / 'original-kroppside-kun-referanse.png')
prompt = '''Use case: precise-object-edit
Asset type: one small transparent HD game-sprite detail, not a whole new helicopter.
Image 1 is the edit target: a crop of our own helicopter drawing. Keep its exact framing, shapes, brown silhouette, surrounding pieces and transparent pixels. Return only this same cropped detail, with real alpha.
Image 2 is the matching original pixel-art body detail, used ONLY to understand the three small flat gray marks and their shape. Its pink is a reference background and must not appear in the output.
Image 3 shows our entire helicopter solely for placement context. Image 4 shows the whole correct original pose. Image 5, approved Sam, is the sole ink/flat-shading style anchor.
Primary request: correct only the three gray body-side marks. In image 1 the smallest rear mark has mistakenly become a round gray button with a dark circular outline. Flatten it into the small irregular gray inset shown in image 2, flush with the brown surface, without a knob, rim, circle or highlight. Preserve the separate larger slanted gray mark above/left and the longer flat gray mark along the body side: three separate groups, in the same places and sizes as the source, with brown space between them. Preserve the gray front trim already at the far right; do not merge it with the side marks.
Keep all surrounding brown, green, silver/gray and black parts, the windshield, diagonal rear silver strut and every cut-out opening unchanged. Preserve the sparse single flat shade, clean dark smooth comic ink, no soft gradients, no texture, no beveled hardware, no text, no new fasteners. Do not outline individual source pixels or turn source pixel steps into large blocks. No change to framing or relative geometry. Only the gray mark shapes change.'''
refs = [K/'egen-kroppside.png', K/'original-kroppside-kun-referanse.png', base, R/'kilder/1990/referanse24x.png', A/'hovedark/sam.png']
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
args = dict(prompt=prompt, referenced_image_paths=list(map(str, refs)), transparent_background=True)
plan = dict(kall=cid, rute=1990, jobb='fig178_02', utfort=False,
            opprettet_utc=datetime.now(timezone.utc).isoformat(),
            formaal='Tre separate flate graafelt, fjern oppdiktet rund sideknapp',
            type='generert_egenutsnitt_skal_plasseres_tilbake_i_egen_tegning',
            egen_heltegning=str(base), egen_heltegning_SHA256=sha(base),
            utsnitt_i_egen_tegning=roi, kildeutsnitt_kun_referanse=True,
            original_kontekst_og_stil_sett=True,
            referanser_SHA256={str(p): sha(p) for p in refs})
(K/'prompt-brukt.txt').write_text(prompt+'\n')
(K/'argumenter.json').write_text(json.dumps(args, ensure_ascii=False, indent=2)+'\n')
(K/'plan.json').write_text(json.dumps(plan, ensure_ascii=False, indent=2)+'\n')
(R/'forbered-smaafelt.py.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps({'kall': cid, 'egen_base': str(base), 'utsnitt': roi, 'referanser': list(map(str, refs))}, ensure_ascii=False))
