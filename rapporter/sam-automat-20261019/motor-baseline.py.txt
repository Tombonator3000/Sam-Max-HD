from pathlib import Path
import importlib.util, json, hashlib, subprocess

ROOT=Path('/home/tom/Sam-Max-HD')
OUT=ROOT/'out/sam-automat-20261019'
REPORT=ROOT/'work/.gpt-gren/rapporter/sam-automat-20261019'
spec=importlib.util.spec_from_file_location('kontroll',ROOT/'out/inventoryknapp-20261016/kontroll.py')
k=importlib.util.module_from_spec(spec);spec.loader.exec_module(k)

pkg=ROOT/'out/felles-release-20261019/linux/squashfs-root/usr'
k.NEW=pkg/'bin/scummvm';k.LIB=pkg/'lib';k.PROJECT=pkg/'share/sam-max-hd'

base={p.relative_to(mod).as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in mod.rglob('*') if p.is_file()}
(REPORT/'grunnlag-filer-d31c30f2.json').write_text(json.dumps(base,indent=2)+'\n')

results=[]
for anim in (1,7,8,9,10):
    env={'DIGHD_TEST_ROOM':'43','DIGHD_TEST_AT':'10','DIGHD_TEST_COSTUME':'220',
         'DIGHD_TEST_CHORE_EVERY':'1000','DIGHD_TEST_STATIC_SCENE':'1',
         'DIGHD_TEST_SCALE':'255','DIGHD_TEST_POS':'160,180','DIGHD_DUMP_EVERY':'4',
         'DIGHD_DEBUG_SOFT':'1','DIGHD_DEBUG_COSTUME':'1','DIGHD_TEST_ANIMATION':str(anim)}
    results.append(k.run(f'baseline220-anim{anim}',k.NEW,env,360,mod))
(REPORT/'motor-baseline.json').write_text(json.dumps(results,indent=2)+'\n')
assert all(r['exit']==0 and r['dumps']>0 for r in results)
