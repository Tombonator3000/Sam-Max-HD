def colored_patch(raw, box, color):
    patch = raw.crop(box).convert('RGB')
    a = np.asarray(patch).astype(float)
    r, g, b = a[..., 0], a[..., 1], a[..., 2]
    if color == 'red':
        score = r - np.maximum(g, b)
    elif color == 'yellow':
        score = np.minimum(r-b, g-b)
    else:
        score = np.minimum(g-r, g-b)
    alpha = np.uint8(np.clip((score-3)/12, 0, 1)*255)
    # Alfa kommer bare fra den nye tegningens farge mot blå himmel.
    # Ingen originalpiksler eller originalalfa brukes i komposisjonen.
    return patch, Image.fromarray(alpha)
