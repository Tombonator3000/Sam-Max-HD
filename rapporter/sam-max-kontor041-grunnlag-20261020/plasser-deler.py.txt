def piece(call,part,box,size):
    a=raw[call].copy();l=labels[call];sel=l==part;yy,xx=np.nonzero(sel);gx,gy=int(xx.min()),int(yy.min());gw,gh=int(xx.max()-gx+1),int(yy.max()-gy+1)
