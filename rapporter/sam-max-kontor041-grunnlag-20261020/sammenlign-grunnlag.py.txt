from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
import os, subprocess, shutil, json, hashlib, re
P = Path('/home/tom/Sam-Max-HD')
O = Path(__file__).resolve().parent
R = P / 'work/.gpt-gren/rapporter' / O.name
PK = P / 'out/felles-release-20261020-sam343/linux/squashfs-root'
D = PK / 'usr/share/sam-max-hd'
B = PK / 'usr/bin/scummvm'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(B) == '8db371698b610d40a5c0d789f777dad34cb5d666b91f305f2be8e062a3427914'
def run(case):
    name, mod = case
    T = O / name
    assert not T.exists()
    T.mkdir()
    h, s = T / 'home', T / 'saves'
    h.mkdir(); s.mkdir()
    for p in (P / 'out/inventoryknapp-20261016/saves').glob('samnmax.s*'):
        shutil.copy2(p, s / p.name)
    env = {k: v for k, v in os.environ.items() if not k.startswith('DIGHD_')}
    env.update(HOME=str(h), XDG_CONFIG_HOME=str(h / '.config'), XDG_DATA_HOME=str(h / '.local/share'),
        XDG_CACHE_HOME=str(h / '.cache'), LD_LIBRARY_PATH=str(PK / 'usr/lib'), SDL_VIDEODRIVER='dummy',
        SDL_AUDIODRIVER='dummy', DIGHD_MOD=str(mod), DIGHD_VERIFY='1', DIGHD_QUIT_AT='320',
        DIGHD_DUMP_DIR=str(T), DIGHD_DUMP_EVERY='4', DIGHD_DUMP_FLAT='1', DIGHD_TEST_ROOM='7',
        DIGHD_TEST_AT='10', DIGHD_TEST_CAMX='160', DIGHD_MANGLER='0', DIGHD_DEBUG_SOFT='1',
        DIGHD_DEBUG_COSTUME='1', DIGHD_TEST_COSTUME='41', DIGHD_TEST_ACTOR='2', DIGHD_TEST_CLIP='2',
        DIGHD_TEST_CHORE_EVERY='1000', DIGHD_TEST_STATIC_SCENE='1', DIGHD_TEST_SCALE='255',
        DIGHD_TEST_POS='237,120', DIGHD_TEST_ANIMATION='6')
    args = [str(B), '--config=' + str(T / 'scummvm.ini'), '--path=' + str(D / 'game'), '--savepath=' + str(s),
        '--save-slot=2', '--gfx-mode=normal', '--no-fullscreen', '--gui-theme=builtin', '--music-driver=null',
        '--debuglevel=1', 'samnmax']
    with (T / 'scummvm.log').open('wb') as f:
        code = subprocess.run(args, env=env, stdout=f, stderr=subprocess.STDOUT, timeout=240).returncode
    log = (T / 'scummvm.log').read_text(errors='replace')
    checks = [dict(frame=int(f), avvik=int(n)) for f, n in re.findall(r'verify frame (\d+): (\d+) screen pixels differ from the whole screen built again', log)]
    draws = [dict(frame=int(f), cel=int(n)) for f,n in re.findall(r'frame (\d+) soft cel\s+(\d+) costume\s*41\b', log)]
    proof = dict(scene=name, exit=code, binar_sha256=sha(B), mod=str(mod),
        mod_json_sha256=sha(mod/'mod.json'), logg_sha256=sha(T/'scummvm.log'), kontroller=checks,
        helprove_bestatt=bool(code==0 and checks and not any(v['avvik'] for v in checks)), tegninger=draws,
        naturlig_gjennomspilling=False, elevation_test=0, elevation_primarskript=-2)
    (T/'kvittering.json').write_text(json.dumps(proof,indent=2)+'\n')
    assert code==0 and len(checks)==80 and set(range(1920,1969)) <= {v['cel'] for v in draws}
    print(name, [(v['frame'],v['avvik']) for v in checks if v['avvik']], '49 kildeposer',flush=True)
    return proof
cases = [('sammenligning2456-ny-binar', P/'out/felles-release-20261020-sam342-ferdig/linux/squashfs-root/usr/share/sam-max-hd/mods/gpt'),
         ('sammenligning2478-ny-binar', D/'mods/gpt')]
with ThreadPoolExecutor(max_workers=2) as pool:
    rows=list(pool.map(run,cases))
same=rows[0]['kontroller']==rows[1]['kontroller']
proof=dict(proever=rows, samme_binar=True, samme_kontrollresultat=same,
    ingen_ny041_grafikk=True, avvik_forklart=False)
(R/'grunnlagssammenligning.json').write_text(json.dumps(proof,indent=2)+'\n')
shutil.copy2(__file__,R/'sammenlign-grunnlag.py.txt')
assert same, 'Grunnlagene gir ulike avvik; undersok for produksjon'
print('Samme kontrollresultat med 2456 og 2478 i faktisk 2478-pakkebinar.',flush=True)
