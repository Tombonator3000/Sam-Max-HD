def prepend(path, text):
    s = path.read_text()
    s = re.sub(r'<!-- ' + marker + r':start -->.*?<!-- ' + marker + r':slutt -->\n*', '', s, flags=re.S)
    lines = s.splitlines(keepends=True)
    block = '<!-- ' + marker + ':start -->\n' + text + '\n<!-- ' + marker + ':slutt -->\n'
    path.write_text(lines[0].rstrip() + '\n\n' + block + '\nTidligere status nedenfor er historikk.\n\n' + ''.join(lines[1:]).lstrip().rstrip() + '\n')
