from pathlib import Path
import hashlib, json, shutil

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(p, q):
    p.write_text(json.dumps(q, ensure_ascii=False, indent=2) + '\n')

# Utføres etter faktisk visning av begge native, gråtavlen, delarket,
# begge hele motorbilder, begge før/etter-par og 24-fasertavlen.
motor = read(R / 'motor-sammenligning.json')
assert motor['alle2_observert'] and motor['uforklarte_HD_endringer'] == 0
assert motor['ingen_nye_eller_okte_bufferavvik'] and motor['atomisk_skriving_bestatt']
vis = read(R / 'motor-visuell-plan.json')
for group in ['varianter', 'hele_motorbilder']:
    assert len(vis[group]) == 2
    for e in vis[group]:
        assert sha(R / e['fil']) == e['sha256']
        e['visuelt_sett'] = True
assert sha(R / vis['animasjonstavel']['fil']) == vis['animasjonstavel']['sha256']
vis['visuelt_sett'] = vis['animasjonstavel']['visuelt_sett'] = True
save(R / 'motor-visuell-plan.json', vis)
motor['visuelt_sett'] = True
save(R / 'motor-sammenligning.json', motor)
q = read(R / 'motor-pikselkontroll.json')
assert len(q) == 2
for e in q: e['visuelt_sett'] = True
save(R / 'motor-pikselkontroll.json', q)

rows = read(R / 'egen-native.json')
assert {e['rute'] for e in rows['ruter']} == {2007, 2009}
for e in rows['ruter']:
    assert sha(A / e['prototype']) == e['sha256']
    e.update(motorobservert=True, native_visuelt_sett=True)
save(R / 'egen-native.json', rows)
for name in ['rutekart-fig178_02.json', 'rutekart-samlet.json']:
    q = read(R / name)
    assert len(q['fig178_02']['ruter']) == 3
    for e in q['fig178_02']['ruter']:
        if e['rute'] in [2007, 2009]:
            e.update(native_mottatt=True, motorobservert=True, native_visuelt_sett=True)
    save(R / name, q)

codes, colors, returns = [], [], []
for c in ['H082', 'H083', 'H084', 'H085']:
    K = R / 'kall' / c
    selected = c in ['H082', 'H085']
    if (K / 'fargekontroll.json').exists(): colors.append({**read(K / 'fargekontroll.json'), 'kall': c})
    if (K / 'kodeflate-kontroll.json').exists(): codes.append({**read(K / 'kodeflate-kontroll.json'), 'kall': c})
    if selected:
        q = read(K / 'kontroll.json')
        co = read(K / 'kodeflate-kontroll.json')
        assert q['tekniske_porter_bestatt'] and q['alle_alfa_fester_innen3HD']
        assert read(K / 'fargekontroll.json')['alle_fargefester_innen3HD']
        assert co['alle_kodefester_innen3HD']
        assert all(e['egne_kodepiksler_over3HD_fra_kilde'] == 0 for e in co['farger'])
        q.update(status='EGENKONTROLLERT_NATIVE_MOTOROBSERVERT', egenkontrollert=True,
                 native_mottatt=True, motorobservert=True, visuell_native_kontroll=True)
        save(K / 'kontroll.json', q)
    elif c == 'H083':
        returns.append(dict(kall=c, omfang='mellomresultat høyre del', brukt_i_sluttgrafikk=False,
                            grunn='Tre gale toppfelt. H085 er nytegnet som erstatning.'))
    else:
        returns.append(dict(kall=c, omfang='tidlig helprototype med H083 høyre del',
                            helprototype_brukt=False, venstre_raadel_brukt=True,
                            brukt_som_venstre_del_i='H085',
                            grunn='Tidlig helprototype avvist. Venstre del består og er brukt med den nye H085-høyredelen.'))
save(R / 'fargekontroll-alle.json', colors)
save(R / 'kodeflate-kontroll-samlet.json', codes)
save(R / 'kvalitetsreturer.json', returns)

replay = read(R / 'kildelos-replay.json')
assert len(replay) == 2 and all(e['begge_PNG_hash_eksakt_gjenlaget'] and not e['originalfil_aapnet'] for e in replay)
for e in replay:
    assert not e['originalmaske'] and not e['kildekoordinater_til_plassering']
q = read(R / 'fremdrift.json')
q['nye_raadeler_direkte_brukt'] = q.pop('egne_nye_rakter_deler_brukt', 3)
q['motorobservert'] = 2
for e in q['delvise_leveranser']: e['motorobservert'] = True
save(R / 'fremdrift.json', q)
save(R / 'sluttkontroll.json', dict(klar_for_additiv_publisering=True, visuelt_sett=True,
    valgte_native=2, motorobservert=2, native=3716, publisert=0, egne_nye_malerier=2,
    nye_raadeler_direkte_brukt=3, hele_ark_levert=0, automatiske_OK=2, automatiske_SJEKK=0,
    egen_native_port_bestatt=2, farge_og_alfafester_bestatt=2, beskyttede=1472, stasher=3,
    Tom_godkjent=False, Claude_godkjent=False, pakker_native=3713, pakker_mangler_native=3))
save(R / 'visuell-valgt-native.json', dict(valgte=['H082', 'H085'],
    kompositt='to-valgte-native-kontroll2x.png', sha256=sha(R / 'to-valgte-native-kontroll2x.png'),
    visuelt_sett=True, ekte_alfa_kompositt_pa_graabakgrunn=True,
    hele_native_og_delark_sett=True, begge_hele_motorbilder_sett=True,
    begge_for_etter_par_sett=True, sekvensfaser1998til2021_sett=True))

old = R / 'elleve-kandidater-kontroll4x.png'
