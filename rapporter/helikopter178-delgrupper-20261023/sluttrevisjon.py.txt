read = lambda p: json.loads(p.read_text())
sha = lambda b: hashlib.sha256(b).hexdigest()
def save(p, q): p.write_text(json.dumps(q, ensure_ascii=False, indent=2) + '\n')
def git(p, *args): return subprocess.check_output(['git', *args], cwd=p, text=True).strip()
def api(s): return json.loads(subprocess.check_output(['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/' + s], cwd=P, text=True))
def inventory(root):
    files = [p for p in root.rglob('*') if p.is_file()]
    with ThreadPoolExecutor(max_workers=8) as pool:
        return dict(pool.map(lambda p: (str(p.relative_to(root)), sha(p.read_bytes())), files))
def release_signature(e):
    return (e['id'], e['tag_name'], e['body'], sorted((a['id'], a['name'], a['size'], a['digest']) for a in e['assets']))

def checkmain(name):
    data = (P / name).read_bytes()
    oid = git(P, 'rev-parse', main + ':' + name)
    blob = subprocess.check_output(['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/git/blobs/' + oid,
        '-H', 'Accept: application/vnd.github.raw'], cwd=P)
    assert data == blob
    return dict(fil=name, sha256=sha(blob), gitblob=oid, faktisk_hentet_fra_GitHub=True)
