    if k==13:
        if c in [0,1,2]:return ('Only the detached BACK of Max\'s head and ears, with no visible eyes, nose, mouth, body or shoulders. '
            'Keep the exact three distinct ear bends and the cream lower head shadow of this source phase. It is an animation layer, not a complete character.')
        if c==128:return ('Max\'s complete source BODY layer with the small orange/tan inanimate clockwork prop at the left. '
            'The separate back-of-head layer is NOT part of this drawing. Preserve all cream body fragments, the long thin arm contacting the prop, '
            'hands, feet, blue-gray prop rim and loose reddish-brown wire exactly as visible. Do not fill the intentionally missing head area.')
        return ('Max\'s headless cream BODY layer only, with belly circle, arms and feet in the exact source pose. '
                'Do not add any head or ears, neck stump, red colors, prop or ground shadow. The head is a separate layer in the game.')
    if k==17:

                          'cream color and limbs. The back views show no face; front views have the exact source grin.')

                          'Keep the exact visible grin, red inner ear, bent limbs, detached marks and flattened ear tips that THIS source shows. '
                          'Do not brighten the dark shadow phases or add missing outlines or body parts.')

                          'orange/tan areas, black negative spaces and the small pale marks visible in the source. '
                          'Do not turn this into a full car, face, head or complete Max. Preserve every detached piece and internal transparent gap.')
        return ('Only the intentionally small dark silhouette or disconnected fragment shown as Max disappears behind the car. '
                'Preserve this exact black/gray shape and any tiny blue-gray or pale colored mark; no completed character, eyes or extra details. '
                'Keep the internal holes and separate dots exactly as shown.')
