def project(source,e,flat):
    # Velg fra kildens egen RGB565-projeksjon, uavhengig av ny HD-tegning.
    # Dette er bare måleflaten. Ingen bildeklipping eller ny toleranse.
    x,y=e['pos'];w,h=e['maal'];candidates=[]
    for mirror in [False,True]:
        src=source[:,::-1] if mirror else source
        rgb=src[:,:,:3].astype(np.uint16);r=rgb[:,:,0]>>3;g=rgb[:,:,1]>>2;b=rgb[:,:,2]>>3
        rgb=np.stack([(r<<3)|(r>>2),(g<<2)|(g>>4),(b<<3)|(b>>2)],2).astype(np.uint8)
        expected=np.repeat(np.repeat(rgb,4,0),4,1);mask=np.repeat(np.repeat(src[:,:,3]>0,4,0),4,1)
        for dx in [-1,0,1]:
            for dy in [-1,0,1]:
                xx,yy=(x+dx)*4,(y+dy)*4
                if min(xx,yy)<0 or xx+w*4>1280 or yy+h*4>800:continue
                crop=flat[yy:yy+h*4,xx:xx+w*4]
                wrong=int((mask&np.any(crop!=expected,axis=2)).sum())
                candidates.append((wrong,abs(dx)+abs(dy),mirror,dx,dy,mask))
    candidates.sort(key=lambda e:e[:5]);best=candidates[0]
    mask=np.zeros((800,1280),bool);xx,yy=(x+best[3])*4,(y+best[4])*4
    mask[yy:yy+h*4,xx:xx+w*4]=best[5]
    proof=dict(speilvendt=best[2],dx_original=best[3],dy_original=best[4],kildefargeavvik=best[0],
        synlige_kildepiksler=int(best[5].sum()),neste_kandidater=[dict(avvik=t[0],speil=t[2],dx=t[3],dy=t[4]) for t in candidates[1:5]],
        kildeprojeksjon_uavhengig_av_HD=True)
    return mask,proof
