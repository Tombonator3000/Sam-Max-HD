from pathlib import Path
from datetime import datetime, timezone
import json, hashlib, subprocess, shutil

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
S = A / 'rapporter/sam-max223-slutt-20261021'
N = A / 'rapporter/sam-max223-kilde-20261021'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
save = lambda p,d: p.write_text(json.dumps(d, ensure_ascii=False, indent=2)+'\n')
api = lambda x: json.loads(subprocess.check_output(['gh','api','repos/Tombonator3000/Sam-Max-HD/'+x], cwd=P, text=True))
s = read(O/'bygg-startet.json')
tag,code,mod = (s[k] for k in ['release_tag','kode','hd_mod'])
manifest = read(R/'moddfiler-3324.json')
old = read(A/'rapporter/felles-release-20261021-max-intro/moddfiler-3276.json')
assert len(manifest)==21567 and all(sha(P/'mods/gpt'/f)==h for f,h in manifest.items())
assert set(old)<=set(manifest) and all(manifest[f]==h for f,h in old.items() if f.endswith('.png'))
protected = read(N/'beskyttede-samlet1252.json')
assert len(protected)==1252 and all(sha(A/f)==h for f,h in protected.items())
base = read(S/'grunnlag-produksjon.json')
assert subprocess.check_output(['git','stash','list','--format=%H'], cwd=P, text=True).splitlines()==base['stash']
assert len(list((A/'jobber').glob('fig*/resultat.png')))==465
motor = read(R/'linux-kontroll.json')
files = read(R/'moddfiler-kontroll.json')
assets = read(O/'assetkontroll.json')
raster = read(R/'rasterkontroll-mot-produksjon.json')
pixels = read(R/'pakke-pikselkontroll.json')
vis = read(R/'visuell-kontroll.json')
assert motor['native']==3324 and motor['nye_native_observert']==48
assert motor['runder']==motor['grunnlagsrunder_samme_binar']==1
assert motor['skjermbufferkontroller']==360 and motor['nullkontroller']==motor['grunnlag_nullkontroller']==359
assert motor['alle_eksakte_avvikskoordinater_og_RGB_par_identiske'] and motor['avviksrekkefolge_identisk']
assert files['moddfiler_per_pakke']==21566 and files['begge_modder_byteidentiske'] and files['begge_faktiske_nedlastinger_verifisert']
assert files['hd_mod']==mod and files['kode']==code
assert len(assets)==4 and all(sha(O/v['plattform']/v['fil'])==v['sha256'] for v in assets)
assert raster['alle_nativefiler_identiske'] and raster['croprasteridentiske']==48 and not raster['ikke_rasteridentiske']
assert len(pixels)==48 and all(v['originalflat_identisk'] and v['utenfor_ny_aktorflate']==0 for v in pixels)
assert vis['visuelt_sett'] and vis['egne_hele_varianter']==19
assert len(vis['hele_motorbilder'])==len(vis['hele_for_etter'])==8
assert all(v['visuelt_sett'] for v in vis['varianter']+vis['hele_for_etter'])
for e in vis['hele_motorbilder']+vis['hele_for_etter']+vis['varianter']:
    assert sha(R/e['fil'])==e['sha256']
assert api('git/ref/heads/hd-mod')['object']['sha']==mod
lr,wr = (s['plattform'][p]['run_id'] for p in ['linux','windows'])
body = f'''Linux og Windows inneholder **3324 native HD-bildelag**, med **48 nye animasjonsruter fra bokscenen** siden forrige fellesutgave. Sju hele figurark er levert fra denne scenen i Conroys soverom. Sam og Max følger hver hel tegning med den store åpne boken. Resten av HD-grafikken er under arbeid.

Last ned **SamMaxHD-Windows.zip** for Windows eller **SamMaxHD-x86_64.AppImage** for Linux. Ctrl+H bytter grafikk, og F5 åpner originalmenyen.

Begge faktiske nedlastinger, fire vedleggsdigester, SHA-filer og alle **21566 runtime-moddfiler per pakke** er kontrollert. Moddene er byteidentiske med publisert hd-mod. Alle 48 nye native ruter er logget i den faktiske nye Linux-binæren. PNG-filene og figurutsnittene er identiske med produksjonsopptakene. Nitten ulike hele tegninger og åtte hele før/etter-par er visuelt sett.

En kildebegrunnet hel animasjon i rom 45 er avspilt med både 3276-grunnlaget og den nye modden, i samme nye Linux-binær. Begge gir **359 av 360 nullkontroller**, med samme kjente avvik på 160 piksler ved bilde 36. Eksakte feilkoordinater, begge RGB-verdier og rekkefølgen er identiske. Alle 48 originalflater er SHA-identiske; ingen endringer finnes utenfor logget aktørflate med åtte HD-pikslers målemargin. Ingen nye eller økte bufferavvik. Hele prøven er ikke avviksfri. Andre aktører og romskript er ikke skjult.

465 av 1758 hele figurjobber har resultat; 1293 gjenstår pluss returer, katalogfeil og samlet sluttkontroll. De sju bokarkene har 19 egne hele malerier og 29 eksakte hele RGBA-aliasmål. Råbilder, faktiske prompter, SHA-festet fremdrift, rutekart og egne alfakanaler er bevart. Faste native grenser er uendret. Seks andre bokark med 46 mål holdes tilbake etter en verktøyavvist retting og visuelle feil i utkast. De er ikke publisert; avviste motiver omprøves ikke automatisk. Uavhengige grafikkjobber fortsetter.

Ingen ny Tom-/Claude-godkjenning eller motor-/pipelineendring. Full naturlig gjennomspilling av bokscenen, interaktiv Windows og ny webtest gjenstår. De kjente SDL-, inventar- og testhoppbegrensningene er fortsatt åpne. Grønt pakkebygg beviser ikke at en sporadisk SDL-startfeil er rettet.

Kode: {code}
hd-mod: {mod}
Linux: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{lr}
Windows: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{wr}
Kontrollrapport: https://github.com/Tombonator3000/Sam-Max-HD/blob/gpt-arbeid/rapporter/{R.name}/RAPPORT.md

Spillet og grafikken tilhører Disney/Lucasfilm.
'''
(O/'utgivelse-notat.md').write_text(body)
shutil.copy2(O/'utgivelse-notat.md', R/'utgivelse-notat.md')
prior = api('releases?per_page=100')
before = read(O/'releaser-foer-bygg.json')
assert not any(v['tag_name']==tag for v in prior)
assert {v['id'] for v in before}<={v['id'] for v in prior}
aset = lambda v: {(a['id'],a['name'],a['size'],a['digest']) for a in v['assets']}
for v in before:
    if v['tag_name'] not in ['linux','windows']:
        assert aset(v)==aset(next(x for x in prior if x['id']==v['id']))
for D in [O,R]: save(D/'releaser-foer.json',prior)
save(R/'bevaringskontroll.json',dict(beskyttede_kilder_SHA_uendret=1252,tre_stasher_uendret=True,alle_tidligere_filstier_bevart=True,alle_tidligere_png_byteidentiske=True,filer3324=21567,figurjobber_med_resultat=465,tidligere_fellesutgaver_og_assets_bevart=True,rullerende_linux_windows_oppdatert_med_autoriserte_bygg=True))
(R/'RAPPORT.md').write_text(body+'\nLinux-binær SHA256 '+motor['binar_sha256']+'. Nativefrys beholdes til bekreftet fellespublisering og kontroll av eventuelle ekstra taggbygg.\n')
for p in O.glob('*.py'): shutil.copy2(p,R/(p.name+'.txt'))
print('Fellesutgave klar: begge nedlastinger, 21566 runtimefiler per pakke, 48 identiske raster, 359/360 null og visuell kontroll bekreftet.',flush=True)
