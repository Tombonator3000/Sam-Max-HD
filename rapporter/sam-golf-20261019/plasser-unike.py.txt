"""Rens generert alfa og plasser tegninger etter originale rektangler.

Ingen original pikselmaske klipper nye tegninger. Små originale grå deler
plasseres separat fra den genererte grå delen, ikke fra originale piksler.
"""
from pathlib import Path
from PIL import Image, ImageDraw, ImageFilter
from scipy import ndimage as nd
import numpy as np, json, hashlib, shutil, subprocess

REPO=Path('/home/tom/Sam-Max-HD');ART=REPO/'work/.gpt-gren'
REPORT=ART/'rapporter/sam-golf-20261019'
plan=json.loads((REPORT/'ruteplan-unik.json').read_text())['ruter']
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
specs={}
assignments={
 'A':(list(range(0,7)),[2,1,3,4,6,7,5],'01'),
 'B':(list(range(7,15)),[1,2,3,4,27,26,28,29],'01'),
 'C':(list(range(15,20)),[2,3,1,4,5],'02'),
 'G':(list(range(36,41)),[2,1,3,4,5],'01')}
for label,(indices,ids,attempt) in assignments.items():
    for idx,part_id in zip(indices,ids):
        specs[plan[idx]['bilde']]={'raw':f'unik-{label}-raa{attempt}.png','ids':[part_id]}
(REPORT/'prototyper-rawvalg.json').write_text(json.dumps(specs,ensure_ascii=False,indent=2)+'\n')

cache={};audit=[];prototypes={}
def drawing(raw,ids,size):
    if raw not in cache:
        a=np.array(Image.open(REPORT/raw).convert('RGBA'))
        # Fjern bare de nygenererte hvite rissene inne i den blå buksen.
        # Ekte alfakontur, skjorte, kølle og råfil er urørt.
        if raw=='unik-C-raa02.png':
            original_alpha=a[:,:,3].copy();ys,xs=np.indices(a.shape[:2])
            area=(xs>=765)&(xs<880)&(ys>=280)&(ys<355)
            bad=area&(a[:,:,0]>150)&(a[:,:,0]>a[:,:,1]-25)&(a[:,:,3]>220)
            clean=(a[:,:,0]>60)&(a[:,:,0]<150)&(a[:,:,1]>a[:,:,0]+10)&(a[:,:,2]>a[:,:,1]+10)&(a[:,:,3]>220)
            _,near=nd.distance_transform_edt(~clean,return_indices=True)
            a[:,:,:3][bad]=a[near[0][bad],near[1][bad],:3]
            assert np.array_equal(original_alpha,a[:,:,3])
            (REPORT/'C-bukserens.json').write_text(json.dumps({'raw':raw,'rekt':[765,280,880,355],'rensede_rgb_piksler':int(bad.sum()),'alfa_uendret':True,'metode':'Nærmeste eksisterende blå buksefarge, ingen nytegning eller silhuettmaske.'},ensure_ascii=False,indent=2)+'\n')
        labels,_=nd.label(a[:,:,3]>220)
        cache[raw]=(a,labels)
    a,labels=cache[raw];core=np.isin(labels,ids)
    if raw=='unik-C-raa02.png' and ids==[4]:
        ys,xs=np.indices(a.shape[:2]);white=(a[:,:,:3].min(2)>215)&(np.ptp(a[:,:,:3].astype(int),axis=2)<20)
        core|=(xs>=77)&(xs<148)&(ys>=588)&(ys<690)&white&(a[:,:,3]>20)
    yy,xx=np.nonzero(core)
    assert len(xx)>10
    x0,y0,x1,y1=int(xx.min()),int(yy.min()),int(xx.max())+1,int(yy.max())+1
    xa,ya,xb,yb=max(0,x0-2),max(0,y0-2),min(a.shape[1],x1+2),min(a.shape[0],y1+2)
    local=core[ya:yb,xa:xb];cut=a[ya:yb,xa:xb].copy()
    distance,nearest=nd.distance_transform_edt(~local,return_indices=True);permitted=distance<=1.5
    cut[:,:,3][~permitted|((labels[ya:yb,xa:xb]>0)&~local)]=0
    edge=permitted&~local;cut[:,:,:3][edge]=cut[nearest[0][edge],nearest[1][edge],:3]
    im=Image.fromarray(cut).crop((x0-xa,y0-ya,x1-xa,y1-ya)).resize(size,Image.Resampling.LANCZOS)
    clean=np.array(im);hit=(clean[:,:,:3]==[255,0,255]).all(2)&(clean[:,:,3]>0)
    if hit.any():
        assert (clean[:,:,3][hit]<=2).all()
        _,nearest=nd.distance_transform_edt(clean[:,:,3]<128,return_indices=True)
        clean[:,:,:3][hit]=clean[nearest[0][hit],nearest[1][hit],:3]
    return Image.fromarray(clean),{'raw':raw,'komponenter':ids,'raw_boks':[x0,y0,x1,y1],'mal':size,'rgb_avrundingsrens':int(hit.sum())}

for pose in plan:
    name=pose['bilde']
    if name not in specs:continue
    spec=specs[name];_,_,w,h=pose['rekt'];im=Image.new('RGBA',(w*4,h*4));details=[]
    parts=spec.get('deler',[{'ids':spec.get('ids'),'rekt':[0,0,w,h]}])
    for part in parts:
        x,y,pw,ph=part['rekt'];draw,detail=drawing(spec['raw'],part['ids'],(pw*4,ph*4))
        if part.get('vend_vertikalt'):draw=draw.transpose(Image.Transpose.FLIP_TOP_BOTTOM)
        im.alpha_composite(draw,(x*4,y*4));details.append({**detail,'plassering_1x':part['rekt'],'kildedel':part.get('kildedel'),'vend_vertikalt':part.get('vend_vertikalt',False)})
    if name=='costume110_1949':
        # Originalene har identisk kropp, grep og kølle. Bare hodeområdet endres.
        old=next(p for p in plan if p['bilde']=='costume110_1942')
        def source_crop(p):
            x,y,w,h=p['rekt']
            return np.array(Image.open(ART/'jobber'/p['kildejobb']/'original_1x.png').convert('RGBA').crop((x,y,x+w,y+h)))
        a,b=source_crop(old),source_crop(pose);changed=(a!=b).any(2)
        yy,xx=np.nonzero(changed);assert [int(xx.min()),int(yy.min()),int(xx.max())+1,int(yy.max())+1]==[17,0,41,23]
        im.save(REPORT/'costume110_1949-for-felleskropp.png')
        mask=Image.new('L',im.size);ImageDraw.Draw(mask).rectangle((64,0,168,96),fill=255)
        weight=np.array(mask.filter(ImageFilter.GaussianBlur(1.25)),dtype=float)[:,:,None]/255
        base=np.array(prototypes['costume110_1942'],dtype=float);head=np.array(im,dtype=float)
        alpha=(1-weight)*base[:,:,3:]/255+weight*head[:,:,3:]/255
        premul=(1-weight)*base[:,:,:3]*base[:,:,3:]/255+weight*head[:,:,:3]*head[:,:,3:]/255
        rgb=np.divide(premul,alpha,out=np.zeros_like(premul),where=alpha>0)
        im=Image.fromarray(np.clip(np.concatenate([rgb,alpha*255],axis=2).round(),0,255).astype('uint8'))
        details.append({'felles_kropp':'costume110_1942-prototype.png','original_endringsboks':[17,0,41,23],
                        'myk_rektangulaer_hodeplassering':[64,0,168,96],'original_silhuettmaske':False,
                        'merknad':'Bare allerede genererte kropp/hode. Ingen originale piksler males inn.'})
    path=REPORT/(name+'-prototype.png');im.save(path);prototypes[name]=im
    audit.append({'bilde':name,'prototype_sha256':sha(path),'deler':details,'original_pikselmaske_brukt':False})

progress={'dato':'2026-10-19','verktøy':'innebygd image_gen','modell':'Eksakt modellnavn ikke vist',
          'faktiske_bildekall':['unik-A-raa01.png','unik-B-raa01.png','unik-D-raa01.png','unik-C-raa02.png','unik-G-raa01.png'],
          'leveranser':[],'merknad':'41 originale RGBA-utsnitt planlagt. 25 unike tegningsgrunnlag gir 64 leverte ruter i fem ark. D er bare utkast, E avvist, C1 avbrutt, F ikke bestilt. Delte kall skal ikke summeres per jobb.'}
for i in range(1,8):
    name=f'fig110_{i:02}';d=ART/'jobber'/name;j=json.loads((d/'jobb.json').read_text());out=Image.new('RGBA',(1536,1024));maps=[]

        assert hashlib.sha256(source.tobytes()).hexdigest()==pose['rgba_sha256']
        im=prototypes[pose['bilde']];assert im.size==(w*4,h*4)
        dx,dy=(j['plassering'][0]+x)*4,(j['plassering'][1]+y)*4;out.alpha_composite(im,(dx,dy))
        yy,xx=np.nonzero(np.array(im)[:,:,3]>128)
        maps.append({'bilde':cel['bilde'],'kostyme':110,'rute':cel['rute'],
                     'rect':[dx+int(xx.min()),dy+int(yy.min()),int(xx.max()-xx.min()+1),int(yy.max()-yy.min()+1)],
                     'representant':pose['bilde'],'kilde_rgba_sha256':pose['rgba_sha256'],
                     'prototype':pose['bilde']+'-prototype.png','raw':specs[pose['bilde']]['raw']})
    out.save(d/'resultat.png');digest=sha(d/'resultat.png')
    (REPORT/f'rutekart-{name}.json').write_text(json.dumps({name:{'sha256_resultat':digest,'ruter':maps}},ensure_ascii=False,indent=2)+'\n')
    progress['leveranser'].append({'jobb':name,'resultat':f'jobber/{name}/resultat.png','sha256':digest,'ruter':len(maps),
        'rutekart':f'rapporter/sam-golf-20261019/rutekart-{name}.json','delte_bildekall':sorted(set(m['raw'] for m in maps)),
        'egenkontroll':'ikke kjørt','native_mottatt':False,'visuelt_kontrollert':False,'motoravspilt':False,
        'importert':False,'godkjent_av_Tom':False,'claude_kontroll':False})
(REPORT/'prototype-kvitteringer.json').write_text(json.dumps(audit,ensure_ascii=False,indent=2)+'\n')
(REPORT/'fremdrift.json').write_text(json.dumps(progress,ensure_ascii=False,indent=2)+'\n')
