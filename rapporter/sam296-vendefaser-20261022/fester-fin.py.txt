from pathlib import Path
from PIL import Image, ImageDraw
from scipy import ndimage as nd
import numpy as np
import hashlib
import json
import math
import sys

O = Path(__file__).resolve().parent
K = O / 'kall' / sys.argv[1]
q = json.loads((K / 'verktøyresultat.json').read_text())
source = np.array(Image.open(K / 'original-native-kontroll.png').convert('RGBA'))
input_name = sys.argv[2] if len(sys.argv) > 2 else 'prototype-under-kontroll.png'
variant = '-fin'
own = Image.open(K / input_name).convert('RGBA')


def mask(image, kind):
    r, g, b = image[:, :, :3].astype(float).transpose(2, 0, 1)
    opaque = image[:, :, 3] >= 128
    if kind == 'hud':
        return opaque & (r > 160) & (g < 175) & (b < 155) & (b > .65 * g) & (r > 1.3 * g)
    if kind == 'blaa':
        return opaque & (r < 120) & (g < 130) & (b > 80) & (b > 1.3 * r) & (b > 1.2 * g)
    maximum = np.maximum.reduce([r, g, b])
    minimum = np.minimum.reduce([r, g, b])
    return opaque & (maximum < 100) & (maximum - minimum < 40)


point_sets = {}
for kind in ['hud', 'blaa']:
    y, x = np.where(mask(source, kind)[9::4, 9::4])
    point_sets[kind] = np.array([9.5 + y * 4, 9.5 + x * 4])

# Kilden brukes bare til kontroll-ROI-er og målepunkter, aldri til klipping.
dark_labels, count = nd.label(mask(source, 'mork'))
anchors = []
for label in range(1, count + 1):
    y, x = np.where(dark_labels == label)
    if len(y) < 16:
        continue
    rect = [max(0, int(x.min()) - 4), max(0, int(y.min()) - 4),
            min(source.shape[1], int(x.max()) + 5), min(source.shape[0], int(y.max()) + 5)]
    anchors.append(dict(navn=f'mork-{label}', kilde_areal=int(len(y)),
                        kilde_midt=[float(x.mean()), float(y.mean())], ROI_HD=rect))

rows = []
for dy in np.arange(-1.5, .5001, .125):
    for dx in np.arange(0, 2.001, .125):
        image = own.transform(own.size, Image.Transform.AFFINE, (1, 0, -float(dx), 0, 1, -float(dy)), Image.Resampling.BICUBIC)
        data = np.array(image)
        assert not data[:2, :, 3].any() and not data[-2:, :, 3].any() and not data[:, :2, 3].any() and not data[:, -2:, 3].any()
        assert abs(int(data[:, :, 3].sum()) / int(np.array(own)[:, :, 3].sum()) - 1) < .005
        colors = {}
        for kind, points in point_sets.items():
            own_mask = mask(data, kind)
            if points.shape[1] == 0:
                distances = np.array([0.0])
            elif not own_mask.any():
                distances = np.full(points.shape[1], 999.0)
            else:
                distances = nd.map_coordinates(nd.distance_transform_edt(~own_mask), points, order=1)
            colors[kind] = dict(kildepunkter=int(points.shape[1]),
                                maks_avstand_HD=float(distances.max()),
                                punkter_over3HD=int((distances > 3).sum()))
        measured = []
        for anchor in anchors:
            x0, y0, x1, y1 = anchor['ROI_HD']
            y, x = np.where(mask(data, 'mork')[y0:y1, x0:x1])
            if len(y):
                center = [float(x.mean() + x0), float(y.mean() + y0)]
                distance = math.dist(center, anchor['kilde_midt'])
            else:
                center, distance = None, 999.0
            measured.append(dict(**anchor, egen_midt=center, avvik_HD=distance))
        old_mask, new_mask = source[:, :, 3] >= 128, data[:, :, 3] >= 128
        overlap = float((old_mask & new_mask).sum() / (old_mask | new_mask).sum())
        coverage = float((old_mask & new_mask).sum() / old_mask.sum())
        worst = max([v['maks_avstand_HD'] for v in colors.values()] + [v['avvik_HD'] for v in measured])
        rows.append(dict(hel_plassering_HD=[dx, dy], maks_avvik_HD=worst,
                         farger=colors, morke_fester=measured, IoU=overlap, dekning=coverage))

rows.sort(key=lambda row: (row['maks_avvik_HD'], -row['IoU']))
best = rows[0]
dx, dy = best['hel_plassering_HD']
image = own.transform(own.size, Image.Transform.AFFINE, (1, 0, -float(dx), 0, 1, -float(dy)), Image.Resampling.BICUBIC)
native_name = 'prototype-festekontroll' + variant + '.png'
image.save(K / native_name)
passed = (q['tekniske_porter_bestatt'] and best['maks_avvik_HD'] <= 3
          and best['IoU'] >= .6 and best['dekning'] >= .8)
result = dict(**best, kall=q['kall'], rute=q['rute'], port_HD=3,
              hel_affin_port=q['hel_affin_port'], teknisk_og_fester_bestatt=passed,
              native_fil=native_name,
              native_sha256=hashlib.sha256((K / native_name).read_bytes()).hexdigest(),
              originalmaske=False, originalfragmenter=False, lokal_flytting=False,
              egen_alfa=True, visuelt_sett=False, native_mottatt=False, publisert=False)
(K / ('feste-alle-helplasseringer' + variant + '.json')).write_text(json.dumps(rows, ensure_ascii=False, indent=2) + '\n')
(K / ('sluttkandidat-kontroll' + variant + '.json')).write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
width = max(800, (image.width * 2 + 30) * 2)
board = Image.new('RGB', (width, max(710, image.height * 2 + 55)), '#304050')
draw = ImageDraw.Draw(board)
for i, (sprite, label) in enumerate([(Image.fromarray(source), 'Original, bare kontroll'),
                                      (image, q['kall'] + ', egen helplassering')]):
    background = Image.new('RGBA', sprite.size, '#87959e')
    background.alpha_composite(sprite)
    board.paste(background.convert('RGB').resize((sprite.width * 2, sprite.height * 2),
                                                 Image.Resampling.NEAREST), (12 + i * (width // 2), 34))
    draw.text((12 + i * (width // 2), 10), label, fill='white')
board.save(K / ('sluttkandidat-visuell' + variant + '.png'))
print(json.dumps(dict(kall=q['kall'], rute=q['rute'], bestatt=passed,
                      maks_HD=best['maks_avvik_HD'], shift=best['hel_plassering_HD'],
                      farger=best['farger'], morke_fester=len(anchors)), ensure_ascii=False))
