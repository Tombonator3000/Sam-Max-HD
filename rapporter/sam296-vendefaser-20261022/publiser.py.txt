def read(path):
    return json.loads(path.read_text())


def save(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(where, *args):
    return subprocess.check_output(['git', *args], cwd=where, text=True).strip()


def api(endpoint):
    return json.loads(subprocess.check_output(
        ['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/' + endpoint],
        cwd=P, text=True))


def remote(branch):
    return git(P, 'ls-remote', 'origin', 'refs/heads/' + branch).split()[0]


def inventory(root):
    paths = [p for p in root.rglob('*')
             if p.is_file() and '.git' not in p.relative_to(root).parts]
    def entry(path):
        return path.relative_to(root).as_posix(), sha(path)
    with ThreadPoolExecutor(max_workers=8) as pool:
        return dict(pool.map(entry, paths))


def copy_reflink(source, target):
    assert not target.exists(), str(target)
    subprocess.run(['cp', '-a', '--reflink=always', str(source), str(target)], check=True)


def check_protected(start):
    assert len(start['beskyttede']) == 1448
    assert all(sha(A / name) == digest for name, digest in start['beskyttede'].items())
    assert git(P, 'stash', 'list', '--format=%H').splitlines() == start['stash']
