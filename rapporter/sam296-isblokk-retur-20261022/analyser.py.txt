def read(path):
    return json.loads(path.read_text())

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

@lru_cache(maxsize=16)
def picture(path, flat=False):
    path = Path(path)
    if flat:
        path = path.with_name(path.stem + '_flat.png')
    return np.array(Image.open(path).convert('RGB'))

def rectangle(event, padding=8):
    x, y = event['pos']
    width, height = event['maal']
    return [max(0, x*4-padding), max(0, y*4-padding),
            min(1280, (x+width)*4+padding), min(800, (y+height)*4+padding)]

def buffer_details(proof, directory):
    result = []
    for check in proof['kontroller']:
        if not check['avvik']:
            continue
        frame = check['frame']
        screen = np.array(Image.open(directory / f'frame_{frame:06}_screen.png').convert('RGB'))
        files = [f for f in directory.glob(f'frame_{frame:06}_room*.png')
                 if not f.stem.endswith('_flat')]
        assert len(files) == 1
        whole = np.array(Image.open(files[0]).convert('RGB'))
        y, x = np.where(np.any(screen != whole, axis=2))
        assert len(x) == check['avvik']
        digest = hashlib.sha256(np.stack([x,y],1).tobytes()
                                +screen[y,x].tobytes()+whole[y,x].tobytes()).hexdigest()
        result.append(dict(frame=frame, avvik=len(x), fil=files[0].name,
                           koordinater_og_begge_RGB_sha256=digest,
                           foer_testaktor296=frame<40))
    return result
