def drawing(raw,ids,size):
    a,labels=load(raw);core=np.isin(labels,ids);yy,xx=np.nonzero(core);assert len(xx)>20
    x0,y0,x1,y1=int(xx.min()),int(yy.min()),int(xx.max())+1,int(yy.max())+1
    xa,ya,xb,yb=max(0,x0-2),max(0,y0-2),min(a.shape[1],x1+2),min(a.shape[0],y1+2)
    local=core[ya:yb,xa:xb];cut=a[ya:yb,xa:xb].copy()
    distance,nearest=nd.distance_transform_edt(~local,return_indices=True);permitted=distance<=1.5
    cut[:,:,3][~permitted|((labels[ya:yb,xa:xb]>0)&~local)]=0
    edge=permitted&~local;cut[:,:,:3][edge]=cut[nearest[0][edge],nearest[1][edge],:3]
    im=Image.fromarray(cut).crop((x0-xa,y0-ya,x1-xa,y1-ya)).resize(size,Image.Resampling.LANCZOS)
    z=np.array(im);hit=(z[:,:,:3]==[255,0,255]).all(2)&(z[:,:,3]>0)

def rope_placement(im,source):
    """Myk posisjonsendring av generert strikk, aldri klipping med kilde-alfa."""
    from scipy.interpolate import PchipInterpolator
    a=np.array(im).astype(float);h,w=a.shape[:2]
    colored=(source[:,:,1]>150)&(source[:,:,2]<120)&(source[:,:,3]>128)
    ys=np.flatnonzero(colored.any(1));assert len(ys)>50
    centers=np.array([np.where(colored[y])[0].mean()+0.5 for y in ys])
    centers=nd.gaussian_filter1d(centers,0.8,mode='nearest')
    interpolate=PchipInterpolator((ys+0.5)*4-0.5,centers*4-0.5,extrapolate=True)
    destination=interpolate(np.arange(h))
    target_width=float(np.median(colored[ys].sum(1))*4)
    gen=(a[:,:,1]>130)&(a[:,:,2]<120)&(a[:,:,3]>220)
    genrows=np.flatnonzero(gen.any(1));assert len(genrows)>180
    start=max(float((ys[0]+0.5)*4),float(genrows[0]+4))
    gx=[];gw=[]
    for y in range(h):
        pixels=np.flatnonzero(a[y,:,3]>220)
        gx.append((pixels[0]+pixels[-1])/2 if len(pixels) else w/2)
        gw.append(max(1,len(pixels)))
    gx=nd.gaussian_filter1d(np.array(gx),1);gw=nd.gaussian_filter1d(np.array(gw,dtype=float),1)
    yy,xx=np.mgrid[:h,:w];weight=np.clip((np.arange(h)-start)/12,0,1)
    mapped=(xx-destination[:,None])/(target_width/gw[:,None])+gx[:,None]
    mapped=xx*(1-weight[:,None])+mapped*weight[:,None]
    out=np.empty_like(a)
    for c in range(4):out[:,:,c]=nd.map_coordinates(a[:,:,c],[yy,mapped],order=1,mode='constant',cval=0)
    return Image.fromarray(np.uint8(np.clip(np.rint(out),0,255))),dict(metode='Myk horisontal plassering langs kildebekreftet strikksenter, PCHIP og glattet posisjon. Bare generert RGBA flyttes; kilde-alfa klipper aldri resultatet.',strikkbredde_hd=target_width,start_rad=start)
