from pathlib import Path
import hashlib, json
from datetime import datetime
from zoneinfo import ZoneInfo
from PIL import Image, ImageDraw
import numpy as np

root = Path(__file__).resolve().parents[2]
r = root / 'rapporter/bestilling002-20261013'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
sources = json.loads((r / 'kildehash.json').read_text())
assert len(sources) == 40
for s in sources:
    assert sha(root / s['fil']) == s['sha256'], s['fil']
jobs = ['obj036_01','rom002','obj050_01','rom003','rom004','rom005','rom006','rom008','rom009_del1av2','rom009_del2av2']
records = {}
for g in 'abc':
    for item in json.loads((r / f'{g}-leveranser.json').read_text())['leveranser']:
        assert item['jobb'] not in records
        records[item['jobb']] = item
assert set(records) == set(jobs)
out = []
checks = []
for j in jobs:
    entry = records[j]
    p = root / entry['resultat']
    assert sha(p) == entry['sha256'], j
    im = Image.open(p)
    assert im.size == (1536,1024), j
    meta = json.loads((root / 'jobber' / j / 'jobb.json').read_text())
    x,y,w,h = meta['bilde_i_lerret']
    arr = np.asarray(im.convert('RGB'))
    outside = np.ones((1024,1536), dtype=bool)
    outside[y:y+h,x:x+w] = False
    assert np.all(arr[outside] == 128), j
    cp = root / entry.get('kontrollfil', entry.get('egenkontroll_logg', f'rapporter/bestilling002-20261013/{j}-kontroll.txt'))
    assert 'Utfall: OK' in cp.read_text(), cp
    assert (root/'jobber'/j/'notat.md').is_file()
    item = dict(entry)
    item.update(egenkontroll='OK', kontrollfil=str(cp.relative_to(root)),
                visuelt_godkjent=False, stil_godkjent_av_Tom=True,
                importert=False, runtime_verifisert=False,
                storrelse=list(im.size), format=im.mode)
    out.append(item)
    checks.append({'jobb':j,'sha256':sha(p),'format':im.mode,'storrelse':list(im.size),
                   'padding_rgb128':True,'egenkontroll':'OK'})
a = np.asarray(Image.open(root/'jobber/rom009_del1av2/resultat.png').convert('RGB'))[112:912]
b = np.asarray(Image.open(root/'jobber/rom009_del2av2/resultat.png').convert('RGB'))[112:912]
assert np.array_equal(a[:,384:], b[:,:1152])
stamp = datetime.now(ZoneInfo('Europe/Oslo')).isoformat(timespec='seconds')
manifest = {'dato':stamp,'gren':'gpt-arbeid','produksjonsagent':'gpt-6-astra',
            'verktøy':'Innebygd ChatGPT-bildegenerator og autorisert presis bildebehandling',
            'modell':'Eksakt bildemodellnavn ikke eksponert','leveranser':out}
(r/'fremdrift.json').write_text(json.dumps(manifest,ensure_ascii=False,indent=2)+'\n')
summary = {'dato':stamp,'antall_leveranser':len(out),'beskyttede_kilder_uendret':len(sources),
           'rom009_overlapp':{'bredde':1152,'hoyde':800,'byte_identisk':True},'leveranser':checks,
           'motorimport':False,'runtime':False,'ny_visuell_brukergodkjenning':False}
(r/'sluttkontroll.json').write_text(json.dumps(summary,ensure_ascii=False,indent=2)+'\n')
contact = Image.new('RGB',(1536,5*286),(210,213,218))
d = ImageDraw.Draw(contact)
for row in range(5):
    for col in range(2):
        j=jobs[row*2+col]
        for k,f in enumerate(['referanse.png','resultat.png']):
            im=Image.open(root/'jobber'/j/f).convert('RGB');im.thumbnail((384,512))
            x=col*768+k*384;y=row*286
            contact.paste(im,(x,y+20))
            d.text((x+7,y+4),j+(' ORIGINAL' if k==0 else ' HD'),fill=(25,25,25))
contact.save(r/'samlet-kontaktark.png')
print(json.dumps({'leveranser':len(out),'kilder_uendret':len(sources),'overlapp':'1152x800 identisk'}))
