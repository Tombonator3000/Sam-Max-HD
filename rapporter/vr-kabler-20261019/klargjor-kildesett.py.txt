"""Pakk kun eksakt unike originalruter som tegningsgrunnlag.

Ingen beskyttet fil endres. Alle 62 produksjonsruter beholdes i rutekoblingen.
"""
from pathlib import Path
from PIL import Image
import json, hashlib

ART = Path('/home/tom/Sam-Max-HD/work/.gpt-gren')
REPORT = ART/'rapporter/vr-kabler-20261019'
plan = []
unique = {}
for i in range(1,11):
    name = f'fig246_{i:02}'
    job = json.loads((ART/'jobber'/name/'jobb.json').read_text())
    original = Image.open(ART/'jobber'/name/'original_1x.png').convert('RGBA')
    for cel in job['ruter']:
        x,y,w,h = cel['rekt']
        source = original.crop((x,y,x+w,y+h))
        digest = hashlib.sha256(source.tobytes()).hexdigest()
        key = (w,h,digest)
        if key not in unique:
            unique[key] = len(plan)
            plan.append({'id':cel['bilde'], 'kildejobb':name, 'kilderekt':[x,y,w,h],
                         'kilde_rgba_sha256':digest, 'mottakere':[]})
        plan[unique[key]]['mottakere'].append({'jobb':name, 'bilde':cel['bilde'], 'rekt':[x,y,w,h]})
assert len(plan) == 19 and sum(len(p['mottakere']) for p in plan) == 62
groups = [('A',plan[:4]), ('B',plan[4:9]), ('C',plan[9:])]
for label, poses in groups:
    canvas = Image.new('RGB',(1536,1024),(128,128,128))
    canvas.paste((255,0,255),(64,32,1472,992))
    x,y = 4,4
    for i, pose in enumerate(poses):
        if label == 'C' and i == 5:
            x,y = 4,124
        sx,sy,w,h = pose['kilderekt']
        src = Image.open(ART/'jobber'/pose['kildejobb']/'original_1x.png').convert('RGB')
        crop = src.crop((sx,sy,sx+w,sy+h))
        assert x+w <= 352 and y+h <= 240
        canvas.paste(crop.resize((w*4,h*4),Image.Resampling.NEAREST),(64+x*4,32+y*4))
        pose['tegneark'] = label
        pose['tegneboks_1x'] = [x,y,w,h]
        x += w+8
    canvas.save(REPORT/f'unik-{label}-referanse.png')
(REPORT/'ruteplan-unik.json').write_text(json.dumps({'unike':19,'produksjonsruter':62,'ruter':plan},ensure_ascii=False,indent=2)+'\n')
print('19 unike ruter fordelt på tre tegningsgrunnlag; alle 62 målruter bevart.')
