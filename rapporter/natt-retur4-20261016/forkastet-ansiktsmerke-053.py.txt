from pathlib import Path
from PIL import Image,ImageFilter
import numpy as np,json
p=Path('rapporter/natt-retur4-20261016');a=Image.open('jobber/lag053_01_del1av2/resultat.png').convert('RGB');b=Image.open('jobber/lag053_02_del1av2/resultat.png').convert('RGB');a.save(p/'lag053_01_del1av2-for-ansiktsmaal.png')
# Det lille separate munnmerket over den brede underkjeven er tre ganger to
# originalpiksler. Det nytegnede merket renses og plasseres i 12 x 8 HD-piksler.
roi=(960,566,985,581);part=a.crop(roi).convert('RGBA');aa=np.array(part);mask=aa[:,:,:3].sum(2,dtype=int)<45;yy,xx=np.where(mask);box=(int(xx.min()),int(yy.min()),int(xx.max()+1),int(yy.max()+1))
part.putalpha(Image.fromarray((mask*255).astype('uint8')).filter(ImageFilter.GaussianBlur(.3)));part=part.crop(box).resize((12,8),Image.Resampling.LANCZOS)
clean=b.crop(roi);a.paste(clean,roi[:2]);a.paste(part,(964,568),part);a.save('jobber/lag053_01_del1av2/resultat.png')
(p/'ansiktsmerke-053.json').write_text(json.dumps({'jobb':'lag053_01_del1av2','bilde':'obj508_01','originalens_separate_munnmerke':[241,114,3,2],'maal_HD':[964,568,12,8],'kilde':'nytegnet gjesteutsnitt, forsok4','rensing_flate':'nytegnet samme ansikt fra lag02; ikke originalpiksler','bredt_munnfelt_under':'uendret'},ensure_ascii=False,indent=2)+'\n')
