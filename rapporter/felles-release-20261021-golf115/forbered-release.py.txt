from pathlib import Path
import json
import hashlib
import subprocess
import shutil

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
S = A / 'rapporter/sam115-golf-20261021'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
api = lambda x: json.loads(subprocess.check_output(['gh', 'api', 'repos/Tombonator3000/Sam-Max-HD/' + x], cwd=P, text=True))
save = lambda p, d: p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')
state = read(O / 'bygg-startet.json')
tag, code, mod = (state[k] for k in ['release_tag', 'kode', 'hd_mod'])
manifest = read(R / 'moddfiler-3098.json')
old = read(A / 'rapporter/felles-release-20261021-max052-delvis/moddfiler-3013.json')
assert len(manifest) == 21341 and all(sha(P / 'mods/gpt' / f) == h for f, h in manifest.items())
assert set(old) <= set(manifest) and all(manifest[f] == h for f, h in old.items() if f.endswith('.png'))
protected = read(A / 'rapporter/sam304-kilde-20261021/beskyttede-samlet1096.json')
assert len(protected) == 1096 and all(sha(A / f) == h for f, h in protected.items())
base = read(S / 'grunnlag-produksjon.json')
assert subprocess.check_output(['git', 'stash', 'list', '--format=%H'], cwd=P, text=True).splitlines() == base['stash']
assert len(list((A / 'jobber').glob('fig*/resultat.png'))) == 427
motor = read(R / 'linux-kontroll.json')
files, assets = read(O / 'moddfiler-kontroll.json'), read(O / 'assetkontroll.json')
assert motor['native'] == 3098 and motor['nye_native_observert'] == 85
assert motor['runder'] == motor['grunnlagsrunder_samme_binar'] == 8
assert motor['skjermbufferkontroller'] == 579 and motor['nullkontroller'] == 578 and not motor['nye_eller_oekte_avvik']
assert files['moddfiler_per_pakke'] == 21340 and files['begge_modder_byteidentiske'] and files['begge_faktiske_nedlastinger_verifisert']
assert files['kode'] == code and files['hd_mod'] == mod and len(assets) == 4
for v in assets:
    assert (O / v['plattform'] / v['fil']).stat().st_size == v['bytes'] and sha(O / v['plattform'] / v['fil']) == v['sha256']
shots = read(R / 'motorbilder.json')
assert len(shots) == 12 and all(sha(R / v['fil']) == v['sha256'] for v in shots)
assert read(R / 'motorobservasjoner85.json')['sett'] == 85
rasters = read(R / 'rasterkontroll-mot-visuelt-sett.json')
assert rasters['alle_nativefiler_identiske'] and rasters['croprasteridentiske'] == 85
assert all(v['utenfor_alle_loggede_aktorer_og_marg'] == 0 for v in read(R / 'pakke-pikselkontroll.json'))
assert api('git/ref/heads/hd-mod')['object']['sha'] == mod
save(R / 'visuell-kontroll.json', dict(hele_motorbilder_sett=12, foer_etter_sett=4,
    native_varianter_sammenlignet=85, nativefiler_og_cropraster_identiske_med_sett_produksjon=True,
    sammensatte_golf_lag_i42_produksjonsopptak_sett=True, andre_aktorer_ikke_skjult=True,
    ny_Tom_Claude_godkjenning=False, naturlig_gjennomspilling=False, interaktiv_windows_test=False,
    ny_webtest=False, filer={v['fil']: v['sha256'] for v in shots}))
lr, wr = (state['plattform'][p]['run_id'] for p in ['linux', 'windows'])
sources = {folder: read(A / 'rapporter' / folder / 'mottak-publisert.json')['grafikkilde']
    for folder in ['max052-kattslutt-20261021', 'sam115-golf-20261021']}
body = f'''Linux og Windows inneholder **3098 native HD-bildelag**, med **85 nye animasjonsruter** siden forrige fellespakke: den siste katteposen ved kontoret og 84 ruter med Sam, golfball og fisk på Gator Golf. Tidligere grafikk og rettinger er bevart. Resten av HD-grafikken er under arbeid.

Last ned **SamMaxHD-Windows.zip** for Windows eller **SamMaxHD-x86_64.AppImage** for Linux. Gjør Linux-filen kjørbar og start den. Ctrl+H bytter grafikk, og F5 åpner originalmenyen.

Begge faktiske nedlastinger, SHA-filer og alle **21340 runtime-moddfiler per pakke** er kontrollert. Moddfilene er byteidentiske med publisert modd. Linux er prøvd med inventar, akvarium, cafeteria, katten i rom 9 og fire kildebekreftede golfanimasjoner i rom 25. Alle 85 nye native ruter er logget i motoren. Alle 85 nye native PNG-filer og 85 av 85 figurutsnitt er identiske med visuelt kontrollert produksjon. Tolv hele pakkebilder og fire før/etter-par er sett.

579 skjermbufferkontroller ga 578 null i både pakken og 3013-grunnlaget med samme nye Linux-binær. Det kjente avviket på 160 piksler ved bilde 36 finnes før figurstart i katteprøven. Alle 51 kontroller fra figurstart i denne og alle 480 golfkontroller er null. Ingen nye eller økte avvik. Hele prøven er ikke avviksfri. Fire før/etter-par har null endringer utenfor alle loggede aktørflater med målemargin. Andre aktører og romskript er ikke skjult.

Fire nye hele figurark er fullført siden forrige fellespakke: fig052_01 og fig115_01..03. 427 av 1758 hele figurjobber har resultat; 1331 gjenstår pluss returer, katalogfeil og samlet sluttkontroll. Golfark 01 og 02 er OK; golfark 03 og kattearket har visuelt vurderte omriss-/størrelsesvarsler. Råbilder og faktiske prompter er bevart. Ekte egen alfa beholdes; ingen hard originalmaske, kildefragmenter eller senkede native grenser inngår i de nye rutene. Den løse svarte kildeprikken i slipset, fire små blågrå merker og korrekte ball-/fiskegrep er kontrollert.

Prøvene tvinger bestemte kildekorrekte animasjoner. Full naturlig gjennomspilling, interaktiv Windows-test og ny webtest gjenstår. Historiske testhoppstans, inventarkant i rom 35, Windows SDL-startfeil og inventarikonets RGB under 044-eksplosjonen er fortsatt åpne. Den nye Windows-pakken passerte sitt første byggforsøk; dette avslutter ikke den kjente sporadiske SDL-feilen. Avviste motiver og ugyldige 344/346-ruter er urørt. Ingen ny Tom-/Claude-godkjenning eller motor-/pipelineendring.

Kode: {code}
hd-mod: {mod}
Grafikkilder: {sources}
Linux: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{lr}
Windows: https://github.com/Tombonator3000/Sam-Max-HD/actions/runs/{wr}
Kontrollrapport: https://github.com/Tombonator3000/Sam-Max-HD/blob/gpt-arbeid/rapporter/{R.name}/RAPPORT.md

Spillet og grafikken tilhører Disney/Lucasfilm.
'''
(O / 'utgivelse-notat.md').write_text(body)
shutil.copy2(O / 'utgivelse-notat.md', R / 'utgivelse-notat.md')
prior = api('releases?per_page=100')
before = read(O / 'releaser-foer-bygg.json')
assert not any(v['tag_name'] == tag for v in prior)
assert {v['id'] for v in before} <= {v['id'] for v in prior}
aset = lambda v: {(a['id'], a['name'], a['size'], a['digest']) for a in v['assets']}
for v in before:
    if v['tag_name'] not in ['linux', 'windows']:
        assert aset(v) == aset(next(x for x in prior if x['id'] == v['id']))
for D in [O, R]:
    save(D / 'releaser-foer.json', prior)
save(R / 'bevaringskontroll.json', dict(beskyttede_kilder_SHA_uendret=1096, tre_stasher_uendret=True,
    alle_tidligere_filstier_bevart=True, alle_tidligere_png_byteidentiske=True, filer3098=21341,
    figurjobber_med_resultat=427, tidligere_felles3013_og_assets_bevart=True,
    rullerende_linux_windows_oppdatert_med_autoriserte_bygg=True))
(R / 'RAPPORT.md').write_text(body + f'\nLinux-binær SHA256 {motor["binar_sha256"]}. Alle {len(prior)} tidligere releaser og gjeldende vedlegg bevares ved fellespublisering. De rullerende linux-/windows-taggenes innhold er oppdatert av de autoriserte byggene; tidligere fellesutgaver er bevart. Nativefrys frigjøres først etter bekreftet fellespublisering.\n')
(R / 'PAAGAR.md').write_text('Begge faktiske pakker er kontrollert: 3098 native, 85 nye, alle 21340 runtimefiler per pakke og fire nedlastingsdigester bekreftet. 578/579 null, samme kjente 160 ved 36, ingen nye eller økte. 85 av 85 figurutsnitt er identiske med visuelt sett produksjon. Publisering av fellesrelease og Latest gjenstår. Nativefrysen gjelder fortsatt.\n')
for p in O.glob('*.py'):
    shutil.copy2(p, R / (p.name + '.txt'))
print('Begge faktiske pakker, 21340 filer hver, 85 nye, 578/579 null og visuell kontroll bekreftet.', flush=True)
