def crop(v):
    x, y = v['pos']
    w, h = v['maal']
    return np.array(Image.open(v['fil']).convert('RGBA').crop((x * 4 - 8, y * 4 - 8, (x + w) * 4 + 8, (y + h) * 4 + 8)))

for e in expected:

diffs = []
for name in ['costume303_1920','costume303_1932','costume304_1920','costume304_1929']:
    v = obs[name]
    T = O / ('grunnlag3098-' + v['label'])
    before = np.array(Image.open(T / Path(v['fil']).name).convert('RGB'))
    after = np.array(Image.open(v['fil']).convert('RGB'))
    delta = np.any(before != after, 2)
    outside = delta.copy()
    states = []
    for path in [T / 'scummvm.log', Path(v['fil']).parent / 'scummvm.log']:
        rects = []
        for m in re.finditer(r'frame (\d+) soft cel\s+(\d+) costume\s*(\d+)\b.*? at (-?\d+),(-?\d+) (\d+)x(\d+)', path.read_text()):
            f, cel, costume, x, y, w, h = map(int, m.groups())
            if v['bildeframe'] - 4 <= f < v['bildeframe']:
                rect = [max(0, x * 4 - 12), max(0, y * 4 - 12), min(1280, (x + w) * 4 + 12), min(800, (y + h) * 4 + 12)]
                outside[rect[1]:rect[3], rect[0]:rect[2]] = False
                rects.append(dict(frame=f, cel=cel, kostyme=costume, rekt=rect))
        states.append(dict(fil=str(path), flater=rects))
    diffs.append(dict(bilde=name, frame=v['bildeframe'], endrede_piksler=int(delta.sum()),
        utenfor_alle_loggede_aktorer_og_marg=int(outside.sum()), aktortilstander=states))
    board = Image.new('RGB', (1280, 430), '#283943')

    board.paste(Image.fromarray(after).resize((640, 400)), (640, 30))
