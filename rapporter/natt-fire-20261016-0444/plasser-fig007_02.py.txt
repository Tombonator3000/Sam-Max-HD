from pathlib import Path
import json
import numpy as np
from PIL import Image
from scipy import ndimage

p = Path('rapporter/natt-fire-20261016-0444')
j = json.loads(Path('jobber/fig007_02/jobb.json').read_text())
a = np.array(Image.open(p/'fig007_02-nytegning.png').convert('RGBA'))
original = np.array(Image.open('jobber/fig007_02/original_1x.png').convert('RGB'))
labels, n = ndimage.label(a[:,:,3] > 128)
components = []
for i, s in enumerate(ndimage.find_objects(labels)):
    if s is not None and np.sum(labels[s] == i+1) > 50:
        components.append({'label': i+1, 'box': [s[1].start,s[0].start,s[1].stop,s[0].stop]})
rows = [sorted([c for c in components if lo <= c['box'][1] < hi], key=lambda c:c['box'][0]) for lo,hi in [(0,225),(225,390),(390,530),(530,640),(640,1024)]]
assert [len(r) for r in rows] == [10,3,2,10,5]
canvas = Image.new('RGBA',(1536,1024))
record=[]

def insert(c, target, name):
    x0,y0,x1,y1=c['box']; margin=3
    piece=a[y0-margin:y1+margin,x0-margin:x1+margin].copy()
    local=labels[y0-margin:y1+margin,x0-margin:x1+margin]
    keep=ndimage.binary_dilation(local==c['label'],iterations=2)
    piece[:,:,3][~keep]=0
    piece[:,:,3][piece[:,:,3]>=250]=255
    crop=Image.fromarray(piece)
    tx0,ty0,tx1,ty1=target
    scale=((tx1-tx0)/(x1-x0),(ty1-ty0)/(y1-y0))
    crop=crop.resize((round(crop.width*scale[0]),round(crop.height*scale[1])),Image.Resampling.LANCZOS)
    xy=(round(tx0-margin*scale[0]),round(ty0-margin*scale[1]))
    canvas.alpha_composite(crop,xy)
    record.append({'bilde':name,'kildeboks':c['box'],'maalboks':target,'skala':scale,'plassering':xy})

for k,(frame,c) in enumerate(zip(j['ruter'][:15],rows[0]+rows[1]+rows[2])):
    x0,y0,x1,y1=frame['synlig']; ox,oy=j['plassering']
    insert(c,[(ox+x0)*4,(oy+y0)*4,(ox+x1)*4,(oy+y1)*4],frame['bilde'])

for i,frame in enumerate(j['ruter'][15:20]):
    x0,y0,x1,y1=frame['synlig']; ox,oy=j['plassering']
    mask=np.any(original[y0:y1,x0:x1]!=[255,0,255],axis=2)
    ys,xs=np.nonzero(mask)
    # De to separate delene følger originalens egne bokser og innbyrdes avstand.
    split=(x1-x0)//2
    for part,select in enumerate([xs<split,xs>=split]):
        sx,sy=xs[select],ys[select]
        target=[(ox+x0+int(sx.min()))*4,(oy+y0+int(sy.min()))*4,(ox+x0+int(sx.max())+1)*4,(oy+y0+int(sy.max())+1)*4]
        insert(rows[3][i*2+part],target,frame['bilde']+f'-del{part+1}')

for frame,c in zip(j['ruter'][20:],rows[4]):
    x0,y0,x1,y1=frame['synlig']; ox,oy=j['plassering']
    insert(c,[(ox+x0)*4,(oy+y0)*4,(ox+x1)*4,(oy+y1)*4],frame['bilde'])

canvas.save('jobber/fig007_02/resultat.png')
(p/'fig007_02-plassering.json').write_text(json.dumps(record,indent=2)+'\n')
