read=lambda p:json.loads(p.read_text());sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
b=read(R/'baseline.json');c=read(R/'kandidat.json');native=read(R/'egen-native.json')['ruter'];cache={};rows=[]
def phase(e):

 if not p.exists():return None
 if p not in cache:cache[p]=sha(p)
 return cache[p]
for t in native:
 cel=int(t['bilde'].split('_')[1]);bm={};cm={}
 for source,target in [(b,bm),(c,cm)]:

 if pairs:row.update(grunnlag=pairs[0][0],kandidat=pairs[0][1],originalflat_sha256=pairs[0][2])
 rows.append(row)
result=dict(regel='Samme kildecel, posisjon, mål og SHA256 av hele faktiske originalflaten. (R/'fasekart.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
