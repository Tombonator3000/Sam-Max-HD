from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
import subprocess, shutil
P=Path('/home/tom/Sam-Max-HD'); O=Path(__file__).resolve().parent
R=P/'work/.gpt-gren/rapporter'/O.name
def run(label):
    name='motor12-seed-'+label
    text=(O/'kilde12-hel.py').read_text().replace('kildemotor12-hel',name).replace('kilde12-hel.py.txt',name+'.py.txt')
    text=text.replace("'--debuglevel=1'", "'--random-seed=20261021','--debuglevel=1'")
    text=text.replace('proof=dict(', 'proof=dict(random_seed=20261021,')
    if label=='kandidat3208':text=text.replace("str(P/'mods/gpt')", "str(O/'mod-intro3208')")
    f=O/(name+'.py'); f.write_text(text)
    q=subprocess.run([str(P/'.venv/bin/python'),str(f)],cwd=P,capture_output=True,text=True,timeout=150)
    (R/(name+'-utskrift.txt')).write_text(q.stdout+q.stderr)
    print(label,q.returncode,q.stdout.strip(),flush=True)
    assert q.returncode==0
with ThreadPoolExecutor(max_workers=2) as pool:list(pool.map(run,['grunnlag3150','kandidat3208']))
shutil.copy2(__file__,R/'motor12-seed.py.txt')
