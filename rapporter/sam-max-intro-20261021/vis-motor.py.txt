from pathlib import Path
from PIL import Image, ImageDraw
from collections import Counter
import json, hashlib, shutil, numpy as np

P = Path('/home/tom/Sam-Max-HD')
A = P / 'work/.gpt-gren'
O = Path(__file__).resolve().parent
R = A / 'rapporter' / O.name
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
save = lambda p, d: p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')

def artifacts(name):
    proof = read(R / (name + '.json'))
    assert proof['exit'] == 0 and len(proof['kontroller']) == 1300
    rows = []
    for e in proof['kontroller']:
        if not e['avvik']:
            continue
        f = e['frame']
        screen = O / name / f'frame_{f:06}_screen.png'
        whole = O / name / f'frame_{f:06}_room002.png'
        x = np.array(Image.open(screen).convert('RGB'))
        y = np.array(Image.open(whole).convert('RGB'))
        yy, xx = np.where(np.any(x != y, axis=2))
        assert len(xx) == e['avvik'] == 12
        bbox = [int(xx.min()), int(yy.min()), int(xx.max()+1), int(yy.max()+1)]
        assert bbox == [544, 620, 568, 639]
        signature = hashlib.sha256(np.stack([xx, yy], 1).tobytes()
                                   + x[yy, xx].tobytes() + y[yy, xx].tobytes()).hexdigest()
        rows.append(dict(frame=f, avvik=12, bbox=bbox, signatur=signature,
                         skjerm_sha256=sha(screen), hel_sha256=sha(whole)))
    return proof, rows

base, ba = artifacts('motor-intro-seed-grunnlag3150')
new, ca = artifacts('motor-intro-seed-kandidat3208')

assert len(secondary['kontroller']) == 120 and secondary['nullkontroller'] == 119

pairs = []
selected = {}
for name in names:
    natural = name in new['tegninger']
    c = (new if natural else secondary)['tegninger'][name]
    b = (base if natural else secondary_base)['tegninger'][name]
    assert c['pos'] == b['pos'] and c['maal'] == b['maal']
    cp, bp = Path(c['fil']), Path(b['fil'])
    cf, bf = cp.with_name(cp.stem+'_flat.png'), bp.with_name(bp.stem+'_flat.png')
    assert sha(cf) == sha(bf), (name, 'ulikt originalbilde')
    before = np.array(Image.open(bp).convert('RGB'))
    after = np.array(Image.open(cp).convert('RGB'))
    delta = np.any(before != after, 2)
    outside = delta.copy()
    x, y = c['pos']; w, h = c['maal']
    box = [max(0,x*4-8), max(0,y*4-8), min(after.shape[1],(x+w)*4+8), min(after.shape[0],(y+h)*4+8)]
    outside[box[1]:box[3], box[0]:box[2]] = False
    assert not outside.any(), (name, int(outside.sum()))
    row = dict(bilde=name, faktisk_linux=True, naturlig_intro=natural,
               rettet_hel_animasjon=not natural, grunnlag_frame=b['bildeframe'],
               kandidat_frame=c['bildeframe'], original_flat_sha256=sha(cf),
               grunnlag_sha256=sha(bp), kandidat_sha256=sha(cp),
               native_sha256=sha(O/'mod-intro3208/costumes'/(name+'_hd.png')),
               rekt_med_8hd_marg=box, endrede_piksler=int(delta.sum()),
               utenfor_ny_aktorflate=0)
    pairs.append(row); selected[name] = (b, c)

shots = []
boards = []
for start in range(0, len(canonical), 6):
    board = Image.new('RGB', (1320, 1050), '#304050'); draw = ImageDraw.Draw(board)
    for j, name in enumerate(canonical[start:start+6]):
        b, c = selected[name]
        x, y = c['pos']; w, h = c['maal']
        box = (x*4-8, y*4-8, (x+w)*4+8, (y+h)*4+8)
        xx = (j % 2)*660; yy = (j//2)*350
        draw.text((xx+8, yy+8), name+' original / egen HD i faktisk motor', fill='white')
        for k, row in enumerate([b,c]):
            cut = Image.open(row['fil']).convert('RGB').crop(box)
            cut.thumbnail((320,310), Image.Resampling.LANCZOS)
            board.paste(cut,(xx+8+k*330,yy+30))
        f = R/(name+'-faktisk-linux.png')
        shutil.copy2(c['fil'], f)
        shots.append(dict(bilde=name, fil=f.name, sha256=sha(f), frame=c['bildeframe']))
    f = R/f'motor-varianter-{start//6+1}.png'; board.save(f)
    boards.append(dict(fil=f.name, sha256=sha(f), ruter=canonical[start:start+6]))
