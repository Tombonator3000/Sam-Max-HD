"""Normaliserer NY imagegen-grafikk, med Toms ja 15. oktober.

Originalen brukes til identitet, objektbokser og mål, aldri som bildemateriale
eller konturmaske. Ingen beskyttede jobbfiler endres.
"""
from pathlib import Path
import hashlib
import json
import numpy as np
from PIL import Image
from scipy import ndimage

ROOT=Path(__file__).resolve().parents[2]
REPORT=Path(__file__).resolve().parent
BG=np.array([0,177,64],dtype=float)
corrections={e['bilde']:e for e in json.loads((REPORT/'retting12-kart.json').read_text())}
proof=[]
exports=[]

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def clean(im):
    a=np.array(im.convert('RGB'),dtype=float)
    # Bare bakgrunnsfargen som henger sammen med kanten fjernes. Grønt
    # INNE I et motiv beholdes, også når det ligner arkets nøkkelfarge.
    distance=np.linalg.norm(a-BG,axis=2)
    near=distance<70
    labels,_=ndimage.label(near)
    edge=np.unique(np.concatenate([labels[0],labels[-1],labels[:,0],labels[:,-1]]))
    flood=np.isin(labels,edge[edge!=0])
    actual_bg=np.median(a[flood],axis=0) if flood.any() else BG
    alpha=np.full(a.shape[:2],255.,dtype=float)
    fringe=flood & ndimage.maximum_filter(~flood,size=3)
    alpha[flood]=0
    alpha[fringe]=np.clip((distance[fringe]-12)/58,0,1)*255
    # Avblanding av grønn bakgrunn i den nye tegningens antialiasrand.
    partial=(alpha>0)&(alpha<255)
    f=alpha[partial,None]/255
    a[partial]=np.clip((a[partial]-actual_bg*(1-f))/f,0,255)
    # Generatoren blander også noen mørke kantpiksler med grønt, selv
    # utenfor den første nøkkelfargeterskelen. Estimer mattens alfa mot
    # nærmeste indre tegnefarge. Dette berører bare en smal ytterrand.
    inside=~flood
    interior=ndimage.binary_erosion(inside,iterations=2)
    if interior.any():
        _,indices=ndimage.distance_transform_edt(~interior,return_indices=True)
        foreground=a[indices[0],indices[1]]
        band=ndimage.distance_transform_edt(inside)<=2.5
        spill=inside & band & (a[:,:,1]>a[:,:,0]+8) & (a[:,:,1]>a[:,:,2]+8)
        v=foreground[spill]-actual_bg
        u=a[spill]-actual_bg
        matte=np.clip(np.sum(u*v,axis=1)/(np.sum(v*v,axis=1)+1e-9),0,1)
        a[spill]=foreground[spill]
        alpha[spill]*=matte
    a[alpha==0]=0
    return Image.fromarray(np.dstack([a,alpha]).round().astype('uint8'))

def bbox(im):
    a=np.array(im)[...,3]>=128
    y,x=np.nonzero(a)
    assert len(x)
    return int(x.min()),int(y.min()),int(x.max()+1),int(y.max()+1)

def fit(im,w,h):
    box=bbox(im)
    left,top,right,bottom=box
    im=im.crop((max(0,left-2),max(0,top-2),min(im.width,right+2),min(im.height,bottom+2)))
    l,t,r,b=bbox(im)
    wanted=(round(im.width*w/(r-l)),round(im.height*h/(b-t)))
    best=None
    for dw in range(-2,3):
        for dh in range(-2,3):
            size=(max(1,wanted[0]+dw),max(1,wanted[1]+dh))
            q=im.convert('RGBa').resize(size,Image.Resampling.LANCZOS).convert('RGBA')
            l,t,r,b=bbox(q)
            score=abs(r-l-w)+abs(b-t-h)
            if best is None or score<best[0]:best=(score,q,(l,t,r,b))
    assert best[0]<=1,(w,h,best[0])
    return best[1],best[2]

selected={}
for name in ['ikon07','ikon08','ikon09']:
    folder=ROOT/'jobber'/name
    job=json.loads((folder/'jobb.json').read_text())
    orig=np.array(Image.open(folder/'original_1x.png').convert('RGB'))
    raw=Image.open(REPORT/(name+'-raa01.png')).convert('RGB')
    canvas=Image.new('RGB',(1536,1024),(128,128,128))
    canvas.paste(tuple(BG.astype(int)),(64,32,1472,992))
    source_hash_to_component={}
    for obj in job['objekter']:
        x,y,w,h=obj['rekt']
        source=orig[y:y+h,x:x+w]
        source_sha=hashlib.sha256(source.tobytes()).hexdigest()
        mask=np.any(source!=job['bakgrunn'],axis=2)
        sy,sx=np.nonzero(mask)
        assert len(sx)
        source_box=[int(sx.min()),int(sy.min()),int(sx.max()+1),int(sy.max()+1)]
        left,top,right,bottom=source_box
        pw,ph=(right-left)*4,(bottom-top)*4
        donor=None
        if source_sha in source_hash_to_component:
            q,box,donor=source_hash_to_component[source_sha]
            origin='byteidentisk kilde, gjenbrukt ny tegning'
        else:
            if obj['bilde'] in corrections:
                entry=corrections[obj['bilde']]
                xx,yy,ww,hh=entry['celle']
                img=Image.open(REPORT/'retting12-raa01.png').crop((xx,yy,xx+ww,yy+hh))
                origin='retting12-raa01.png'
            else:
                # Marginen er bare til den nye antialiasranden, ikke nabobilder.
                img=raw.crop((64+x*4-5,32+y*4-5,64+(x+w)*4+5,32+(y+h)*4+5))
                origin=name+'-raa01.png'
            q,box=fit(clean(img),pw,ph)
            source_hash_to_component[source_sha]=(q,box,obj['bilde'])
        qpath=REPORT/'komponenter'/(obj['bilde']+'.png')
        qpath.parent.mkdir(exist_ok=True)
        q.save(qpath)
        # Plasseres etter den NYE tegningens kjerne, uten originalkonturklipp.
        px,py=64+(x+left)*4-box[0],32+(y+top)*4-box[1]
        canvas.paste(q,(px,py),q)
        if obj['rom'] in (93,94):
            tile=Image.new('RGBA',(w*4,h*4))
            tile.alpha_composite(q,(left*4-box[0],top*4-box[1]))
            kind='verb-og-pekere' if obj['rom']==93 else 'inventar-og-samtale'
            dest=REPORT/'ikoner'/kind/(obj['bilde']+'.png')
            dest.parent.mkdir(parents=True,exist_ok=True)
            tile.save(dest)
            exports.append({'bilde':obj['bilde'],'rom':obj['rom'],'fil':str(dest.relative_to(ROOT)),
                            'sha256':sha(dest),'storrelse':[w*4,h*4]})
        proof.append({'jobb':name,'bilde':obj['bilde'],'rom':obj['rom'],'rekt':obj['rekt'],
                      'kilde_synlig':source_box,'ny_kjerne':list(box),'plassering':[px,py],
                      'raa':origin,'donor':donor,'kilde_RGB_sha256':source_sha,
                      'komponent':str(qpath.relative_to(ROOT)),'sha256':sha(qpath)})
    canvas.save(folder/'resultat.png')
    selected[name]={'sha256':sha(folder/'resultat.png'),'objektbilder':len(job['objekter'])}

(REPORT/'komponentkart.json').write_text(json.dumps(proof,indent=2)+'\n')
(REPORT/'ikonmanifest.json').write_text(json.dumps(exports,indent=2)+'\n')
(REPORT/'leveransehash.json').write_text(json.dumps(selected,indent=2)+'\n')
assert len(exports)==111

    preview.alpha_composite(im,(i*192+(192-im.width)//2,30+(150-im.height)//2))
    draw.text((i*192+55,205),name,fill='white')
preview.convert('RGB').save(REPORT/'verbhjul-forhandsvisning.png')

print(json.dumps(selected))
