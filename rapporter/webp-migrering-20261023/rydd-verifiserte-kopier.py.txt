from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
import os, json, hashlib, gzip, time, subprocess
P=Path('/home/tom/Sam-Max-HD'); O=P/'out/webp-migrering-20261023'; SOURCE=P/'mods/gpt'
KEEP={'webp-migrering-20261023','helikopter178-retur2001og2000-20261023','felles-release-20261023-helikopter','formatproeve-webp-avif-20261023'}
SHA='2ba07d9cb94c99053fd0e97b0dfab903f3b01aff'
def digest(p):
 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def free():return os.statvfs(P).f_bavail*os.statvfs(P).f_frsize
inventory={}
for p in SOURCE.rglob('*'):
 if p.is_file() and p.name not in {'README.md','mod.json','SHA256SUMS'}:
  inventory[p.relative_to(SOURCE).as_posix()]={'size':p.stat().st_size,'sha256':digest(p)}
(O/'beholdt-png-grunnlag.json').write_text(json.dumps({'remote_sha':SHA,'restore':'git archive '+SHA,'files':inventory},indent=2)+'\n')
roots=[]
for base,ds,fs in os.walk(P/'out',followlinks=False):
 path=Path(base); rel=path.relative_to(P/'out')
 ds[:]=[d for d in ds if d not in {'.git','__pycache__'} and not (path==P/'out' and d in KEEP)]
 if 'mod.json' in fs and 'costumes' in ds:
  try:m=json.loads((path/'mod.json').read_text())
  except (OSError,ValueError):continue
  if m.get('format')==1 and m.get('scale')==4:
   roots.append(path);ds.clear()
(O/'ryddekandidater.json').write_text(json.dumps({'roots':[str(p)for p in roots], 'excluded_top':sorted(KEEP)},indent=2)+'\n')
summary={'start':time.time(),'free_before':free(),'roots':len(roots),'removed':0,'bytes_logical':0,'kept_different':0,'restore_commit':SHA,'unique_images_removed':False,'stashes_before':subprocess.check_output(['git','-C',str(P),'stash','list','--format=%H'],text=True).splitlines()}
print('Found',len(roots),'old mod copies. Verifying all bytes before removing exact duplicates.',flush=True)
# Every removed file matches a retained current PNG and an immutable published commit.
def process(root):
 rows=[]; kept=0; total=0
 for rel,expected in inventory.items():
  p=root/rel
  if not p.is_file() or p.is_symlink():continue
  st=p.stat()
  if st.st_size==expected['size'] and digest(p)==expected['sha256']:
   # Compare the source again only once per root through its unchanged inode snapshot below.
   p.unlink();rows.append([str(p.relative_to(P)),expected['sha256'],st.st_size]);total+=st.st_size
  else:kept+=1
 return root,rows,kept,total
with gzip.open(O/'slettede-duplikater.jsonl.gz','wt',compresslevel=6) as log:
 with ThreadPoolExecutor(max_workers=2) as pool:
  for i,(root,rows,kept,total) in enumerate(pool.map(process,roots),1):
   for row in rows:log.write(json.dumps(row,separators=(',',':'))+'\n')
   summary['removed']+=len(rows);summary['bytes_logical']+=total;summary['kept_different']+=kept
   summary['roots_done']=i;summary['free_now']=free()
   (O/'rydding-status.json').write_text(json.dumps(summary,indent=2)+'\n')
   print(i,'/',len(roots),root.name,'removed',len(rows),'total GiB',round(summary['bytes_logical']/2**30,2),flush=True)
summary.update(end=time.time(),free_after=free(),manifest_sha256=digest(O/'slettede-duplikater.jsonl.gz'))
summary['stashes_after']=subprocess.check_output(['git','-C',str(P),'stash','list','--format=%H'],text=True).splitlines()
assert summary['stashes_before']==summary['stashes_after']
(O/'rydding-status.json').write_text(json.dumps(summary,indent=2)+'\n')
print('DONE',json.dumps(summary),flush=True)
