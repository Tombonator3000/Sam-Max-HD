def clean_key(im,key):
 a=np.array(im.convert('RGB'),float);key=np.array(key,float)
 dist=np.linalg.norm(a-key,axis=2);near=dist<70
 lab,_=ndimage.label(near);edge=np.unique(np.concatenate([lab[0],lab[-1],lab[:,0],lab[:,-1]]));flood=np.isin(lab,edge[edge>0])
 alpha=np.full(a.shape[:2],255.);fringe=flood & ndimage.maximum_filter(~flood,size=3);alpha[flood]=0;alpha[fringe]=np.clip((dist[fringe]-12)/58,0,1)*255
 part=(alpha>0)&(alpha<255);f=alpha[part,None]/255;a[part]=np.clip((a[part]-key*(1-f))/f,0,255);a[alpha==0]=0
 return Image.fromarray(np.dstack([a,alpha]).round().astype('uint8'))

def core(im):
 y,x=np.nonzero(np.array(im)[...,3]>=128);assert len(x)
 return int(x.min()),int(y.min()),int(x.max()+1),int(y.max()+1)

def fit(im,box,size):
 # Kun synlig boks, aldri originalkontur, styrer skalering/plassering.
 l,t,r,b=core(im);q=im.crop((max(0,l-2),max(0,t-2),min(im.width,r+2),min(im.height,b+2)))
 l,t,r,b=core(q);left,top,right,bottom=box;w,h=right-left,bottom-top
 target=(round(q.width*w/(r-l)),round(q.height*h/(b-t)))
 best=None
 for dx in range(-2,3):
  for dy in range(-2,3):
   z=q.convert('RGBa').resize((max(1,target[0]+dx),max(1,target[1]+dy)),Image.Resampling.LANCZOS).convert('RGBA');a,c,d,e=core(z);score=abs(d-a-w)+abs(e-c-h)
   if best is None or score<best[0]:best=(score,z,(a,c,d,e))
 assert best[0]<=1,(w,h,best[0]);_,z,(a,c,d,e)=best
 out=Image.new('RGBA',size);out.alpha_composite(z,(left-a,top-c));return out
