from pathlib import Path
from PIL import Image
from collections import Counter
import numpy as np,json,hashlib,shutil

P=Path('/home/tom/Sam-Max-HD')
O=Path(__file__).resolve().parent
R=P/'work/.gpt-gren/rapporter'/O.name
read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
motor=read(R/'kilde220-start1-y60.json')
assert motor['exit']==0 and motor['miljo']['DIGHD_TEST_POS']=='220,60'
assert motor['binar_sha256']=='651aa78c84411784671f600085147bbbdce5dc6281e77bd113eb7db7f1c0ed27'
rows=[]
for n in list(range(1986,1995))+[1995]:
    event=motor['tegninger'][f'costume178_{n}']
    x,y=event['pos'];w,h=event['maal']
    assert x>=0 and y>=0 and x+w<=320 and y+h<=200
    native=Path(event['fil'])
    flat=native.with_name(native.stem+'_flat.png')
    idxfile=P/f'mods/gpt/costumes/costume178_{n}_idx.png'
    idximage=Image.open(idxfile)
    idx=np.array(idximage)
    assert idx.shape==(h,w) and idximage.info['transparency']==255 and not np.any(idx==13)
    a=np.array(Image.open(flat).convert('RGBA'))
    rgb=a[y*4+2:(y+h)*4:4,x*4+2:(x+w)*4:4,:3]
    on=idx!=255
    assert rgb.shape==(h,w,3)
    # Hele den faktiske originalblokken skal ha samme palettfarge.
    block=a[y*4:(y+h)*4,x*4:(x+w)*4,:3].reshape(h,4,w,4,3)
    assert not np.any(np.any(block!=rgb[:,None,:,None,:],axis=(1,3,4))&on)
    for value in np.unique(idx[on]):
        assert len(np.unique(rgb[idx==value],axis=0))==1,(n,int(value),'overtegnet originalindeks')
    # Originalpaletten er kanonisk. SDLs RGB565-visning ekspanderer bitene.
    # Målte skjerm-RGB skal ikke feilaktig bli en ny, endret tegnepalett.
    source=np.array(idximage.convert('RGBA'))
    source[~on]=0
    bits=np.array([3,2,3],dtype='uint8')
    quant=source[:,:,:3]>>bits
    expanded=(quant<<bits)|(quant>>(np.array([8,8,8],dtype='uint8')-2*bits))
    assert np.array_equal(expanded[on],rgb[on]),(n,'RGB565-pallettavvik')
    if n==1995:
        assert np.array_equal(source,np.array(Image.open(R/'kilder/1995/original-1x.png').convert('RGBA')))
        continue
    S=R/'kilder-neste'/str(n)
    S.mkdir(parents=True,exist_ok=False)
    Image.fromarray(source).save(S/'original-1x.png')
    shutil.copyfile(idxfile,S/'original-indeks.png')
    shutil.copyfile(native,S/'hel-motor.png')
    shutil.copyfile(flat,S/'hel-originalmotor.png')
    bg=Image.new('RGBA',(w,h),'#87959e')
    bg.alpha_composite(Image.fromarray(source))
    bg.resize((w*24,h*24),Image.Resampling.NEAREST).convert('RGB').save(S/'referanse24x.png')
    record=dict(rute=n,maal_1x=[w,h],opaque=int(on.sum()),frame=event['bildeframe'],pos=[x,y],
        original_RGBA_SHA256=sha(S/'original-1x.png'),indeks_SHA256=sha(S/'original-indeks.png'),
        hel_motor_SHA256=sha(S/'hel-motor.png'),hel_originalmotor_SHA256=sha(S/'hel-originalmotor.png'),
        alle_originalblokker_byteidentiske=True,ingen_indeks_med_blandet_farge=True,
        originalmotor_RGB565_avvik=0,kanonisk_indekspalett_bevart=True,
        hel_synlig=True,source_only=True,motorkode_og_spillskript_uendret=True,
        kontrollert_aktorpos220_60=True,ikke_naturlig_full_gjennomspilling=True,
        ingen_originale_jobbfiler_eller_gamle_kilder_endret=True,visuelt_sett=False,
        farger=[list(c)+[count]for c,count in sorted(Counter(map(tuple,source[on,:3].tolist())).items())])
    (S/'kilde.json').write_text(json.dumps(record,ensure_ascii=False,indent=2)+'\n')
    rows.append(record)
proof=dict(kildefaser_logget=len(motor['tegninger']),nye_hele_kilder=rows,
    tidligere_1995_kilde_RGBA_byteeksakt_gjenfunnet=True,fase1990_hel_kilde_na_tilgjengelig=True,
    kilder_brukes_bare_til_referanse_og_kontroll=True,ny_grafikk_ikke_laget=True,neste_kall_ikke_utfort='H176')
(R/'helkilder1986til1994.json').write_text(json.dumps(proof,ensure_ascii=False,indent=2)+'\n')
(R/'helkilde-1986til1994.py.txt').write_bytes(Path(__file__).read_bytes())
print('Ni hele originalkilder, også1990.97 kildefaser logget.1995 byteeksakt gjenfunnet.')
