def loops(mask):

def rdp(p,epsilon):
 keep={0,len(p)-1};stack=[(0,len(p)-1)]
 while stack:
  lo,hi=stack.pop()
  if hi-lo<=1:continue
  a,b=p[lo],p[hi];v=b-a;den=float(v@v)
  if den: t=np.clip(((p[lo+1:hi]-a)@v)/den,0,1);d=np.linalg.norm(p[lo+1:hi]-a-t[:,None]*v,axis=1)
  else:d=np.linalg.norm(p[lo+1:hi]-a,axis=1)
  j=int(np.argmax(d))+lo+1
  if d[j-lo-1]>epsilon:keep.add(j);stack.extend([(lo,j),(j,hi)])
 return p[sorted(keep)]
